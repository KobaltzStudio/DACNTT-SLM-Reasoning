def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path) -> dict:
    value = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def read_jsonl(path: Path) -> list[dict]:
    rows = []
    with path.open("r", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, start=1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSONL at {path}:{line_number}: {exc}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected JSON object at {path}:{line_number}"
                )
            rows.append(row)

    return rows


def write_temp_exclusive(path: Path, content: bytes) -> Path:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp_path = path.with_name(
        path.name + f".{os.getpid()}.tmp"
    )

    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
    descriptor = os.open(str(temp_path), flags, 0o600)

    try:
        with os.fdopen(descriptor, "wb") as handle:
            handle.write(content)
            handle.flush()
            os.fsync(handle.fileno())
    except Exception:
        try:
            temp_path.unlink()
        except OSError:
            pass
        raise

    return temp_path


def publish_pair_no_overwrite(
    train_set_path: Path,
    selection_log_path: Path,
    train_set_bytes: bytes,
    selection_log_bytes: bytes,
) -> None:
    if train_set_path.exists():
        raise FileExistsError(
            f"Refusing to overwrite existing file: {train_set_path}"
        )

    if selection_log_path.exists():
        raise FileExistsError(
            f"Refusing to overwrite existing file: {selection_log_path}"
        )

    train_temp = write_temp_exclusive(train_set_path, train_set_bytes)
    log_temp = None

    try:
        log_temp = write_temp_exclusive(
            selection_log_path,
            selection_log_bytes,
        )

        if train_set_path.exists() or selection_log_path.exists():
            raise FileExistsError(
                "An output appeared during publication; refusing to overwrite."
            )

        os.rename(train_temp, train_set_path)
        os.rename(log_temp, selection_log_path)

    except Exception:
        for temp_path in (train_temp, log_temp):
            if temp_path is not None:
                try:
                    temp_path.unlink(missing_ok=True)
                except OSError:
                    pass
        raise


def load_pool(
    pool_path: Path,
    d_pool_path: Path,
) -> tuple[dict[str, dict], dict[str, str]]:
    rows = read_jsonl(pool_path)
    d_pool_ids = {
        line.strip()
        for line in d_pool_path.read_text(encoding="utf-8").splitlines()
        if line.strip()
    }

    pool_by_id: dict[str, dict] = {}
    problems: dict[str, str] = {}

    for index, row in enumerate(rows, start=1):
        tid = row.get("trajectory_id")
        pid = row.get("problem_id")

        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Pool row {index}: invalid trajectory_id")

        if not isinstance(pid, str) or not pid:
            raise ValueError(f"Pool row {index}: invalid problem_id")

        if tid in pool_by_id:
            raise ValueError(f"Duplicate trajectory_id in pool: {tid}")

        metadata = row.get("metadata")
        if not isinstance(metadata, dict):
            raise ValueError(f"{tid}: metadata is missing or invalid")

        if metadata.get("verification_status") != "valid":
            raise ValueError(f"{tid}: verification_status is not valid")

        capabilities = row.get("capabilities")
        if not isinstance(capabilities, list) or not capabilities:
            raise ValueError(f"{tid}: capability list is empty or invalid")

        if len(capabilities) != len(set(capabilities)):
            raise ValueError(f"{tid}: duplicate capability tags")

        unknown = sorted(set(capabilities) - set(TAXONOMY))
        if unknown:
            raise ValueError(f"{tid}: unknown capability tags: {unknown}")

        if pid not in d_pool_ids:
            raise ValueError(f"{tid}: problem_id is outside D_pool: {pid}")

        pool_by_id[tid] = row
        problems[tid] = pid

    if not pool_by_id:
        raise ValueError("Verified pool is empty")

    return pool_by_id, problems


def load_tokens(
    tokens_path: Path,
    pool_path: Path,
    problems: dict[str, str],
) -> dict[str, int]:
    summary_path = tokens_path.with_suffix(".summary.json")

    if not summary_path.is_file():
        raise FileNotFoundError(
            f"Token summary not found: {summary_path}"
        )

    summary = read_json(summary_path)
    actual_pool_hash = sha256_file(pool_path)

    if summary.get("pool_sha256") != actual_pool_hash:
        raise ValueError(
            "Token summary pool_sha256 does not match the current pool."
        )

    rows = read_jsonl(tokens_path)
    tokens: dict[str, int] = {}

    for index, row in enumerate(rows, start=1):
        tid = row.get("trajectory_id")
        pid = row.get("problem_id")
        count = row.get("response_tokens")

        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Token row {index}: invalid trajectory_id")

        if tid in tokens:
            raise ValueError(f"Duplicate token row: {tid}")

        if tid not in problems:
            raise ValueError(f"Token row not present in verified pool: {tid}")

        if problems[tid] != pid:
            raise ValueError(f"Token problem_id mismatch: {tid}")

        if not isinstance(count, int) or isinstance(count, bool) or count <= 0:
            raise ValueError(f"Invalid response_tokens for {tid}: {count}")

        tokens[tid] = count

    if set(tokens) != set(problems):
        missing = sorted(set(problems) - set(tokens))
        extra = sorted(set(tokens) - set(problems))
        raise ValueError(
            f"Pool/token IDs differ. Missing={missing[:5]}, extra={extra[:5]}"
        )

    token_sum = sum(tokens.values())

    if len(rows) != summary.get("n_trajectories"):
        raise ValueError("Token summary n_trajectories mismatch")

    if token_sum != summary.get("response_tokens_sum"):
        raise ValueError("Token summary response_tokens_sum mismatch")

    return tokens


def load_capability_deficits(state_path: Path, method: str, seed: int, round_number: int) -> dict[str, float]:
    state = read_json(state_path)

    if state.get("schema_version") != "p1p2.capability-state.v1":
        raise ValueError("Unsupported capability state schema")

    target = state.get("target", {})

    if target.get("method") != method:
        raise ValueError(
            f"State target method mismatch: expected {method}, got {target.get('method')}"
        )

    if target.get("seed") != seed:
        raise ValueError(
            f"State target seed mismatch: expected {seed}, got {target.get('seed')}"
        )

    if target.get("selection_round") != round_number:
        raise ValueError(
            "State selection_round mismatch: "
            f"expected {round_number}, got {target.get('selection_round')}"
        )

    capabilities = state.get("capabilities")
    if not isinstance(capabilities, dict):
        raise ValueError("Capability state has no capabilities object")

    deficits = {}

    for capability in TAXONOMY:
        item = capabilities.get(capability)
        if not isinstance(item, dict):
            raise ValueError(f"Missing capability state: {capability}")

        value = item.get("deficit")
        if (
            not isinstance(value, (int, float))
            or isinstance(value, bool)
            or not math.isfinite(value)
            or not 0.0 <= value <= 1.0
        ):
            raise ValueError(f"Invalid deficit for {capability}: {value}")

        deficits[capability] = float(value)

    return deficits


def load_previous_train_ids(paths: list[Path], problems: dict[str, str]) -> list[str]:
    trained = set()

    for path in paths:
        data = read_json(path)
        ids = data.get("trajectory_ids")

        if not isinstance(ids, list):
            raise ValueError(f"Invalid trajectory_ids in {path}")

        for tid in ids:
            if not isinstance(tid, str) or tid not in problems:
                raise ValueError(
                    f"Previous train set contains unknown trajectory: {tid}"
                )
            trained.add(tid)

    return sorted(trained)


def load_nll(
    nll_path: Path,
    tokens: dict[str, int],
    source_checkpoint: str,
) -> tuple[dict[str, float], str]:
    rows = read_jsonl(nll_path)
    ell_by_id = {}

    for index, row in enumerate(rows, start=1):
        tid = row.get("trajectory_id")

        if not isinstance(tid, str) or not tid:
            raise ValueError(f"NLL row {index}: invalid trajectory_id")

        if tid in ell_by_id:
            raise ValueError(f"Duplicate NLL trajectory_id: {tid}")

        if tid not in tokens:
            raise ValueError(f"NLL trajectory not present in token table: {tid}")

        if row.get("checkpoint") != source_checkpoint:
            raise ValueError(
                f"NLL checkpoint mismatch for {tid}: "
                f"expected {source_checkpoint}, got {row.get('checkpoint')}"
            )

        if row.get("n_tokens") != tokens[tid]:
            raise ValueError(f"NLL token count mismatch for {tid}")

        ell = row.get("ell")
        if (
            not isinstance(ell, (int, float))
            or isinstance(ell, bool)
            or not math.isfinite(ell)
        ):
            raise ValueError(f"Invalid ell value for {tid}: {ell}")

        ell_by_id[tid] = float(ell)

    if set(ell_by_id) != set(tokens):
        missing = sorted(set(tokens) - set(ell_by_id))
        extra = sorted(set(ell_by_id) - set(tokens))
        raise ValueError(
            f"NLL does not cover the token table. "
            f"Missing={missing[:5]}, extra={extra[:5]}"
        )

    summary_path = nll_path.with_suffix(".summary.json")
    if not summary_path.is_file():
        raise FileNotFoundError(f"NLL summary not found: {summary_path}")

    summary = read_json(summary_path)

    if summary.get("checkpoint") != source_checkpoint:
        raise ValueError("NLL summary checkpoint mismatch")

    if summary.get("pool_sha256") != sha256_file(POOL_DEFAULT):
        raise ValueError("NLL summary pool_sha256 mismatch")

    if summary.get("n_trajectories") != len(tokens):
        raise ValueError("NLL summary trajectory count mismatch")

    return ell_by_id, sha256_file(nll_path)


def average_percentile_ranks(scores: dict[str, float]) -> dict[str, float]:
    if not scores:
        raise ValueError("Cannot rank an empty score mapping")

    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    n = len(ordered)

    if n == 1:
        return {ordered[0][0]: 0.5}

    result = {}
    index = 0

    while index < n:
        end = index + 1
        value = ordered[index][1]

        while end < n and ordered[end][1] == value:
            end += 1

        average_rank = ((index + 1) + end) / 2.0
        percentile = (average_rank - 1.0) / (n - 1.0)

        for position in range(index, end):
            result[ordered[position][0]] = percentile

        index = end

    return result


def calculate_p1_scores(
    pool_by_id: dict[str, dict],
    deficits: dict[str, float],
) -> dict[str, float]:
    result = {}

    for tid, row in pool_by_id.items():
        tags = row["capabilities"]
        result[tid] = sum(deficits[tag] for tag in tags) / len(tags)

    return result

#!/usr/bin/env python3
"""
Selection rules of B2, B3, B4, B5 and the shuffled-scores ablation (analysis_plan.md section 5,
task7_decisions.md T7-1, T7-2, T7-3). Pure Python: no torch, no files; select_round.py does the I/O.

Budget filling (protocol section 16.1, tokens(B_t) <= beta_round): walk the order and stop at the first
trajectory that would exceed the budget. Later, shorter trajectories are not used to top up, because that
would favour short trajectories over the method's own order. B3 and B4 keep the stopping trajectory for
the next round.

Tokens are response tokens under the Student tokenizer without the end-of-turn token, the unit of L_bar
and therefore of beta_round.

Seeds fixed by the config: B2 random.Random(training_seed * 100 + round); B3 random.Random(training_seed).
The ablation's seed is not in the config; ABLATION_SEED_OFFSET keeps it apart from B2.
"""
import random
from collections import Counter

from learnability import score_checkpoint

METHODS = ("B2", "B3", "B4", "B5", "ablation_shuffled_scores")
NEEDS_NLL = ("B4", "B5", "ablation_shuffled_scores")
ABLATION_SEED_OFFSET = 50
SELECTED_CRITERION = {"B2": "random", "B3": "shortest_correct", "B4": "top_score", "B5": "top_score",
                      "ablation_shuffled_scores": "shuffled_score"}


def fill_budget(order, problems, tokens, budget, n_cap, unavailable=()):
    """Returns (selected ids in order, {id: reason}, tokens used, whether the budget stopped the walk)."""
    unavailable = set(unavailable)
    selected, reasons, per_problem = [], {}, Counter()
    used, stopped = 0, False
    for tid in order:
        if tid in unavailable:
            reasons[tid] = "already_trained"
        elif stopped:
            reasons[tid] = "budget_exhausted"
        elif per_problem[problems[tid]] >= n_cap:
            reasons[tid] = "n_cap_reached"
        elif used + tokens[tid] > budget:
            stopped = True
            reasons[tid] = "budget_exhausted"
        else:
            selected.append(tid)
            reasons[tid] = "selected"
            per_problem[problems[tid]] += 1
            used += tokens[tid]
    return selected, reasons, used, stopped


def rank_by_score(scores):
    """Descending score, ties by trajectory_id ascending (decision #23)."""
    return sorted(scores, key=lambda tid: (-scores[tid], tid))


def random_order(ids, seed):
    order = sorted(ids)
    random.Random(seed).shuffle(order)
    return order


def shuffled_scores(scores, seed):
    """The same score values assigned to trajectories at random."""
    ids = sorted(scores)
    values = [scores[tid] for tid in ids]
    random.Random(seed).shuffle(values)
    return dict(zip(ids, values))


def shortest_per_problem(problems, tokens):
    """Decision #24: per problem the shortest trajectory, ties by smallest trajectory_id."""
    best = {}
    for tid in sorted(problems):
        pid = problems[tid]
        if pid not in best or tokens[tid] < tokens[best[pid]]:
            best[pid] = tid
    return best


def b3_order(problems, tokens, training_seed):
    """B3 order: the shortest trajectory of each problem, problems shuffled once [T7-2]."""
    best = shortest_per_problem(problems, tokens)
    return [best[pid] for pid in random_order(best, training_seed)]


def static_round(order, problems, tokens, budget, n_cap, round_index):
    """
    Round round_index of a once-computed order (B3, B4): walk it without repeating a trajectory; a
    trajectory skipped by n_cap or by the budget stays available for later rounds [T7-1, T7-2].
    """
    used_before = set()
    for r in range(round_index + 1):
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap, used_before)
        if r < round_index:
            used_before |= set(selected)
    return selected, reasons, used, stopped


def select(method, round_index, training_seed, problems, tokens, budget, n_cap, ell_by_id=None, trained_ids=()):
    """
    One round's selection over the eligible pool. Returns the selected ids, tokens used, band parameters
    and one log entry per eligible trajectory (fields of selection_log_schema.json, null where unused).
    ell_by_id: ell at S_t for B5 and the ablation, at S_0 for B4 in every round.
    trained_ids: trained in this run before this round (reference set of B5 and the ablation [T7-3]).
    """
    if method not in METHODS:
        raise ValueError(f"unknown method {method}")
    scoring, scores = None, None
    if method in NEEDS_NLL:
        if ell_by_id is None or set(ell_by_id) != set(problems):
            raise ValueError(f"{method} needs ell for exactly the eligible pool")
        # B4 ranks at S_0, where nothing has been trained: the reference set is the whole eligible pool.
        scoring = score_checkpoint(ell_by_id, () if method == "B4" else trained_ids)
        scores = scoring["l_proxy"]

    if method == "B2":
        order = random_order(problems, training_seed * 100 + round_index)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    elif method == "B3":
        order = b3_order(problems, tokens, training_seed)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    elif method == "B4":
        order = rank_by_score(scores)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    else:
        if method == "ablation_shuffled_scores":
            scores = shuffled_scores(scores, training_seed * 100 + round_index + ABLATION_SEED_OFFSET)
        order = rank_by_score(scores)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    if not stopped:
        raise ValueError(f"{method} round {round_index}: order exhausted before the budget ({used}/{budget} tokens)")

    ranks = {tid: i for i, tid in enumerate(order, start=1)}
    has_rank = method != "B2"
    rows = []
    for tid in sorted(problems):
        reason = reasons.get(tid, "not_shortest")
        rows.append({
            "problem_id": problems[tid],
            "trajectory_id": tid,
            "tokens": tokens[tid],
            "nll": ell_by_id[tid] if scoring else None,
            "ell_star": scoring["ell_star"] if scoring else None,
            "sigma": scoring["sigma"] if scoring else None,
            "l_proxy": scoring["l_proxy"][tid] if scoring else None,
            "score": scores[tid] if scores else None,
            "rank": ranks.get(tid) if has_rank else None,
            "selected": reason == "selected",
            "criterion": SELECTED_CRITERION[method] if reason == "selected" else reason,
            "outcome": None,
        })
    return {
        "selected": selected,
        "tokens_used": used,
        "ell_star": scoring["ell_star"] if scoring else None,
        "sigma": scoring["sigma"] if scoring else None,
        "n_reference": scoring["n_reference"] if scoring else None,
        "rows": rows,
    }


def probe_groups(tokens, group_size, seed, n_bins=5):
    """
    L_emp groups (config budget.l_emp_groups): token-length quintile bins of the eligible pool, group_size
    trajectories sampled per bin. n_cap is not applied: the rule does not ask for it.
    """
    ordered = sorted(tokens, key=lambda tid: (tokens[tid], tid))
    bins = [ordered[i * len(ordered) // n_bins: (i + 1) * len(ordered) // n_bins] for i in range(n_bins)]
    rng = random.Random(seed)
    return [sorted(rng.sample(b, group_size)) for b in bins]

#!/usr/bin/env python3
"""
L_proxy (analysis_plan.md section 4, config "learnability"). Pure Python: no torch, no files.

  ell*_t  = median of ell over the reference set at S_t
  sigma_t = IQR of ell over the reference set at S_t / 1.349
  L_proxy = exp(-(ell - ell*_t)^2 / (2 sigma_t^2)), computed for every eligible trajectory

Reference set [T7-3]: eligible trajectories not yet trained in this run (at S_0: all eligible).

Quartiles use statistics.quantiles(n=4) (method "exclusive"), the same call as the Task 5 smoke analysis
(analyze_smoke_test.py). The choice cannot change the ranking: L_proxy order depends only on |ell - ell*_t|.
"""
import math
import statistics


def band_parameters(ells):
    """(ell_star, sigma) of a list of ell values."""
    if len(ells) < 2:
        raise ValueError(f"reference set too small: {len(ells)}")
    if not all(math.isfinite(x) for x in ells):
        raise ValueError("non-finite ell in the reference set")
    q = statistics.quantiles(ells, n=4)
    sigma = (q[2] - q[0]) / 1.349
    if sigma <= 0:
        raise ValueError("IQR of ell is zero: L_proxy is undefined")
    return statistics.median(ells), sigma


def l_proxy(ell, ell_star, sigma):
    return math.exp(-((ell - ell_star) ** 2) / (2 * sigma ** 2))


def score_checkpoint(ell_by_id, trained_ids=()):
    """
    L_proxy of every trajectory in ell_by_id (the whole eligible pool at one checkpoint).
    trained_ids: trajectories already trained in this run; excluded from the reference set only.
    """
    unknown = set(trained_ids) - set(ell_by_id)
    if unknown:
        raise ValueError(f"trained trajectories missing from the scored pool: {sorted(unknown)[:3]}")
    trained = set(trained_ids)
    reference = [ell for tid, ell in ell_by_id.items() if tid not in trained]
    ell_star, sigma = band_parameters(reference)
    return {
        "ell_star": ell_star,
        "sigma": sigma,
        "n_reference": len(reference),
        "l_proxy": {tid: l_proxy(ell, ell_star, sigma) for tid, ell in ell_by_id.items()},
    }
