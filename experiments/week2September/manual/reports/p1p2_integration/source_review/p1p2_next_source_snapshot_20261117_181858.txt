def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish a completed staging directory without overwriting a target.

    Retries transient Windows PermissionError failures. On terminal failure,
    the staging directory is intentionally preserved for recovery.
    """
    retry_delays = (0.25, 0.5, 1.0, 2.0, 4.0)

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)
            return
        except PermissionError as exc:
            if out_dir.exists():
                raise FileExistsError(
                    "OUTPUT_COLLISION: destination appeared during "
                    "publication; refusing to overwrite: " + str(out_dir)
                ) from exc

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)

def hash_path(path: Path) -> str:
    path = path.resolve()

    if path.is_file():
        return sha256_file(path)

    if not path.is_dir():
        raise FileNotFoundError(f"Input does not exist: {path}")

    files = sorted(p for p in path.rglob("*") if p.is_file())
    if not files:
        raise ValueError(f"Checkpoint directory is empty: {path}")

    digest = hashlib.sha256()
    for file_path in files:
        relative = file_path.relative_to(path).as_posix()
        digest.update(relative.encode("utf-8"))
        digest.update(b"\0")
        digest.update(sha256_file(file_path).encode("ascii"))
        digest.update(b"\n")

    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:

def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []

    with path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSONL at {path}:{line_no}: {exc}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected JSON object at {path}:{line_no}"
                )
            rows.append(row)

    return rows


def read_id_file(path: Path) -> set[str]:
    values = [
        line.strip()
        for line in path.read_text(encoding="utf-8-sig").splitlines()
        if line.strip()
    ]

    if len(values) != len(set(values)):
        raise ValueError(f"Duplicate IDs in {path}")

    return set(values)


def wilson_interval(
    correct: int,
    support: int,
    z: float = 1.959963984540054,
) -> tuple[float | None, float | None]:
    if support <= 0:
        return None, None

    p = correct / support
    z2 = z * z
    denominator = 1.0 + z2 / support
    center = (p + z2 / (2.0 * support)) / denominator
    margin = (
        z
        * (
            p * (1.0 - p) / support
            + z2 / (4.0 * support * support)
        ) ** 0.5
        / denominator
    )

    return max(0.0, center - margin), min(1.0, center + margin)


def strict_correct(row: dict[str, Any]) -> bool:
    """P2 primary strict metric: only status == 'valid' is correct."""
    status = row.get("status")

    if status not in {"valid", "invalid_answer", "invalid_format"}:
        raise ValueError(
            "Prediction must contain a recognized P2 status; "
            f"got status={status!r}, problem_id={row.get('problem_id')!r}"
        )

    return status == "valid"

        extra = sorted(approval_ids - gold_ids)
        raise ValueError(
            "Approval/Gold ID set mismatch: "
            f"missing_approval={missing[:10]}, "
            f"extra_approval={extra[:10]}"
        )

    for problem_id, row in approval_by_id.items():
        review_status = str(row.get("review_status", "")).strip()
        human_approval = str(row.get("human_approval", "")).strip()

        approved_review_statuses = {

def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish without overwrite; retry transient Windows permission failures."""
    retry_delays = (0.25, 0.5, 1.0, 2.0, 4.0)

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)
            return
        except PermissionError as exc:
            if out_dir.exists():
                raise FileExistsError(
                    "OUTPUT_COLLISION: destination appeared during "
                    "publication; refusing to overwrite: " + str(out_dir)
                ) from exc

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)


def main() -> None:
    args = parse_args()

    input_hashes = build_input_hashes(args)
    out_dir = Path(args.out_root) / args.run_id

    if out_dir.exists():
        verify_existing_sidecar(out_dir, input_hashes)
        return

    out_dir.parent.mkdir(parents=True, exist_ok=True)

    stage = Path(tempfile.mkdtemp(
        prefix=f".{args.run_id}.staging.",
        dir=out_dir.parent,
    ))

    try:
        build_sidecar(args, stage, input_hashes)
        publish_staged_directory(stage, out_dir)
    except Exception as exc:
        if stage.exists():
            print(f"RECOVERY_STAGING_PRESERVED={stage}")
            print(f"RECOVERY_ERROR={type(exc).__name__}: {exc}")
        raise

    print(f"CAPABILITY_EVAL: PASS {out_dir}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Selection rules of B2, B3, B4, B5 and the shuffled-scores ablation (analysis_plan.md section 5,
task7_decisions.md T7-1, T7-2, T7-3). Pure Python: no torch, no files; select_round.py does the I/O.

Budget filling (protocol section 16.1, tokens(B_t) <= beta_round): walk the order and stop at the first
trajectory that would exceed the budget. Later, shorter trajectories are not used to top up, because that
would favour short trajectories over the method's own order. B3 and B4 keep the stopping trajectory for
the next round.

Tokens are response tokens under the Student tokenizer without the end-of-turn token, the unit of L_bar
and therefore of beta_round.

Seeds fixed by the config: B2 random.Random(training_seed * 100 + round); B3 random.Random(training_seed).
The ablation's seed is not in the config; ABLATION_SEED_OFFSET keeps it apart from B2.
"""
import random
from collections import Counter

from learnability import score_checkpoint

METHODS = ("B2", "B3", "B4", "B5", "ablation_shuffled_scores")
NEEDS_NLL = ("B4", "B5", "ablation_shuffled_scores")
ABLATION_SEED_OFFSET = 50
SELECTED_CRITERION = {"B2": "random", "B3": "shortest_correct", "B4": "top_score", "B5": "top_score",
                      "ablation_shuffled_scores": "shuffled_score"}


def fill_budget(order, problems, tokens, budget, n_cap, unavailable=()):
    """Returns (selected ids in order, {id: reason}, tokens used, whether the budget stopped the walk)."""
    unavailable = set(unavailable)
    selected, reasons, per_problem = [], {}, Counter()
    used, stopped = 0, False
    for tid in order:
        if tid in unavailable:
            reasons[tid] = "already_trained"
        elif stopped:
            reasons[tid] = "budget_exhausted"
        elif per_problem[problems[tid]] >= n_cap:
            reasons[tid] = "n_cap_reached"
        elif used + tokens[tid] > budget:
            stopped = True
            reasons[tid] = "budget_exhausted"
        else:
            selected.append(tid)
            reasons[tid] = "selected"
            per_problem[problems[tid]] += 1
            used += tokens[tid]
    return selected, reasons, used, stopped


def rank_by_score(scores):
    """Descending score, ties by trajectory_id ascending (decision #23)."""
    return sorted(scores, key=lambda tid: (-scores[tid], tid))


def random_order(ids, seed):
    order = sorted(ids)
    random.Random(seed).shuffle(order)
    return order


def shuffled_scores(scores, seed):
    """The same score values assigned to trajectories at random."""
    ids = sorted(scores)
    values = [scores[tid] for tid in ids]
    random.Random(seed).shuffle(values)
    return dict(zip(ids, values))


def shortest_per_problem(problems, tokens):
    """Decision #24: per problem the shortest trajectory, ties by smallest trajectory_id."""
    best = {}
    for tid in sorted(problems):
        pid = problems[tid]
        if pid not in best or tokens[tid] < tokens[best[pid]]:
            best[pid] = tid
    return best


def b3_order(problems, tokens, training_seed):
    """B3 order: the shortest trajectory of each problem, problems shuffled once [T7-2]."""
    best = shortest_per_problem(problems, tokens)
    return [best[pid] for pid in random_order(best, training_seed)]


def static_round(order, problems, tokens, budget, n_cap, round_index):
    """
    Round round_index of a once-computed order (B3, B4): walk it without repeating a trajectory; a
    trajectory skipped by n_cap or by the budget stays available for later rounds [T7-1, T7-2].
    """
    used_before = set()
    for r in range(round_index + 1):
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap, used_before)
        if r < round_index:
            used_before |= set(selected)
    return selected, reasons, used, stopped


def select(method, round_index, training_seed, problems, tokens, budget, n_cap, ell_by_id=None, trained_ids=()):
    """
    One round's selection over the eligible pool. Returns the selected ids, tokens used, band parameters
    and one log entry per eligible trajectory (fields of selection_log_schema.json, null where unused).
    ell_by_id: ell at S_t for B5 and the ablation, at S_0 for B4 in every round.
    trained_ids: trained in this run before this round (reference set of B5 and the ablation [T7-3]).
    """
    if method not in METHODS:
        raise ValueError(f"unknown method {method}")
    scoring, scores = None, None
    if method in NEEDS_NLL:
        if ell_by_id is None or set(ell_by_id) != set(problems):
            raise ValueError(f"{method} needs ell for exactly the eligible pool")
        # B4 ranks at S_0, where nothing has been trained: the reference set is the whole eligible pool.
        scoring = score_checkpoint(ell_by_id, () if method == "B4" else trained_ids)
        scores = scoring["l_proxy"]

    if method == "B2":
        order = random_order(problems, training_seed * 100 + round_index)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    elif method == "B3":
        order = b3_order(problems, tokens, training_seed)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    elif method == "B4":
        order = rank_by_score(scores)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    else:
        if method == "ablation_shuffled_scores":
            scores = shuffled_scores(scores, training_seed * 100 + round_index + ABLATION_SEED_OFFSET)
        order = rank_by_score(scores)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    if not stopped:
        raise ValueError(f"{method} round {round_index}: order exhausted before the budget ({used}/{budget} tokens)")

    ranks = {tid: i for i, tid in enumerate(order, start=1)}
    has_rank = method != "B2"
    rows = []
    for tid in sorted(problems):
        reason = reasons.get(tid, "not_shortest")
        rows.append({
            "problem_id": problems[tid],
            "trajectory_id": tid,
            "tokens": tokens[tid],
            "nll": ell_by_id[tid] if scoring else None,
            "ell_star": scoring["ell_star"] if scoring else None,
            "sigma": scoring["sigma"] if scoring else None,
            "l_proxy": scoring["l_proxy"][tid] if scoring else None,
            "score": scores[tid] if scores else None,
            "rank": ranks.get(tid) if has_rank else None,
            "selected": reason == "selected",
            "criterion": SELECTED_CRITERION[method] if reason == "selected" else reason,
            "outcome": None,
        })
    return {
        "selected": selected,
        "tokens_used": used,
        "ell_star": scoring["ell_star"] if scoring else None,
        "sigma": scoring["sigma"] if scoring else None,
        "n_reference": scoring["n_reference"] if scoring else None,
        "rows": rows,
    }


def probe_groups(tokens, group_size, seed, n_bins=5):
    """
    L_emp groups (config budget.l_emp_groups): token-length quintile bins of the eligible pool, group_size
    trajectories sampled per bin. n_cap is not applied: the rule does not ask for it.
    """
    ordered = sorted(tokens, key=lambda tid: (tokens[tid], tid))
    bins = [ordered[i * len(ordered) // n_bins: (i + 1) * len(ordered) // n_bins] for i in range(n_bins)]
    rng = random.Random(seed)
    return [sorted(rng.sample(b, group_size)) for b in bins]

#!/usr/bin/env python3
"""
L_proxy (analysis_plan.md section 4, config "learnability"). Pure Python: no torch, no files.

  ell*_t  = median of ell over the reference set at S_t
  sigma_t = IQR of ell over the reference set at S_t / 1.349
  L_proxy = exp(-(ell - ell*_t)^2 / (2 sigma_t^2)), computed for every eligible trajectory

Reference set [T7-3]: eligible trajectories not yet trained in this run (at S_0: all eligible).

Quartiles use statistics.quantiles(n=4) (method "exclusive"), the same call as the Task 5 smoke analysis
(analyze_smoke_test.py). The choice cannot change the ranking: L_proxy order depends only on |ell - ell*_t|.
"""
import math
import statistics


def band_parameters(ells):
    """(ell_star, sigma) of a list of ell values."""
    if len(ells) < 2:
        raise ValueError(f"reference set too small: {len(ells)}")
    if not all(math.isfinite(x) for x in ells):
        raise ValueError("non-finite ell in the reference set")
    q = statistics.quantiles(ells, n=4)
    sigma = (q[2] - q[0]) / 1.349
    if sigma <= 0:
        raise ValueError("IQR of ell is zero: L_proxy is undefined")
    return statistics.median(ells), sigma


def l_proxy(ell, ell_star, sigma):
    return math.exp(-((ell - ell_star) ** 2) / (2 * sigma ** 2))


def score_checkpoint(ell_by_id, trained_ids=()):
    """

        raise ValueError(f"trained trajectories missing from the scored pool: {sorted(unknown)[:3]}")

        "sigma": sigma,
        "n_reference": len(reference),
        "l_proxy": {tid: l_proxy(ell, ell_star, sigma) for tid, ell in ell_by_id.items()},
    }

    POOL_VERIFIED, add_model_args, encode_example, load_config, load_pool, load_tokenizer, sha256_file,
    student_from_args, write_json,
)

DEFAULT_OUT = POOL_VERIFIED.with_name("pool_v1_tokens.jsonl")


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--pool", type=Path, default=POOL_VERIFIED)
    ap.add_argument("--out", type=Path, default=DEFAULT_OUT)
    add_model_args(ap)
    args = ap.parse_args()

    cfg = load_config()
    model_name, revision = student_from_args(args, cfg)
    tok = load_tokenizer(model_name, revision)
    pool = sorted(load_pool(args.pool), key=lambda r: r["trajectory_id"])
    args.out.parent.mkdir(parents=True, exist_ok=True)
    total = 0
    with args.out.open("w", encoding="utf-8") as f:
        for r in pool:
            n = encode_example(tok, r["question"], r["raw_generation"], add_eos=False)["n_traj"]
            total += n
            f.write(json.dumps({"trajectory_id": r["trajectory_id"], "problem_id": r["problem_id"],
                                "response_tokens": n}) + "\n")
    write_json(args.out.with_suffix(".summary.json"), {
        "pool": str(args.pool), "pool_sha256": sha256_file(args.pool),
        "tokenizer": model_name, "tokenizer_revision": revision,
        "n_trajectories": len(pool), "response_tokens_sum": total,
    })
    print(f"{len(pool)} trajectories, {total} response tokens -> {args.out}")


if __name__ == "__main__":
    main()
