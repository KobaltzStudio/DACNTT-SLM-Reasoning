"""Unified, no-overwrite controller for P2 B2-B5 and P1/P2 methods.

User-facing rounds are checkpoint numbers: 1 means train/evaluate S1.
P1 selector rounds are 1-based; select_round.py rounds are 0-based.

def sha256(path: Path) -> str:
    h = hashlib.sha256()
    if path.is_file():
        with path.open("rb") as f:
            for chunk in iter(lambda: f.read(1024 * 1024), b""):
                h.update(chunk)
        return h.hexdigest()
    if not path.is_dir():
        raise FileNotFoundError(path)
    files = sorted(x for x in path.rglob("*") if x.is_file())
    if not files:
        raise ValueError(f"Empty checkpoint directory: {path}")
    for item in files:
        h.update(item.relative_to(path).as_posix().encode())
        h.update(b"\0")
        h.update(sha256(item).encode())
        h.update(b"\n")
    return h.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    value = json.loads(path.read_text(encoding="utf-8-sig"))
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows = []
    with path.open("r", encoding="utf-8-sig") as f:
        for n, line in enumerate(f, 1):
            if not line.strip():
                continue
            row = json.loads(line)
            if not isinstance(row, dict):
                raise ValueError(f"Expected object at {path}:{n}")
            rows.append(row)
    return rows


def atomic_json(path: Path, value: Any, *, replace: bool = False) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if path.exists() and not replace:
        raise FileExistsError(f"Refusing to overwrite: {path}")
    fd, tmp_name = tempfile.mkstemp(prefix=f".{path.name}.", suffix=".tmp", dir=path.parent)
    tmp = Path(tmp_name)
    try:
        with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as f:
            json.dump(value, f, ensure_ascii=False, indent=2)
            f.write("\n")
            f.flush(); os.fsync(f.fileno())
        if replace:
            os.replace(tmp, path)
        else:
            os.link(tmp, path)
    finally:
        tmp.unlink(missing_ok=True)


def inside(path: Path, root: Path) -> bool:
    try:
        path.resolve().relative_to(root.resolve())
        return True
    except ValueError:
        return False

    def _subprocess(self, cmd: list[str]) -> None:
        subprocess.run(cmd, cwd=self.root, check=True)

    def _validate_root_inputs(self) -> None:
        for p in (self.config, self.pool, self.tokens, self.tokens.with_suffix(".summary.json"), self.d_pool, self.dev_ids):
            if not p.is_file():
                raise FileNotFoundError(f"Required integration input missing: {p}")
        cfg = read_json(self.config)
        if cfg.get("config_version") != "v2":
            raise ValueError("Only frozen experiment_config.json v2 is supported")
        if cfg.get("selection", {}).get("alpha") != 0.5 or cfg.get("selection", {}).get("alpha_tuned") is not False:
            raise ValueError("Frozen P1P2 alpha policy must remain 0.5 and untuned")
        if cfg.get("training", {}).get("method") != "LoRA":
            raise ValueError("Frozen training method is not LoRA")
        pool_hash = sha256(self.pool)
        # Do not hard-code a collaborator-specific digest. The active pool hash
        # is canonicalized by verifying the token summary against these exact bytes.
        summary = read_json(self.tokens.with_suffix(".summary.json"))
        if summary.get("pool_sha256") != pool_hash:
            raise ValueError("Token summary does not match verified pool")
        dev = [x.strip() for x in self.dev_ids.read_text(encoding="utf-8-sig").splitlines() if x.strip()]
        if len(dev) != 500 or len(set(dev)) != 500:
            raise ValueError("D_dev must contain exactly 500 unique IDs")

    def namespace(self, seed: int) -> tuple[Path, Path, Path]:
        suffix = "" if seed in REGISTERED_SEEDS else "_custom"
        return (self.root / f"runs/p1p2_integration{suffix}",
                self.root / f"models/p1p2_integration{suffix}",
                self.root / f"reports/p1p2_integration{suffix}")

    def safe_output(self, path: Path, kind: str) -> Path:
        path = path.resolve()
        for forbidden in FORBIDDEN:
            if inside(path, self.root / forbidden):
                raise ValueError(f"Forbidden output namespace: {path}")
        if kind == "run" and not (inside(path, self.root / "runs/p1p2_integration") or inside(path, self.root / "runs/p1p2_integration_custom")):
            raise ValueError(f"Run output outside approved namespace: {path}")
        if kind == "model" and not (inside(path, self.root / "models/p1p2_integration") or inside(path, self.root / "models/p1p2_integration_custom")):
            raise ValueError(f"Model output outside approved namespace: {path}")
        return path

    def emit(self, label: str, cmd: list[str], *, outputs: list[Path] = ()) -> None:
        self.actions.append({"stage": label, "command": [str(x) for x in cmd], "outputs": [str(x) for x in outputs]})
        print(f"[{ 'PLAN' if self.dry_run else 'RUN' }] {label}")
        print("  " + subprocess.list2cmdline([str(x) for x in cmd]))
        if not self.dry_run:
            self.runner([str(x) for x in cmd])

    def _valid_prediction(self, path: Path, expected_tag: str | None = None) -> bool:
        summary_path = path.with_suffix(".summary.json")
        if not path.is_file() or not summary_path.is_file(): return False
        rows = read_jsonl(path)
        ids = [str(r.get("problem_id", "")).strip() for r in rows]
        expected = {x.strip() for x in self.dev_ids.read_text(encoding="utf-8-sig").splitlines() if x.strip()}
        summary = read_json(summary_path)
        tag_ok = expected_tag is None or all(r.get("checkpoint") == expected_tag for r in rows)
        summary_ok = (summary.get("split") == "dev" and summary.get("n_problems") == 500
                      and (expected_tag is None or summary.get("checkpoint") == expected_tag))
        return (len(rows) == 500 and len(set(ids)) == 500 and set(ids) == expected
                and tag_ok and summary_ok)

    def _valid_nll(self, path: Path, tag: str) -> bool:
        summary_path = path.with_suffix(".summary.json")
        if not path.is_file() or not summary_path.is_file(): return False
        rows = read_jsonl(path); summary = read_json(summary_path)
        token_rows = read_jsonl(self.tokens)
        expected = {r["trajectory_id"]: r["response_tokens"] for r in token_rows}
        return (summary.get("checkpoint") == tag and summary.get("pool_sha256") == sha256(self.pool)
                and summary.get("n_trajectories") == len(expected)
                and len(rows) == len(expected) and len({r.get("trajectory_id") for r in rows}) == len(expected)
                and {r.get("trajectory_id") for r in rows} == set(expected)
                and all(r.get("checkpoint") == tag and r.get("n_tokens") == expected.get(r.get("trajectory_id")) for r in rows))

    def _ensure_nll(self, run_dir: Path, ckpt: int, adapter: Path | None, *, tag_override: str | None = None) -> Path:
        tag = tag_override or f"S{ckpt}"
        out_dir = run_dir / tag
        out = out_dir / "nll.jsonl"
        out_key = str(out.resolve()).casefold()
        if self._valid_nll(out, tag):
            if not self.resume and not self.dry_run and out_key not in self.generated_nll:
                raise FileExistsError(f"Validated NLL already exists; rerun with --resume to reuse: {out}")
            print(f"[REUSE] validated NLL {out}")
            return out
        if self.dry_run and out_key in self.generated_nll:
            return out
        if out.exists() or out.with_suffix(".summary.json").exists():
            raise RuntimeError(f"Partial or invalid NLL exists; preserve and investigate, do not overwrite: {out}")
        cmd = [self.py, self.scripts / "compute_nll.py", "--tag", tag, "--out", out,
               "--pool", self.pool, "--batch-size", str(self.nll_batch), "--dtype", self.dtype]
        if adapter is not None: cmd += ["--adapter", adapter]
        self.emit(f"compute_nll_{tag}", cmd, outputs=[out, out.with_suffix(".summary.json")])
        self.generated_nll.add(out_key)
        return out

    def _ensure_selection_p1(self, method: str, seed: int, round_no: int, run_dir: Path,

        train, log = out_dir / "train_set.json", out_dir / "selection_log.jsonl"

            data = read_json(train)
            recorded_state = Path(str(data.get("state_file", "")))
            if not recorded_state.is_absolute():
                recorded_state = (self.root / recorded_state).resolve()

                raise FileNotFoundError(f"Existing P1 selection references missing state: {recorded_state}")

                               "training_seed": seed, "round": round_no,
                               "source_checkpoint": f"S{round_no-1}",
                               "pool_sha256": sha256(self.pool),
                               "tokens_sha256": sha256(self.tokens),
                               "config_sha256": sha256(self.config),
                               "state_sha256": sha256(recorded_state)}

                if data.get(key) != expected:
                    raise RuntimeError(f"Existing P1 train-set {key} mismatch: {train}")
            if method == "P1-DEFICIT":
                if data.get("nll_file") is not None or data.get("nll_sha256") is not None:

                raise RuntimeError(f"Existing P1 selection log does not cover the full verified pool: {log}")

            raise RuntimeError(f"Incomplete P1 selection pair exists; refusing overwrite: {out_dir}")

               "--pool", self.pool, "--tokens", self.tokens, "--d-pool", self.d_pool, "--config", self.config]
        for previous in range(1, round_no): cmd += ["--previous-train-set", run_dir / f"round_{previous:03d}" / "train_set.json"]
        if nll_path is not None: cmd += ["--nll", nll_path]
        self.emit(f"select_{method}_round_{round_no}", cmd, outputs=[train, log])

    def _ensure_selection_p2(self, method: str, seed: int, target_ckpt: int, run_dir: Path,

    def _hook(self, method: str, seed: int, ckpt: int, prediction: Path, checkpoint: Path,
              run_dir: Path, model_dir: Path, report_dir: Path, train_set: Path, selection_log: Path,
              shared_group: str | None = None) -> Path:
        run_id = f"{method}_seed{seed}_S{ckpt}"
        out_dir = report_dir / "capability_sidecars" / run_id
        train_log = checkpoint / "train_log.json"
        checkpoint_log = run_dir / "checkpoint_log.jsonl"
        cmd = [self.py, self.scripts / "run_p1p2_post_eval.py", "--method", method, "--seed", str(seed),
               "--round", str(ckpt), "--prediction", prediction, "--checkpoint", checkpoint,
               "--output-dir", out_dir, "--manual-root", self.root,
               "--train-set", train_set, "--selection-log", selection_log,
               "--train-log", train_log, "--checkpoint-log", checkpoint_log]
        if shared_group:
            cmd += ["--shared-checkpoint-group", shared_group]
        self.emit(f"post_eval_{run_id}", cmd, outputs=[out_dir])
        return out_dir

    def _state(self, sidecar: Path, method: str, seed: int, selection_round: int, report_dir: Path) -> Path:
        out = report_dir / "capability_state" / f"{method}_seed{seed}" / f"selection_round_{selection_round:03d}.json"
        cmd = [self.py, self.scripts / "p1p2_capability_state.py", "--sidecar-dir", sidecar,
               "--target-method", method, "--target-seed", str(seed), "--out", out]
        self.emit(f"capability_state_{method}_seed{seed}_selection_{selection_round}", cmd, outputs=[out])
        return out

    def _checkpoint_log(self, run_dir: Path, method: str, seed: int, ckpt: int,
                        train_set: Path, model_dir: Path, train_log: Path) -> Path:
        path = run_dir / "checkpoint_log.jsonl"
        row = {"record_type": "checkpoint", "checkpoint": f"S{ckpt}", "parent_checkpoint": f"S{ckpt-1}",
               "round": ckpt, "method": method, "training_seed": seed, "data_order_seed": seed,
               "train_set": str(train_set), "train_set_sha256": sha256(train_set),
               "checkpoint_path": str(model_dir), "checkpoint_sha256": sha256(model_dir),
               "train_log": str(train_log), "train_log_sha256": sha256(train_log)}
        if path.exists():
            rows = read_jsonl(path)
            existing = [r for r in rows if r.get("checkpoint") == f"S{ckpt}"]
            if existing:
                if existing[0] != row: raise RuntimeError(f"Checkpoint log provenance collision: {path} S{ckpt}")
                return path
            with path.open("a", encoding="utf-8") as f: f.write(json.dumps(row, ensure_ascii=False) + "\n")
        else:
            path.parent.mkdir(parents=True, exist_ok=True)
            with path.open("x", encoding="utf-8") as f: f.write(json.dumps(row, ensure_ascii=False) + "\n")
        return path

    def _try_b4_s1_alias(self, seed: int, b4_train_set: Path, b4_model_dir: Path,
                         runs_root: Path, models_root: Path) -> bool:
        """Reuse B5 S1 only after exact selection and training provenance checks."""
        b5_run = runs_root / f"B5_seed{seed}"
        b5_train = b5_run / "round_000" / "train_set.json"
        b5_model = models_root / f"B5_seed{seed}" / "student_ckpt_001"
        b5_log = b5_model / "train_log.json"
        if not (b5_train.is_file() and b5_log.is_file() and b5_model.is_dir()):
            return False
        own, other, train_log = read_json(b4_train_set), read_json(b5_train), read_json(b5_log)
        required_equal = (own.get("trajectory_ids") == other.get("trajectory_ids")
                          and own.get("pool_sha256") == other.get("pool_sha256")
                          and own.get("config_sha256") == other.get("config_sha256")
                          and own.get("nll_sha256") == other.get("nll_sha256")
                          and own.get("source_checkpoint") == "S0"
                          and other.get("source_checkpoint") == "S0")
        train_provenance = (train_log.get("seed") == seed
                            and train_log.get("checkpoint_parent") == "student_ckpt_000"
                            and train_log.get("train_set_sha256") == sha256(b5_train))
        if not required_equal or not train_provenance:
            raise RuntimeError("B4/B5 S1 alias rejected: selection IDs, source provenance, or B5 train provenance differ")
        if b4_model_dir.exists():
            raise FileExistsError(f"B4 S1 adapter destination already exists: {b4_model_dir}")
        if self.dry_run:
            self.actions.append({"stage": "reuse_B5_S1_for_B4_S1_after_provenance_check",
                                 "command": ["CONTROLLER_INTERNAL", "copy_verified_B5_S1_adapter", str(b5_model), str(b4_model_dir)],
                                 "outputs": [str(b4_model_dir)]})
        else:
            b4_model_dir.parent.mkdir(parents=True, exist_ok=True)
            shutil.copytree(b5_model, b4_model_dir)
        print(f"[ALIAS] B4 seed {seed} S1 reuses B5 S1 after exact ID/provenance verification")
        return True

    def run_method_seed(self, method: str, seed: int, checkpoints: list[int]) -> None:
        runs_root, models_root, reports_root = self.namespace(seed)
        run_dir = self.safe_output(runs_root / f"{method}_seed{seed}", "run")
        model_run = self.safe_output(models_root / f"{method}_seed{seed}", "model")
        run_dir.mkdir(parents=True, exist_ok=True) if not self.dry_run else None
        model_run.mkdir(parents=True, exist_ok=True) if not self.dry_run else None
        # S0 reference is immutable and shared; it is never trained by this controller.
        state_s0 = None
        if method in P1_METHODS:
            if not self.s0_sidecar.is_dir() or not self.s0_reference.is_file():
                raise FileNotFoundError("P1 requires validated SHARED S0 sidecar and s0_base_model_reference.json")
            state_s0 = self._state(self.s0_sidecar, method, seed, 1, reports_root)
        for ckpt in checkpoints:
            p2_round = ckpt - 1
            round_no = ckpt
            model_dir = model_run / f"student_ckpt_{ckpt:03d}"
            prediction = run_dir / f"S{ckpt}" / "eval_dev.jsonl"
            adapter_parent = model_run / f"student_ckpt_{ckpt-1:03d}"
            if ckpt == 1:
                adapter_parent = None
            if model_dir.exists():
                train_log_existing = model_dir / "train_log.json"
                if not ((self.resume or self.dry_run) and train_log_existing.is_file()):
                    raise FileExistsError(f"Adapter directory collision or interrupted training; refusing to write: {model_dir}")
                if not any(model_dir.glob("adapter_model.*")) and not (model_dir / "adapter_model.safetensors").is_file():
                    raise RuntimeError(f"Existing adapter lacks adapter weights: {model_dir}")
            else:
                train_set = run_dir / (f"round_{round_no:03d}" if method in P1_METHODS else f"round_{p2_round:03d}") / "train_set.json"
                selection_log = train_set.with_name("selection_log.jsonl")
                if method == "P1-DEFICIT":
                    nll_path = None
                elif method == "P1P2":
                    existing_round1 = run_dir / "round_001" / "train_set.json"
                    if ckpt == 1 and existing_round1.is_file():
                        recorded = read_json(existing_round1).get("nll_file")
                        if not recorded:
                            raise RuntimeError("Existing P1P2 round-1 train set has no recorded S0 NLL path")
                        nll_path = Path(recorded)
                        if not nll_path.is_absolute(): nll_path = (self.root / nll_path).resolve()
                        if not self._valid_nll(nll_path, "S0"):
                            raise RuntimeError(f"Existing P1P2 round-1 NLL is not valid/full-coverage: {nll_path}")
                    else:
                        nll_path = self._ensure_nll(run_dir, ckpt - 1, adapter_parent)
                elif method == "B4":
                    nll_path = self._ensure_nll(run_dir, 0, None)
                elif method == "B5":
                    nll_path = self._ensure_nll(run_dir, ckpt - 1, adapter_parent)
                else:
                    nll_path = None
                if method in P1_METHODS:
                    state_path = state_s0 if ckpt == 1 else reports_root / "capability_state" / f"{method}_seed{seed}" / f"selection_round_{ckpt:03d}.json"

def parse_checkpoints(args: argparse.Namespace) -> list[int]:
    if args.round is not None:
        if args.from_checkpoint or args.to_checkpoint:
            raise ValueError("Do not combine --round with checkpoint range flags")
        return [args.round]
    if args.rounds != ["all"] and (args.from_checkpoint or args.to_checkpoint):
        raise ValueError("Do not combine --rounds with checkpoint range flags")
    if args.from_checkpoint or args.to_checkpoint:
        if not args.from_checkpoint or not args.to_checkpoint:
            raise ValueError("--from-checkpoint and --to-checkpoint must be supplied together")
        a, b = int(args.from_checkpoint[1:]), int(args.to_checkpoint[1:])
        if a not in (0,1,2,3) or b not in (1,2,3) or a >= b:
            raise ValueError("Checkpoint range must be increasing, e.g. S1 through S3")
        return list(range(a + 1, b + 1))
    if args.rounds == ["all"]: return [1,2,3]
    try: values = sorted({int(x) for x in args.rounds})
    except ValueError as exc: raise ValueError("--rounds accepts all or checkpoint numbers 1 2 3") from exc
    if not values or any(x not in (1,2,3) for x in values):
        raise ValueError("--rounds values must be 1, 2, or 3")
    return values

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    try:
        value = json.loads(Path(path).read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise ValueError(f"Invalid JSON in {path}: {exc}") from exc
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    with Path(path).open("r", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(f"Invalid JSONL at {path}:{line_number}: {exc}") from exc
            if not isinstance(row, dict):
                raise ValueError(f"Expected JSON object at {path}:{line_number}")
            rows.append(row)
    return rows


def _require_file(path: Path, label: str) -> Path:
    path = Path(path).resolve()
    if not path.is_file():
        raise FileNotFoundError(f"{label} not found: {path}")
    return path


def load_pool(pool_path: Path, d_pool_path: Path) -> tuple[dict[str, dict], dict[str, str]]:
    pool_path = _require_file(pool_path, "Verified pool")
    d_pool_path = _require_file(d_pool_path, "D_pool ID file")
    d_pool_ids = {line.strip() for line in d_pool_path.read_text(encoding="utf-8").splitlines() if line.strip()}
    pool_by_id: dict[str, dict] = {}
    problems: dict[str, str] = {}
    for index, row in enumerate(read_jsonl(pool_path), 1):
        tid, pid = row.get("trajectory_id"), row.get("problem_id")
        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Pool row {index}: invalid trajectory_id")
        if not isinstance(pid, str) or not pid:
            raise ValueError(f"Pool row {index}: invalid problem_id")
        if tid in pool_by_id:
            raise ValueError(f"Duplicate trajectory_id in pool: {tid}")
        metadata = row.get("metadata")
        if not isinstance(metadata, dict) or metadata.get("verification_status") != "valid":
            raise ValueError(f"{tid}: metadata.verification_status must be valid")
        capabilities = row.get("capabilities")
        if not isinstance(capabilities, list) or not capabilities:
            raise ValueError(f"{tid}: capability list is empty or invalid")
        if any(not isinstance(tag, str) for tag in capabilities):
            raise ValueError(f"{tid}: capability tags must be strings")
        if len(capabilities) != len(set(capabilities)):
            raise ValueError(f"{tid}: duplicate capability tags")
        unknown = sorted(set(capabilities) - set(TAXONOMY))
        if unknown:
            raise ValueError(f"{tid}: unknown capability tags: {unknown}")
        if pid not in d_pool_ids:
            raise ValueError(f"{tid}: problem_id is outside D_pool: {pid}")
        pool_by_id[tid] = row
        problems[tid] = pid
    if not pool_by_id:
        raise ValueError("Verified pool is empty")
    return pool_by_id, problems


def load_tokens(tokens_path: Path, pool_path: Path, problems: dict[str, str]) -> dict[str, int]:
    tokens_path = _require_file(tokens_path, "Token table")
    pool_path = _require_file(pool_path, "Verified pool")
    summary_path = tokens_path.with_suffix(".summary.json")
    summary = read_json(_require_file(summary_path, "Token summary"))
    actual_pool_hash = sha256_file(pool_path)
    if summary.get("pool_sha256") != actual_pool_hash:
        raise ValueError("Token summary pool_sha256 does not match the supplied pool file")
    rows = read_jsonl(tokens_path)
    tokens: dict[str, int] = {}
    for index, row in enumerate(rows, 1):
        tid, pid, count = row.get("trajectory_id"), row.get("problem_id"), row.get("response_tokens")
        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Token row {index}: invalid trajectory_id")
        if tid in tokens:
            raise ValueError(f"Duplicate token row: {tid}")
        if tid not in problems:
            raise ValueError(f"Token row not present in verified pool: {tid}")
        if problems[tid] != pid:
            raise ValueError(f"Token problem_id mismatch: {tid}")
        if not isinstance(count, int) or isinstance(count, bool) or count <= 0:
            raise ValueError(f"Invalid response_tokens for {tid}: {count}")
        tokens[tid] = count
    if set(tokens) != set(problems):
        missing, extra = sorted(set(problems) - set(tokens)), sorted(set(tokens) - set(problems))
        raise ValueError(f"Pool/token IDs differ. Missing={missing[:5]}, extra={extra[:5]}")
    if len(rows) != summary.get("n_trajectories"):
        raise ValueError("Token summary n_trajectories mismatch")
    if sum(tokens.values()) != summary.get("response_tokens_sum"):
        raise ValueError("Token summary response_tokens_sum mismatch")
    return tokens


def load_capability_deficits(state_path: Path, method: str, seed: int, round_number: int) -> dict[str, float]:
    state = read_json(_require_file(state_path, "Capability state"))
    if state.get("schema_version") != "p1p2.capability-state.v1":
        raise ValueError("Unsupported capability state schema")
    target = state.get("target")
    if not isinstance(target, dict):
        raise ValueError("Capability state target is missing or invalid")
    expected = {"method": method, "seed": seed, "selection_round": round_number}
    for key, value in expected.items():
        if target.get(key) != value:
            raise ValueError(f"State target {key} mismatch: expected {value}, got {target.get(key)}")
    capabilities = state.get("capabilities")
    if not isinstance(capabilities, dict):
        raise ValueError("Capability state has no capabilities object")
    deficits: dict[str, float] = {}
    for capability in TAXONOMY:
        item = capabilities.get(capability)
        if not isinstance(item, dict):
            raise ValueError(f"Missing capability state: {capability}")
        value = item.get("deficit")
        if not isinstance(value, (int, float)) or isinstance(value, bool) or not math.isfinite(value) or not 0 <= value <= 1:
            raise ValueError(f"Invalid deficit for {capability}: {value}")
        deficits[capability] = float(value)
    return deficits


def load_previous_train_ids(paths: list[Path], problems: dict[str, str], method: str, seed: int,
                            current_round: int, pool_sha256: str) -> list[str]:
    trained: set[str] = set()
    seen_rounds: set[int] = set()
    for path in paths:
        data = read_json(_require_file(path, "Previous train set"))
        if data.get("schema_version") != "p1p2.train-set.v1":
            raise ValueError(f"Unsupported previous train-set schema: {path}")
        if data.get("method") != method:
            raise ValueError(f"Previous train-set method mismatch: {path}")
        if data.get("training_seed") != seed:
            raise ValueError(f"Previous train-set seed mismatch: {path}")
        previous_round = data.get("round")
        if not isinstance(previous_round, int) or isinstance(previous_round, bool) or not 1 <= previous_round < current_round:
            raise ValueError(f"Previous train-set round must be in [1, {current_round - 1}]: {path}")
        if previous_round in seen_rounds:
            raise ValueError(f"Duplicate previous train-set round {previous_round}")
        seen_rounds.add(previous_round)
        if data.get("pool_sha256") != pool_sha256:
            raise ValueError(f"Previous train-set pool hash mismatch: {path}")
        ids = data.get("trajectory_ids")
        if not isinstance(ids, list) or not ids:
            raise ValueError(f"Invalid or empty trajectory_ids in {path}")
        for tid in ids:
            if not isinstance(tid, str) or tid not in problems:
                raise ValueError(f"Previous train set contains unknown trajectory: {tid}")
            trained.add(tid)
    expected_rounds = set(range(1, current_round))
    if seen_rounds != expected_rounds:
        raise ValueError(f"Previous train-set rounds incomplete: expected {sorted(expected_rounds)}, got {sorted(seen_rounds)}")
    return sorted(trained)


def load_nll(nll_path: Path, tokens: dict[str, int], source_checkpoint: str,

        if tid in ell_by_id:

        if tid not in tokens:
            raise ValueError(f"NLL trajectory not present in token table: {tid}")
        if row.get("checkpoint") != source_checkpoint:
            raise ValueError(f"NLL checkpoint mismatch for {tid}: expected {source_checkpoint}, got {row.get('checkpoint')}")
        if row.get("n_tokens") != tokens[tid]:
            raise ValueError(f"NLL token count mismatch for {tid}")
        ell = row.get("ell")
        if not isinstance(ell, (int, float)) or isinstance(ell, bool) or not math.isfinite(ell):
            raise ValueError(f"Invalid ell value for {tid}: {ell}")
        ell_by_id[tid] = float(ell)
    if set(ell_by_id) != set(tokens):
        missing, extra = sorted(set(tokens) - set(ell_by_id)), sorted(set(ell_by_id) - set(tokens))

def average_percentile_ranks(scores: dict[str, float]) -> dict[str, float]:
    if not scores:
        raise ValueError("Cannot rank an empty score mapping")
    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    n = len(ordered)
    if n == 1:
        return {ordered[0][0]: 0.5}
    result: dict[str, float] = {}
    index = 0
    while index < n:
        end = index + 1
        value = ordered[index][1]
        while end < n and ordered[end][1] == value:
            end += 1
        average_rank = ((index + 1) + end) / 2.0
        percentile = (average_rank - 1.0) / (n - 1.0)
        for position in range(index, end):
            result[ordered[position][0]] = percentile
        index = end
    return result


def calculate_p1_scores(pool_by_id: dict[str, dict], deficits: dict[str, float]) -> dict[str, float]:
    result: dict[str, float] = {}
    for tid, row in pool_by_id.items():
        tags = row["capabilities"]
        result[tid] = sum(deficits[tag] for tag in tags) / len(tags)
    return result


def publish_pair_no_overwrite(train_set_path: Path, selection_log_path: Path,
                              train_set_bytes: bytes, selection_log_bytes: bytes) -> None:
    train_set_path, selection_log_path = Path(train_set_path), Path(selection_log_path)
    if train_set_path.resolve() == selection_log_path.resolve():
        raise ValueError("Train-set and selection-log paths must be different")
    train_set_path.parent.mkdir(parents=True, exist_ok=True)
    selection_log_path.parent.mkdir(parents=True, exist_ok=True)
    if train_set_path.exists() or selection_log_path.exists():
        raise FileExistsError("Refusing to overwrite an existing selector output")
    nonce = f"{os.getpid()}"
    train_tmp = train_set_path.with_name(train_set_path.name + f".{nonce}.tmp")
    log_tmp = selection_log_path.with_name(selection_log_path.name + f".{nonce}.tmp")
    created_train = False
    created_temps: list[Path] = []
    try:
        for temp_path, content in ((train_tmp, train_set_bytes), (log_tmp, selection_log_bytes)):
            flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
            fd = os.open(str(temp_path), flags, 0o600)
            created_temps.append(temp_path)
            with os.fdopen(fd, "wb") as handle:
                handle.write(content)
                handle.flush()
                os.fsync(handle.fileno())
        os.link(train_tmp, train_set_path)
        created_train = True
        os.link(log_tmp, selection_log_path)
    except Exception:
        if created_train and train_set_path.exists() and sha256_file(train_set_path) == hashlib.sha256(train_set_bytes).hexdigest():
            train_set_path.unlink()
        raise
    finally:
        for temp_path in created_temps:
            temp_path.unlink(missing_ok=True)

def validate_metrics(metrics_path: Path) -> dict[str, dict[str, Any]]:
    by_capability: dict[str, dict[str, Any]] = {}

    with metrics_path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue

            row = json.loads(line)
            if not isinstance(row, dict):
                raise ValueError(f"Invalid metric row at line {line_no}")

            capability = row.get("capability")
            if capability not in CAPABILITIES:
                raise ValueError(f"Unknown capability: {capability!r}")

            if capability in by_capability:
                raise ValueError(f"Duplicate capability row: {capability}")

            if row.get("strict_metric") != "status_equals_valid":
                raise ValueError("Capability row does not use strict metric")

            if row.get("gold_policy") != "approved_85_primary":
                raise ValueError("Capability row uses an invalid Gold policy")

            support = int(row["n_support"])
            correct = int(row["n_correct"])
            incorrect = int(row["n_incorrect"])

            if support != EXPECTED_SUPPORT[capability]:
                raise ValueError(
                    f"Support mismatch for {capability}: "
                    f"{support} != {EXPECTED_SUPPORT[capability]}"
                )

            if correct < 0 or incorrect < 0:
                raise ValueError(f"Negative metric count: {capability}")

            if correct + incorrect != support:
                raise ValueError(f"Count mismatch: {capability}")

            mastery = correct / support
            deficit = 1.0 - mastery

            if not math.isfinite(float(row["mastery"])):
                raise ValueError(f"Non-finite mastery: {capability}")

            if not math.isfinite(float(row["deficit"])):
                raise ValueError(f"Non-finite deficit: {capability}")

            if abs(float(row["mastery"]) - mastery) > 1e-12:
                raise ValueError(f"Mastery mismatch: {capability}")

            if abs(float(row["deficit"]) - deficit) > 1e-12:
                raise ValueError(f"Deficit mismatch: {capability}")

            by_capability[capability] = {
                "n_support": support,
                "n_correct": correct,
                "n_incorrect": incorrect,
                "mastery": mastery,
                "deficit": deficit,
                "ci95_low": row.get("ci95_low"),
                "ci95_high": row.get("ci95_high"),
            }

    if set(by_capability) != set(CAPABILITIES):
        raise ValueError(
            "Capability coverage is incomplete: "
            f"{sorted(by_capability)}"
        )

    return by_capability


def publish_exclusive(output: Path, content: bytes) -> None:
    """Publish through a same-directory hard link; never replace a target."""
    output.parent.mkdir(parents=True, exist_ok=True)

    fd, temp_name = tempfile.mkstemp(
        prefix=f".{output.name}.staging.",
        suffix=".tmp",
        dir=str(output.parent),
    )
    temp_path = Path(temp_name)

    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(content)
            stream.flush()
            os.fsync(stream.fileno())

        try:
            os.link(temp_path, output)
        except FileExistsError:
            raise
        except OSError as exc:
            raise RuntimeError(
                "ATOMIC_NO_OVERWRITE_UNAVAILABLE: hard-link publication "
                f"failed; temporary file preserved at {temp_path}: {exc}"
            ) from exc

        temp_path.unlink(missing_ok=True)

    except Exception:
        if temp_path.exists():
            print(f"RECOVERY_TEMP_PRESERVED={temp_path}")
        raise

def sha256_file(path: Path) -> str:
    """Return the SHA-256 digest of a regular file."""
    source = Path(path).resolve()

    return digest.hexdigest()


def canonical_json_bytes(value: Any) -> bytes:
    """Serialize JSON deterministically for stable fingerprints."""
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n").encode("utf-8")


def sha256_json(value: Any) -> str:
    return hashlib.sha256(canonical_json_bytes(value)).hexdigest()


def fingerprint_files(paths: Mapping[str, Path]) -> dict[str, Any]:
    """Fingerprint named input files; names are stable logical identifiers."""
    if not paths:
        raise ValueError("At least one input file is required for a fingerprint")
    entries: dict[str, dict[str, Any]] = {}
    resolved_seen: dict[str, str] = {}
    for name in sorted(paths):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Fingerprint input names must be non-empty strings")
        path = Path(paths[name]).resolve()
        if not path.is_file():
            raise FileNotFoundError(f"Fingerprint input '{name}' not found: {path}")
        normalized = str(path).casefold()
        if normalized in resolved_seen:
            raise ValueError(f"Duplicate fingerprint path for '{name}' and '{resolved_seen[normalized]}'")
        resolved_seen[normalized] = name
        entries[name] = {
            "path": str(path),
            "size_bytes": path.stat().st_size,
            "sha256": sha256_file(path),
        }
    payload = {"schema_version": "p1p2.input-fingerprint.v1", "files": entries}
    payload["fingerprint_sha256"] = sha256_json(payload)
    return payload

        raise ValueError("Fingerprint is missing files or fingerprint_sha256")
    unsigned = {key: value for key, value in fingerprint.items() if key != "fingerprint_sha256"}
    actual = sha256_json(unsigned)
    if actual != digest:
        raise ValueError("Fingerprint digest mismatch")
    return True


def verify_fingerprint_files(fingerprint: Mapping[str, Any]) -> bool:
    """Verify that every recorded file still has its recorded size and hash."""
    verify_fingerprint(fingerprint)
    files = fingerprint["files"]
    for name, entry in files.items():
        if not isinstance(entry, Mapping):
            raise ValueError(f"Invalid fingerprint entry: {name}")
        path = Path(entry.get("path", ""))
        if not path.is_file():
            raise FileNotFoundError(f"Fingerprint source missing: {name}: {path}")
        if path.stat().st_size != entry.get("size_bytes"):
            raise ValueError(f"Fingerprint size mismatch: {name}")
        if sha256_file(path) != entry.get("sha256"):
            raise ValueError(f"Fingerprint SHA-256 mismatch: {name}")
    return True


def ensure_no_overwrite(paths: Iterable[Path]) -> list[Path]:
    """Resolve and preflight output paths without creating or changing them."""
    resolved: list[Path] = []
    seen: set[str] = set()
    for raw_path in paths:
        path = Path(raw_path).resolve()
        key = str(path).casefold()
        if key in seen:
            raise ValueError(f"Duplicate output path: {path}")
        seen.add(key)
        if path.exists():
            raise FileExistsError(f"Refusing to overwrite existing artifact: {path}")
        resolved.append(path)
    if not resolved:
        raise ValueError("At least one output path is required")
    return resolved


def _write_temp_for_target(target: Path, content: bytes) -> Path:
    target.parent.mkdir(parents=True, exist_ok=True)
    fd, raw_temp = tempfile.mkstemp(prefix=f".{target.name}.", suffix=".tmp", dir=str(target.parent))
    temp_path = Path(raw_temp)
    try:
        with os.fdopen(fd, "wb") as handle:
            handle.write(content)
            handle.flush()
            os.fsync(handle.fileno())
    except Exception:
        temp_path.unlink(missing_ok=True)
        raise
    return temp_path


def publish_bytes_no_overwrite(target: Path, content: bytes) -> Path:
    """Atomically publish one file without replacing an existing path."""
    if not isinstance(content, bytes):
        raise TypeError("content must be bytes")
    destination = Path(target).resolve()
    ensure_no_overwrite([destination])
    temp_path = _write_temp_for_target(destination, content)
    try:
        # A same-filesystem hard link is atomic and fails if destination exists.
        os.link(temp_path, destination)
    finally:
        temp_path.unlink(missing_ok=True)
    return destination


def publish_json_no_overwrite(target: Path, value: Any) -> Path:
    return publish_bytes_no_overwrite(target, canonical_json_bytes(value))


def publish_pair_no_overwrite(first_path: Path, first_content: bytes,
                              second_path: Path, second_content: bytes) -> tuple[Path, Path]:
    """Publish two files; roll back the first if publishing the second fails."""
    first = Path(first_path).resolve()
    second = Path(second_path).resolve()
    ensure_no_overwrite([first, second])
    first_temp = _write_temp_for_target(first, first_content)
    second_temp: Path | None = None
    first_published = False
    try:
        second_temp = _write_temp_for_target(second, second_content)
        os.link(first_temp, first)
        first_published = True
        os.link(second_temp, second)
    except Exception:
        if first_published and first.exists():
            expected = hashlib.sha256(first_content).hexdigest()
            if sha256_file(first) == expected:
                first.unlink()
        raise
    finally:
        first_temp.unlink(missing_ok=True)
        if second_temp is not None:
            second_temp.unlink(missing_ok=True)
    return first, second


def publish_manifest_no_overwrite(target: Path, manifest: Mapping[str, Any],
                                   input_paths: Mapping[str, Path]) -> Path:
    """Attach a reproducible input fingerprint and publish a run manifest."""
    if not isinstance(manifest, Mapping):
        raise TypeError("manifest must be a mapping")
    payload = dict(manifest)
    if "input_fingerprint" in payload:
        raise ValueError("manifest must not supply input_fingerprint directly")
    payload["input_fingerprint"] = fingerprint_files(input_paths)
    return publish_json_no_overwrite(target, payload)

def collate(examples, pad_id):
    width = max(len(e["input_ids"]) for e in examples)
    ids, labels, mask = [], [], []
    for e in examples:
        pad = width - len(e["input_ids"])
        ids.append(e["input_ids"] + [pad_id] * pad)          # right padding for training
        labels.append(e["labels"] + [IGNORE_INDEX] * pad)
        mask.append([1] * len(e["input_ids"]) + [0] * pad)
    return torch.tensor(ids), torch.tensor(labels), torch.tensor(mask)

def generate_hf(todo, tok, model_name, revision, dtype, adapter, max_new_tokens, batch_size):
    """Yields lists of (row, text, n_output_tokens), one list per batch."""
    from transformers import GenerationConfig
    model = load_model(model_name, revision, dtype, adapter=adapter)
    gen_cfg = GenerationConfig(
        do_sample=False,
        max_new_tokens=max_new_tokens,
        repetition_penalty=1.0,
        eos_token_id=[end_of_turn_id(tok), tok.eos_token_id],
        pad_token_id=tok.pad_token_id,
    )
    for i in range(0, len(todo), batch_size):
        batch = todo[i: i + batch_size]
        prompts = [build_prompt(tok, r["question"]) for r in batch]
        enc = tok(prompts, return_tensors="pt", padding=True, add_special_tokens=False).to(model.device)
        with torch.no_grad():
            out = model.generate(**enc, generation_config=gen_cfg)
        results = []
        for r, ids in zip(batch, out[:, enc["input_ids"].shape[1]:]):
            ids = ids.tolist()
            n_out = len(ids)
            while n_out and ids[n_out - 1] == tok.pad_token_id:
                n_out -= 1
            results.append((r, tok.decode(ids[:n_out], skip_special_tokens=True), n_out))
        yield results

#!/usr/bin/env python3
"""
Score every verified pool trajectory under a Student checkpoint (one forward pass, no backward).

Per trajectory tau (trajectory tokens only, the end-of-turn token excluded):
  nll_sum    = sum_t -log p(tau_t | x, tau_<t)
  ell        = nll_sum / n_tokens            (length-normalized NLL; input of L_proxy, config "learnability")
  brier_sum  = sum_t || pi_t - onehot(tau_t) ||^2
               (Brier residual; stored so the LARK score, arXiv 2605.30651 Eq. 7, can be computed
                later without another pass. Exploratory, not part of the frozen L_proxy.)

L_proxy itself is not computed here: it needs ell over the whole pool (median / IQR rule) and
is derived in the analysis step.

--pool defaults to the official pool v1 (trajectory/pool/pool_v1_verified.jsonl). The Task 5 smoke test
scored the preliminary pool; reproduce it with --pool trajectory/pool/batch_100_teacher_verified.jsonl.

Usage (from experiments/week2September/manual/):

      --pool trajectory/pool/batch_100_teacher_verified.jsonl

      --out reports/task5/smoke_test/nll_pool_S1_seed13.jsonl --pool trajectory/pool/batch_100_teacher_verified.jsonl

            width = max(len(e["input_ids"]) for e in batch)
            ids = torch.tensor([e["input_ids"] + [tok.pad_token_id] * (width - len(e["input_ids"])) for e in batch])
            mask = torch.tensor([[1] * len(e["input_ids"]) + [0] * (width - len(e["input_ids"])) for e in batch])

                  flush=True)
    elapsed = time.time() - t0

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def hash_path(path: Path) -> str:
    path = path.resolve()

    if path.is_file():
        return sha256_file(path)

    if not path.is_dir():
        raise FileNotFoundError(f"Input does not exist: {path}")

    files = sorted(p for p in path.rglob("*") if p.is_file())
    if not files:
        raise ValueError(f"Checkpoint directory is empty: {path}")

    digest = hashlib.sha256()
    for file_path in files:
        relative = file_path.relative_to(path).as_posix()
        digest.update(relative.encode("utf-8"))
        digest.update(b"\0")
        digest.update(sha256_file(file_path).encode("ascii"))
        digest.update(b"\n")

    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    with path.open("r", encoding="utf-8-sig") as stream:

    if not isinstance(value, dict):

    return value


def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []

    with path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSONL at {path}:{line_no}: {exc}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected JSON object at {path}:{line_no}"
                )
            rows.append(row)

    return rows

def wilson_interval(
    correct: int,
    support: int,
    z: float = 1.959963984540054,
) -> tuple[float | None, float | None]:
    if support <= 0:
        return None, None

    p = correct / support
    z2 = z * z
    denominator = 1.0 + z2 / support
    center = (p + z2 / (2.0 * support)) / denominator
    margin = (
        z
        * (
            p * (1.0 - p) / support
            + z2 / (4.0 * support * support)
        ) ** 0.5
        / denominator
    )

    return max(0.0, center - margin), min(1.0, center + margin)

def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish without overwrite; retry transient Windows permission failures."""
    retry_delays = (0.25, 0.5, 1.0, 2.0, 4.0)

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)
            return
        except PermissionError as exc:
            if out_dir.exists():
                raise FileExistsError(
                    "OUTPUT_COLLISION: destination appeared during "
                    "publication; refusing to overwrite: " + str(out_dir)
                ) from exc

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)

class Runner:
    """Runs steps once, keeps markers and timing."""

    def __init__(self, runs_dir, dry_run=False):
        self.runs_dir = Path(runs_dir)
        self.dry_run = dry_run

    def marker(self, scope_dir, step):
        return Path(scope_dir) / ".done" / f"{step}.json"

    def is_done(self, scope_dir, step):
        return self.marker(scope_dir, step).exists()

    def mark(self, scope_dir, step, info):
        path = self.marker(scope_dir, step)
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(info, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")

    def run(self, scope_dir, step, cmd):
        printable = " ".join(shlex.quote(str(c)) for c in cmd)
        if self.is_done(scope_dir, step):
            print(f"[skip] {scope_dir}/{step}")
            return
        print(f"[run] {scope_dir}/{step}\n      {printable}", flush=True)
        if self.dry_run:
            return
        t0 = time.time()
        subprocess.run([str(c) for c in cmd], check=True)
        elapsed = round(time.time() - t0, 1)
        self.mark(scope_dir, step, {"command": printable, "elapsed_seconds": elapsed})
        self.runs_dir.mkdir(parents=True, exist_ok=True)
        with (self.runs_dir / "timing.jsonl").open("a", encoding="utf-8") as f:
            f.write(json.dumps({"scope": str(scope_dir), "step": step, "elapsed_seconds": elapsed}) + "\n")

def write_atomic(path, text):
    tmp = path.with_name(path.name + ".tmp")
    tmp.write_text(text, encoding="utf-8")
    os.replace(tmp, path)


def load_tokens(tokens_path, pool_path):
    summary = json.loads(Path(tokens_path).with_suffix(".summary.json").read_text(encoding="utf-8"))
    if summary["pool_sha256"] != sha256_file(pool_path):
        raise ValueError(f"{tokens_path} was built from another pool")
    rows = read_jsonl(tokens_path)
    return {r["trajectory_id"]: r["problem_id"] for r in rows}, {r["trajectory_id"]: r["response_tokens"] for r in rows}


def load_ell(nll_path, tokens):
    """ell by trajectory; the NLL file must cover the pool with the token counts of the budget."""
    rows = read_jsonl(nll_path)
    ell = {r["trajectory_id"]: r["ell"] for r in rows}
    if len(ell) != len(rows) or set(ell) != set(tokens):
        raise ValueError(f"{nll_path} does not cover exactly the eligible pool")
    wrong = [r["trajectory_id"] for r in rows if r["n_tokens"] != tokens[r["trajectory_id"]]]
    if wrong:
        raise ValueError(f"{len(wrong)} token counts in {nll_path} differ from the token table: {wrong[:3]}")
    return ell


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--method", required=True, choices=METHODS)
    ap.add_argument("--seed", required=True, type=int)
    ap.add_argument("--round", required=True, type=int)
    ap.add_argument("--run-dir", required=True, type=Path)
    ap.add_argument("--pool", type=Path, default=POOL_VERIFIED)
    ap.add_argument("--tokens", type=Path, default=DEFAULT_TOKENS)
    ap.add_argument("--nll", type=Path, default=None)
    ap.add_argument("--budget", type=int, default=None, help="pilot only (default: config beta_round_tokens)")
    ap.add_argument("--pilot", action="store_true")
    args = ap.parse_args()

    cfg = load_config()
    if not args.pilot:
        if args.seed not in cfg["seeds"]["training_seeds"] + cfg["seeds"]["optional_extra_seeds"]:
            raise SystemExit(f"seed {args.seed} is not in the config")
        if args.budget is not None:
            raise SystemExit("--budget needs --pilot")
    if not 0 <= args.round < cfg["schedule"]["rounds_M"]:
        raise SystemExit(f"round must be in 0..{cfg['schedule']['rounds_M'] - 1}")
    if args.method in NEEDS_NLL and args.nll is None:
        raise SystemExit(f"{args.method} needs --nll")
    budget = args.budget or cfg["budget"]["beta_round_tokens"]
    n_cap = cfg["selection"]["n_cap"]

    problems, tokens = load_tokens(args.tokens, args.pool)
    trained = []
    for t in range(args.round):
        trained += json.loads((round_dir(args.run_dir, t) / "train_set.json").read_text(encoding="utf-8"))["trajectory_ids"]
    ell = load_ell(args.nll, tokens) if args.method in NEEDS_NLL else None

    res = select(args.method, args.round, args.seed, problems, tokens, budget, n_cap, ell, sorted(set(trained)))

    config_hash = sha256_file(CONFIG_PATH)
    source_t = 0 if args.method == "B4" else args.round  # B4 is ranked at S_0 in every round
    source = cfg["schedule"]["checkpoint_name"].format(t=source_t)
    out_dir = round_dir(args.run_dir, args.round)
    out_dir.mkdir(parents=True, exist_ok=True)
    write_atomic(out_dir / "selection_log.jsonl", "".join(
        json.dumps({"record_type": "selection", "method": args.method, "round": args.round,
                    "source_checkpoint": source, "training_seed": args.seed, **row, "config_hash": config_hash},
                   ensure_ascii=False) + "\n"
        for row in res["rows"]))
    write_atomic(out_dir / "train_set.json", json.dumps({
        "pilot": args.pilot,
        "method": args.method, "round": args.round, "training_seed": args.seed, "source_checkpoint": source,
        "pool": rel(args.pool), "pool_sha256": sha256_file(args.pool),
        "config_sha256": config_hash,
        "nll_file": rel(args.nll) if ell else None,
        "nll_sha256": sha256_file(args.nll) if ell else None,
        "ell_star": res["ell_star"], "sigma": res["sigma"], "n_reference": res["n_reference"],
        "n_eligible": len(problems), "n_trained_before": len(set(trained)),
        "beta_round_tokens": budget, "n_cap": n_cap,
        "tokens_used": res["tokens_used"], "n_selected": len(res["selected"]),
        "trajectory_ids": res["selected"],
    }, ensure_ascii=False, indent=2) + "\n")
    print(f"[{args.method} seed {args.seed} round {args.round}] {len(res['selected'])} trajectories, "
          f"{res['tokens_used']}/{budget} tokens -> {out_dir}")


if __name__ == "__main__":
    main()

def canonical_sha256(value):
    payload = json.dumps(
        value,
        sort_keys=True,
        separators=(",", ":"),
    ).encode("utf-8")
    return hashlib.sha256(payload).hexdigest()


def write_json(path, value):
    path.write_text(
