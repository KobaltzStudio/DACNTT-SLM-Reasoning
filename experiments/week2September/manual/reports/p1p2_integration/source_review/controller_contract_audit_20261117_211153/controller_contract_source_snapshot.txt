Train one round of the Student with LoRA: S_t + B_t -> S_{t+1}.

Loss: token-level mean over all supervised (response) tokens of the effective batch;
prompt tokens are masked. Seeds: torch seed (LoRA init, dropout) = data order seed = --seed.

Pool: the trajectories are looked up in the pool named by the train set ("pool", checked against

def collate(examples, pad_id):
    width = max(len(e["input_ids"]) for e in examples)
    ids, labels, mask = [], [], []
    for e in examples:
        pad = width - len(e["input_ids"])
        ids.append(e["input_ids"] + [pad_id] * pad)          # right padding for training
        labels.append(e["labels"] + [IGNORE_INDEX] * pad)
        mask.append([1] * len(e["input_ids"]) + [0] * pad)
    return torch.tensor(ids), torch.tensor(labels), torch.tensor(mask)

def generate_hf(todo, tok, model_name, revision, dtype, adapter, max_new_tokens, batch_size):
    """Yields lists of (row, text, n_output_tokens), one list per batch."""
    from transformers import GenerationConfig
    model = load_model(model_name, revision, dtype, adapter=adapter)
    gen_cfg = GenerationConfig(
        do_sample=False,
        max_new_tokens=max_new_tokens,
        repetition_penalty=1.0,
        eos_token_id=[end_of_turn_id(tok), tok.eos_token_id],
        pad_token_id=tok.pad_token_id,
    )
    for i in range(0, len(todo), batch_size):
        batch = todo[i: i + batch_size]
        prompts = [build_prompt(tok, r["question"]) for r in batch]
        enc = tok(prompts, return_tensors="pt", padding=True, add_special_tokens=False).to(model.device)
        with torch.no_grad():
            out = model.generate(**enc, generation_config=gen_cfg)
        results = []
        for r, ids in zip(batch, out[:, enc["input_ids"].shape[1]:]):
            ids = ids.tolist()
            n_out = len(ids)
            while n_out and ids[n_out - 1] == tok.pad_token_id:
                n_out -= 1
            results.append((r, tok.decode(ids[:n_out], skip_special_tokens=True), n_out))
        yield results


def generate_vllm(todo, tok, model_name, revision, adapter, max_new_tokens, batch_size, gpu_memory_utilization):
    """Same contract as generate_hf. batch_size here is the number of problems per generate() call."""
    from vllm import LLM, SamplingParams
    if not torch.cuda.is_available() or not torch.cuda.is_bf16_supported():
        raise SystemExit("the vllm backend needs a GPU with bf16 (T6-5)")
    stop_ids = sorted({end_of_turn_id(tok), tok.eos_token_id})
    params = SamplingParams(n=1, temperature=0.0, top_p=1.0, repetition_penalty=1.0,
                            max_tokens=max_new_tokens, stop_token_ids=stop_ids)
    with tempfile.TemporaryDirectory(prefix="merged_student_") as merged:
        if adapter:
            model = load_model(model_name, revision, torch.bfloat16, adapter=adapter)
            model.save_pretrained(merged)
            tok.save_pretrained(merged)
            del model
            gc.collect()
            torch.cuda.empty_cache()
            llm = LLM(model=merged, dtype="bfloat16", generation_config="vllm",
                      gpu_memory_utilization=gpu_memory_utilization)
        else:
            llm = LLM(model=model_name, revision=revision, tokenizer_revision=revision, dtype="bfloat16",
                      generation_config="vllm", gpu_memory_utilization=gpu_memory_utilization)
        for i in range(0, len(todo), batch_size):
            batch = todo[i: i + batch_size]
            outputs = llm.generate([build_prompt(tok, r["question"]) for r in batch], params, use_tqdm=False)
            yield [(r, o.outputs[0].text, len(o.outputs[0].token_ids)) for r, o in zip(batch, outputs)]

#!/usr/bin/env python3
"""
Score every verified pool trajectory under a Student checkpoint (one forward pass, no backward).

Per trajectory tau (trajectory tokens only, the end-of-turn token excluded):
  nll_sum    = sum_t -log p(tau_t | x, tau_<t)
  ell        = nll_sum / n_tokens            (length-normalized NLL; input of L_proxy, config "learnability")
  brier_sum  = sum_t || pi_t - onehot(tau_t) ||^2
               (Brier residual; stored so the LARK score, arXiv 2605.30651 Eq. 7, can be computed
                later without another pass. Exploratory, not part of the frozen L_proxy.)

L_proxy itself is not computed here: it needs ell over the whole pool (median / IQR rule) and
is derived in the analysis step.

--pool defaults to the official pool v1 (trajectory/pool/pool_v1_verified.jsonl). The Task 5 smoke test
scored the preliminary pool; reproduce it with --pool trajectory/pool/batch_100_teacher_verified.jsonl.

Usage (from experiments/week2September/manual/):
  python scripts/compute_nll.py --tag S0 --out reports/task5/smoke_test/nll_pool_S0.jsonl \

    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    t0 = time.time()
    args.out.parent.mkdir(parents=True, exist_ok=True)
    with args.out.open("w", encoding="utf-8") as f:
        for i in range(0, len(examples), args.batch_size):
            batch = examples[i: i + args.batch_size]
            width = max(len(e["input_ids"]) for e in batch)
            ids = torch.tensor([e["input_ids"] + [tok.pad_token_id] * (width - len(e["input_ids"])) for e in batch])
            mask = torch.tensor([[1] * len(e["input_ids"]) + [0] * (width - len(e["input_ids"])) for e in batch])
            labels = torch.tensor([e["labels"] + [IGNORE_INDEX] * (width - len(e["labels"])) for e in batch])
            ids, mask, labels = ids.to(model.device), mask.to(model.device), labels.to(model.device)

                    "trajectory_id": r["trajectory_id"], "problem_id": r["problem_id"], "checkpoint": args.tag,
                    "n_tokens": int(sel.sum()), "nll_sum": nll_sum, "ell": nll_sum / int(sel.sum()),
                    "brier_sum": brier,
                }) + "\n")
            print(f"[{args.tag}] {min(i + args.batch_size, len(examples))}/{len(examples)}  {time.time() - t0:.0f}s",
                  flush=True)
    elapsed = time.time() - t0
    write_json(args.out.with_suffix(".summary.json"), {
        "checkpoint": args.tag, "adapter": args.adapter, "model": model_name, "revision": revision,
        "dtype": str(dtype).replace("torch.", ""), "n_trajectories": len(examples),
        "pool": str(args.pool), "pool_sha256": sha256_file(args.pool),

        "peak_gpu_memory_gb": round(torch.cuda.max_memory_allocated() / 2**30, 2) if torch.cuda.is_available() else None,
        "environment": environment_info(torch),
    })
    print(f"[{args.tag}] scored {len(examples)} trajectories in {elapsed:.0f}s -> {args.out}")

    main()

#!/usr/bin/env python3
"""
Selection rules of B2, B3, B4, B5 and the shuffled-scores ablation (analysis_plan.md section 5,
task7_decisions.md T7-1, T7-2, T7-3). Pure Python: no torch, no files; select_round.py does the I/O.

Budget filling (protocol section 16.1, tokens(B_t) <= beta_round): walk the order and stop at the first
trajectory that would exceed the budget. Later, shorter trajectories are not used to top up, because that
would favour short trajectories over the method's own order. B3 and B4 keep the stopping trajectory for
the next round.

Tokens are response tokens under the Student tokenizer without the end-of-turn token, the unit of L_bar
and therefore of beta_round.

Seeds fixed by the config: B2 random.Random(training_seed * 100 + round); B3 random.Random(training_seed).
The ablation's seed is not in the config; ABLATION_SEED_OFFSET keeps it apart from B2.
"""
import random
from collections import Counter

from learnability import score_checkpoint

METHODS = ("B2", "B3", "B4", "B5", "ablation_shuffled_scores")
NEEDS_NLL = ("B4", "B5", "ablation_shuffled_scores")
ABLATION_SEED_OFFSET = 50
SELECTED_CRITERION = {"B2": "random", "B3": "shortest_correct", "B4": "top_score", "B5": "top_score",
                      "ablation_shuffled_scores": "shuffled_score"}


def fill_budget(order, problems, tokens, budget, n_cap, unavailable=()):
    """Returns (selected ids in order, {id: reason}, tokens used, whether the budget stopped the walk)."""
    unavailable = set(unavailable)
    selected, reasons, per_problem = [], {}, Counter()
    used, stopped = 0, False
    for tid in order:
        if tid in unavailable:
            reasons[tid] = "already_trained"
        elif stopped:
            reasons[tid] = "budget_exhausted"
        elif per_problem[problems[tid]] >= n_cap:
            reasons[tid] = "n_cap_reached"
        elif used + tokens[tid] > budget:
            stopped = True
            reasons[tid] = "budget_exhausted"
        else:
            selected.append(tid)
            reasons[tid] = "selected"
            per_problem[problems[tid]] += 1
            used += tokens[tid]
    return selected, reasons, used, stopped


def rank_by_score(scores):
    """Descending score, ties by trajectory_id ascending (decision #23)."""
    return sorted(scores, key=lambda tid: (-scores[tid], tid))


def random_order(ids, seed):
    order = sorted(ids)
    random.Random(seed).shuffle(order)
    return order


def shuffled_scores(scores, seed):
    """The same score values assigned to trajectories at random."""
    ids = sorted(scores)
    values = [scores[tid] for tid in ids]
    random.Random(seed).shuffle(values)
    return dict(zip(ids, values))


def shortest_per_problem(problems, tokens):
    """Decision #24: per problem the shortest trajectory, ties by smallest trajectory_id."""
    best = {}
    for tid in sorted(problems):
        pid = problems[tid]
        if pid not in best or tokens[tid] < tokens[best[pid]]:
            best[pid] = tid
    return best


def b3_order(problems, tokens, training_seed):
    """B3 order: the shortest trajectory of each problem, problems shuffled once [T7-2]."""
    best = shortest_per_problem(problems, tokens)
    return [best[pid] for pid in random_order(best, training_seed)]


def static_round(order, problems, tokens, budget, n_cap, round_index):
    """
    Round round_index of a once-computed order (B3, B4): walk it without repeating a trajectory; a
    trajectory skipped by n_cap or by the budget stays available for later rounds [T7-1, T7-2].
    """
    used_before = set()
    for r in range(round_index + 1):
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap, used_before)
        if r < round_index:
            used_before |= set(selected)
    return selected, reasons, used, stopped


def select(method, round_index, training_seed, problems, tokens, budget, n_cap, ell_by_id=None, trained_ids=()):
    """
    One round's selection over the eligible pool. Returns the selected ids, tokens used, band parameters
    and one log entry per eligible trajectory (fields of selection_log_schema.json, null where unused).
    ell_by_id: ell at S_t for B5 and the ablation, at S_0 for B4 in every round.
    trained_ids: trained in this run before this round (reference set of B5 and the ablation [T7-3]).
    """
    if method not in METHODS:
        raise ValueError(f"unknown method {method}")
    scoring, scores = None, None
    if method in NEEDS_NLL:
        if ell_by_id is None or set(ell_by_id) != set(problems):
            raise ValueError(f"{method} needs ell for exactly the eligible pool")
        # B4 ranks at S_0, where nothing has been trained: the reference set is the whole eligible pool.
        scoring = score_checkpoint(ell_by_id, () if method == "B4" else trained_ids)
        scores = scoring["l_proxy"]

    if method == "B2":
        order = random_order(problems, training_seed * 100 + round_index)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    elif method == "B3":
        order = b3_order(problems, tokens, training_seed)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    elif method == "B4":
        order = rank_by_score(scores)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    else:
        if method == "ablation_shuffled_scores":
            scores = shuffled_scores(scores, training_seed * 100 + round_index + ABLATION_SEED_OFFSET)
        order = rank_by_score(scores)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    if not stopped:
        raise ValueError(f"{method} round {round_index}: order exhausted before the budget ({used}/{budget} tokens)")

    ranks = {tid: i for i, tid in enumerate(order, start=1)}
    has_rank = method != "B2"
    rows = []
    for tid in sorted(problems):
        reason = reasons.get(tid, "not_shortest")
        rows.append({
            "problem_id": problems[tid],
            "trajectory_id": tid,
            "tokens": tokens[tid],
            "nll": ell_by_id[tid] if scoring else None,
            "ell_star": scoring["ell_star"] if scoring else None,
            "sigma": scoring["sigma"] if scoring else None,
            "l_proxy": scoring["l_proxy"][tid] if scoring else None,
            "score": scores[tid] if scores else None,
            "rank": ranks.get(tid) if has_rank else None,
            "selected": reason == "selected",
            "criterion": SELECTED_CRITERION[method] if reason == "selected" else reason,
            "outcome": None,
        })
    return {
        "selected": selected,
        "tokens_used": used,
        "ell_star": scoring["ell_star"] if scoring else None,
        "sigma": scoring["sigma"] if scoring else None,
        "n_reference": scoring["n_reference"] if scoring else None,
        "rows": rows,
    }


def probe_groups(tokens, group_size, seed, n_bins=5):
    """
    L_emp groups (config budget.l_emp_groups): token-length quintile bins of the eligible pool, group_size
    trajectories sampled per bin. n_cap is not applied: the rule does not ask for it.
    """
    ordered = sorted(tokens, key=lambda tid: (tokens[tid], tid))
    bins = [ordered[i * len(ordered) // n_bins: (i + 1) * len(ordered) // n_bins] for i in range(n_bins)]
    rng = random.Random(seed)
    return [sorted(rng.sample(b, group_size)) for b in bins]

#!/usr/bin/env python3
"""
L_proxy (analysis_plan.md section 4, config "learnability"). Pure Python: no torch, no files.

  ell*_t  = median of ell over the reference set at S_t
  sigma_t = IQR of ell over the reference set at S_t / 1.349
  L_proxy = exp(-(ell - ell*_t)^2 / (2 sigma_t^2)), computed for every eligible trajectory

Reference set [T7-3]: eligible trajectories not yet trained in this run (at S_0: all eligible).

Quartiles use statistics.quantiles(n=4) (method "exclusive"), the same call as the Task 5 smoke analysis
(analyze_smoke_test.py). The choice cannot change the ranking: L_proxy order depends only on |ell - ell*_t|.
"""
import math
import statistics


def band_parameters(ells):
    """(ell_star, sigma) of a list of ell values."""
    if len(ells) < 2:
        raise ValueError(f"reference set too small: {len(ells)}")
    if not all(math.isfinite(x) for x in ells):
        raise ValueError("non-finite ell in the reference set")
    q = statistics.quantiles(ells, n=4)
    sigma = (q[2] - q[0]) / 1.349
    if sigma <= 0:
        raise ValueError("IQR of ell is zero: L_proxy is undefined")
    return statistics.median(ells), sigma


def l_proxy(ell, ell_star, sigma):
    return math.exp(-((ell - ell_star) ** 2) / (2 * sigma ** 2))


def score_checkpoint(ell_by_id, trained_ids=()):
    """
    L_proxy of every trajectory in ell_by_id (the whole eligible pool at one checkpoint).
    trained_ids: trajectories already trained in this run; excluded from the reference set only.
    """
    unknown = set(trained_ids) - set(ell_by_id)
    if unknown:
        raise ValueError(f"trained trajectories missing from the scored pool: {sorted(unknown)[:3]}")
    trained = set(trained_ids)
    reference = [ell for tid, ell in ell_by_id.items() if tid not in trained]
    ell_star, sigma = band_parameters(reference)
    return {
        "ell_star": ell_star,
        "sigma": sigma,
        "n_reference": len(reference),
        "l_proxy": {tid: l_proxy(ell, ell_star, sigma) for tid, ell in ell_by_id.items()},
    }

def read_jsonl_resumable(path):
    """Rows of an append-only file; a line cut off by an interrupted run is dropped from the file."""
    path = Path(path)
    if not path.exists():
        return []
    lines = path.read_text(encoding="utf-8").splitlines(keepends=True)
    rows = []
    for i, line in enumerate(lines):
        try:
            rows.append(json.loads(line))
        except json.JSONDecodeError:
            if i != len(lines) - 1:
                raise
            path.write_text("".join(lines[:-1]), encoding="utf-8")
            print(f"[resume] dropped a truncated last line of {path}")
    return rows

def build_prompt(tokenizer, question):
    """Chat-formatted prompt v1, ending with the assistant generation header."""
    messages = [{"role": "user", "content": PROMPT_TEMPLATE.format(question=question)}]
    return tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)


def response_text(raw_generation):
    """Training/scoring target: the Teacher's raw output (REASONING: ... ANSWER: ...)."""
    return raw_generation.strip()


def encode_example(tokenizer, question, raw_generation, add_eos=True):
    """
    Token ids for prompt + response, labels masked (-100) on the prompt.

    Prompt and response are tokenized separately, which is exactly what the model sees at
    generation time. The end-of-turn token is appended to the supervised part when add_eos,
    so the Student learns to stop. n_traj counts only the trajectory tokens (no end-of-turn),
    matching L_bar in the config.
    """
    prompt_ids = tokenizer(build_prompt(tokenizer, question), add_special_tokens=False)["input_ids"]
    traj_ids = tokenizer(response_text(raw_generation), add_special_tokens=False)["input_ids"]
    resp_ids = traj_ids + ([end_of_turn_id(tokenizer)] if add_eos else [])
    input_ids = prompt_ids + resp_ids
    labels = [IGNORE_INDEX] * len(prompt_ids) + resp_ids
    return {
        "input_ids": input_ids,
        "labels": labels,
        "n_prompt": len(prompt_ids),
        "n_traj": len(traj_ids),
        "n_supervised": len(resp_ids),
    }


def end_of_turn_id(tokenizer):
    tid = tokenizer.convert_tokens_to_ids("<|im_end|>")
    if tid is None or tid == tokenizer.unk_token_id:
        raise ValueError("tokenizer has no <|im_end|> token")
    return tid


def pick_dtype(torch, requested="auto"):
    """bf16 when the GPU supports it (config), otherwise fp32 (fp16 is avoided: Qwen2.5 can overflow)."""
    if requested != "auto":
        return {"bf16": torch.bfloat16, "fp32": torch.float32, "fp16": torch.float16}[requested]
    if torch.cuda.is_available() and torch.cuda.is_bf16_supported():
        return torch.bfloat16
    return torch.float32


def load_tokenizer(model_name, revision):
    from transformers import AutoTokenizer
    tok = AutoTokenizer.from_pretrained(model_name, revision=revision)
    tok.padding_side = "left"
    if tok.pad_token is None:
        tok.pad_token = tok.eos_token
    return tok


def load_model(model_name, revision, dtype, adapter=None, merge=True):
    """Base Student at the pinned revision, optionally with a LoRA adapter (merged for speed)."""
    import torch
    from transformers import AutoModelForCausalLM
    model = AutoModelForCausalLM.from_pretrained(model_name, revision=revision, dtype=dtype)
    if adapter:
        from peft import PeftModel
        model = PeftModel.from_pretrained(model, adapter)
        if merge:
            model = model.merge_and_unload()
    model.to("cuda" if torch.cuda.is_available() else "cpu")
    model.eval()
    return model

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    try:
        value = json.loads(Path(path).read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise ValueError(f"Invalid JSON in {path}: {exc}") from exc
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    with Path(path).open("r", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(f"Invalid JSONL at {path}:{line_number}: {exc}") from exc
            if not isinstance(row, dict):
                raise ValueError(f"Expected JSON object at {path}:{line_number}")
            rows.append(row)
    return rows


def _require_file(path: Path, label: str) -> Path:

def load_pool(pool_path: Path, d_pool_path: Path) -> tuple[dict[str, dict], dict[str, str]]:
    pool_path = _require_file(pool_path, "Verified pool")
    d_pool_path = _require_file(d_pool_path, "D_pool ID file")
    d_pool_ids = {line.strip() for line in d_pool_path.read_text(encoding="utf-8").splitlines() if line.strip()}
    pool_by_id: dict[str, dict] = {}
    problems: dict[str, str] = {}
    for index, row in enumerate(read_jsonl(pool_path), 1):
        tid, pid = row.get("trajectory_id"), row.get("problem_id")
        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Pool row {index}: invalid trajectory_id")
        if not isinstance(pid, str) or not pid:
            raise ValueError(f"Pool row {index}: invalid problem_id")
        if tid in pool_by_id:
            raise ValueError(f"Duplicate trajectory_id in pool: {tid}")
        metadata = row.get("metadata")
        if not isinstance(metadata, dict) or metadata.get("verification_status") != "valid":
            raise ValueError(f"{tid}: metadata.verification_status must be valid")
        capabilities = row.get("capabilities")
        if not isinstance(capabilities, list) or not capabilities:
            raise ValueError(f"{tid}: capability list is empty or invalid")
        if any(not isinstance(tag, str) for tag in capabilities):
            raise ValueError(f"{tid}: capability tags must be strings")
        if len(capabilities) != len(set(capabilities)):
            raise ValueError(f"{tid}: duplicate capability tags")
        unknown = sorted(set(capabilities) - set(TAXONOMY))
        if unknown:
            raise ValueError(f"{tid}: unknown capability tags: {unknown}")
        if pid not in d_pool_ids:
            raise ValueError(f"{tid}: problem_id is outside D_pool: {pid}")
        pool_by_id[tid] = row
        problems[tid] = pid
    if not pool_by_id:
        raise ValueError("Verified pool is empty")
    return pool_by_id, problems


def load_tokens(tokens_path: Path, pool_path: Path, problems: dict[str, str]) -> dict[str, int]:
    tokens_path = _require_file(tokens_path, "Token table")
    pool_path = _require_file(pool_path, "Verified pool")
    summary_path = tokens_path.with_suffix(".summary.json")
    summary = read_json(_require_file(summary_path, "Token summary"))
    actual_pool_hash = sha256_file(pool_path)
    if summary.get("pool_sha256") != actual_pool_hash:
        raise ValueError("Token summary pool_sha256 does not match the supplied pool file")
    rows = read_jsonl(tokens_path)
    tokens: dict[str, int] = {}
    for index, row in enumerate(rows, 1):
        tid, pid, count = row.get("trajectory_id"), row.get("problem_id"), row.get("response_tokens")
        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Token row {index}: invalid trajectory_id")
        if tid in tokens:
            raise ValueError(f"Duplicate token row: {tid}")
        if tid not in problems:
            raise ValueError(f"Token row not present in verified pool: {tid}")
        if problems[tid] != pid:
            raise ValueError(f"Token problem_id mismatch: {tid}")
        if not isinstance(count, int) or isinstance(count, bool) or count <= 0:
            raise ValueError(f"Invalid response_tokens for {tid}: {count}")
        tokens[tid] = count
    if set(tokens) != set(problems):
        missing, extra = sorted(set(problems) - set(tokens)), sorted(set(tokens) - set(problems))
        raise ValueError(f"Pool/token IDs differ. Missing={missing[:5]}, extra={extra[:5]}")
    if len(rows) != summary.get("n_trajectories"):
        raise ValueError("Token summary n_trajectories mismatch")
    if sum(tokens.values()) != summary.get("response_tokens_sum"):
        raise ValueError("Token summary response_tokens_sum mismatch")
    return tokens


def load_capability_deficits(state_path: Path, method: str, seed: int, round_number: int) -> dict[str, float]:
    state = read_json(_require_file(state_path, "Capability state"))
    if state.get("schema_version") != "p1p2.capability-state.v1":
        raise ValueError("Unsupported capability state schema")
    target = state.get("target")
    if not isinstance(target, dict):
        raise ValueError("Capability state target is missing or invalid")
    expected = {"method": method, "seed": seed, "selection_round": round_number}
    for key, value in expected.items():
        if target.get(key) != value:
            raise ValueError(f"State target {key} mismatch: expected {value}, got {target.get(key)}")
    capabilities = state.get("capabilities")
    if not isinstance(capabilities, dict):
        raise ValueError("Capability state has no capabilities object")
    deficits: dict[str, float] = {}
    for capability in TAXONOMY:
        item = capabilities.get(capability)
        if not isinstance(item, dict):
            raise ValueError(f"Missing capability state: {capability}")
        value = item.get("deficit")
        if not isinstance(value, (int, float)) or isinstance(value, bool) or not math.isfinite(value) or not 0 <= value <= 1:
            raise ValueError(f"Invalid deficit for {capability}: {value}")
        deficits[capability] = float(value)
    return deficits


def load_previous_train_ids(paths: list[Path], problems: dict[str, str], method: str, seed: int,
                            current_round: int, pool_sha256: str) -> list[str]:
    trained: set[str] = set()
    seen_rounds: set[int] = set()
    for path in paths:
        data = read_json(_require_file(path, "Previous train set"))
        if data.get("schema_version") != "p1p2.train-set.v1":
            raise ValueError(f"Unsupported previous train-set schema: {path}")
        if data.get("method") != method:
            raise ValueError(f"Previous train-set method mismatch: {path}")
        if data.get("training_seed") != seed:
            raise ValueError(f"Previous train-set seed mismatch: {path}")
        previous_round = data.get("round")
        if not isinstance(previous_round, int) or isinstance(previous_round, bool) or not 1 <= previous_round < current_round:
            raise ValueError(f"Previous train-set round must be in [1, {current_round - 1}]: {path}")
        if previous_round in seen_rounds:
            raise ValueError(f"Duplicate previous train-set round {previous_round}")
        seen_rounds.add(previous_round)
        if data.get("pool_sha256") != pool_sha256:
            raise ValueError(f"Previous train-set pool hash mismatch: {path}")
        ids = data.get("trajectory_ids")
        if not isinstance(ids, list) or not ids:
            raise ValueError(f"Invalid or empty trajectory_ids in {path}")
        for tid in ids:
            if not isinstance(tid, str) or tid not in problems:
                raise ValueError(f"Previous train set contains unknown trajectory: {tid}")
            trained.add(tid)
    expected_rounds = set(range(1, current_round))
    if seen_rounds != expected_rounds:
        raise ValueError(f"Previous train-set rounds incomplete: expected {sorted(expected_rounds)}, got {sorted(seen_rounds)}")
    return sorted(trained)


def load_nll(nll_path: Path, tokens: dict[str, int], source_checkpoint: str,
             pool_path: Path) -> tuple[dict[str, float], str]:
    nll_path = _require_file(nll_path, "NLL file")

    ell_by_id: dict[str, float] = {}
    for index, row in enumerate(read_jsonl(nll_path), 1):
        tid = row.get("trajectory_id")
        if not isinstance(tid, str) or not tid:
            raise ValueError(f"NLL row {index}: invalid trajectory_id")
        if tid in ell_by_id:
            raise ValueError(f"Duplicate NLL trajectory_id: {tid}")
        if tid not in tokens:
            raise ValueError(f"NLL trajectory not present in token table: {tid}")
        if row.get("checkpoint") != source_checkpoint:
            raise ValueError(f"NLL checkpoint mismatch for {tid}: expected {source_checkpoint}, got {row.get('checkpoint')}")
        if row.get("n_tokens") != tokens[tid]:
            raise ValueError(f"NLL token count mismatch for {tid}")
        ell = row.get("ell")
        if not isinstance(ell, (int, float)) or isinstance(ell, bool) or not math.isfinite(ell):
            raise ValueError(f"Invalid ell value for {tid}: {ell}")
        ell_by_id[tid] = float(ell)
    if set(ell_by_id) != set(tokens):
        missing, extra = sorted(set(tokens) - set(ell_by_id)), sorted(set(ell_by_id) - set(tokens))

        raise ValueError("NLL summary pool_sha256 mismatch against supplied pool")
    if summary.get("n_trajectories") != len(tokens):
        raise ValueError("NLL summary trajectory count mismatch")

def average_percentile_ranks(scores: dict[str, float]) -> dict[str, float]:
    if not scores:
        raise ValueError("Cannot rank an empty score mapping")
    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    n = len(ordered)
    if n == 1:
        return {ordered[0][0]: 0.5}
    result: dict[str, float] = {}
    index = 0
    while index < n:
        end = index + 1
        value = ordered[index][1]
        while end < n and ordered[end][1] == value:
            end += 1
        average_rank = ((index + 1) + end) / 2.0
        percentile = (average_rank - 1.0) / (n - 1.0)
        for position in range(index, end):
            result[ordered[position][0]] = percentile
        index = end
    return result


def calculate_p1_scores(pool_by_id: dict[str, dict], deficits: dict[str, float]) -> dict[str, float]:
    result: dict[str, float] = {}
    for tid, row in pool_by_id.items():
        tags = row["capabilities"]
        result[tid] = sum(deficits[tag] for tag in tags) / len(tags)
    return result


def publish_pair_no_overwrite(train_set_path: Path, selection_log_path: Path,
                              train_set_bytes: bytes, selection_log_bytes: bytes) -> None:
    train_set_path, selection_log_path = Path(train_set_path), Path(selection_log_path)
    if train_set_path.resolve() == selection_log_path.resolve():
        raise ValueError("Train-set and selection-log paths must be different")
    train_set_path.parent.mkdir(parents=True, exist_ok=True)
    selection_log_path.parent.mkdir(parents=True, exist_ok=True)
    if train_set_path.exists() or selection_log_path.exists():
        raise FileExistsError("Refusing to overwrite an existing selector output")
    nonce = f"{os.getpid()}"
    train_tmp = train_set_path.with_name(train_set_path.name + f".{nonce}.tmp")
    log_tmp = selection_log_path.with_name(selection_log_path.name + f".{nonce}.tmp")
    created_train = False
    created_temps: list[Path] = []
    try:
        for temp_path, content in ((train_tmp, train_set_bytes), (log_tmp, selection_log_bytes)):
            flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
            fd = os.open(str(temp_path), flags, 0o600)
            created_temps.append(temp_path)
            with os.fdopen(fd, "wb") as handle:
                handle.write(content)
                handle.flush()
                os.fsync(handle.fileno())
        os.link(train_tmp, train_set_path)
        created_train = True
        os.link(log_tmp, selection_log_path)
    except Exception:
        if created_train and train_set_path.exists() and sha256_file(train_set_path) == hashlib.sha256(train_set_bytes).hexdigest():
            train_set_path.unlink()
        raise
    finally:
        for temp_path in created_temps:
            temp_path.unlink(missing_ok=True)

def run_selection(args: argparse.Namespace) -> dict[str, Any]:
    if args.seed < 0:
        raise ValueError("seed must be non-negative")
    if args.method == "P1-DEFICIT" and args.nll is not None:
        raise ValueError("P1-DEFICIT does not accept an NLL input")
    if args.method == "P1P2" and args.nll is None:
        raise ValueError("P1P2 requires --nll for the current source checkpoint")
    source_checkpoint = f"S{args.round - 1}"
    config_path = _require_file(args.config, "Frozen experiment config")
    config = read_json(config_path)
    selection_cfg = config.get("selection")
    budget_cfg = config.get("budget")
    if not isinstance(selection_cfg, dict) or not isinstance(budget_cfg, dict):
        raise ValueError("Frozen config is missing selection or budget section")
    alpha = selection_cfg.get("alpha")
    alpha_tuned = selection_cfg.get("alpha_tuned")
    if alpha != EXPECTED_ALPHA or alpha_tuned is not False:
        raise ValueError(f"Frozen alpha policy mismatch: expected alpha={EXPECTED_ALPHA}, alpha_tuned=false; got {alpha}, {alpha_tuned}")
    budget = budget_cfg.get("beta_round_tokens")
    n_cap = selection_cfg.get("n_cap")
    if not isinstance(budget, int) or isinstance(budget, bool) or budget <= 0:
        raise ValueError("Invalid beta_round_tokens in frozen config")
    if n_cap != 1:
        raise ValueError(f"Frozen protocol requires n_cap=1, found {n_cap}")
    if selection_cfg.get("allow_reselect_across_rounds") is not True:
        raise ValueError("Frozen protocol requires allow_reselect_across_rounds=true")

    pool_path = _require_file(args.pool, "Verified pool")
    tokens_path = _require_file(args.tokens, "Token table")
    pool_hash = sha256_file(pool_path)
    pool_by_id, problems = load_pool(pool_path, args.d_pool)
    tokens = load_tokens(tokens_path, pool_path, problems)
    deficits = load_capability_deficits(args.state, args.method, args.seed, args.round)
    trained_ids = load_previous_train_ids(args.previous_train_set, problems, args.method, args.seed, args.round, pool_hash)
    p1_scores = calculate_p1_scores(pool_by_id, deficits)
    p2_scores = None

def validate_metrics(metrics_path: Path) -> dict[str, dict[str, Any]]:
    by_capability: dict[str, dict[str, Any]] = {}

    with metrics_path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue

            row = json.loads(line)
            if not isinstance(row, dict):
                raise ValueError(f"Invalid metric row at line {line_no}")

            capability = row.get("capability")
            if capability not in CAPABILITIES:
                raise ValueError(f"Unknown capability: {capability!r}")

            if capability in by_capability:
                raise ValueError(f"Duplicate capability row: {capability}")

            if row.get("strict_metric") != "status_equals_valid":
                raise ValueError("Capability row does not use strict metric")

            if row.get("gold_policy") != "approved_85_primary":
                raise ValueError("Capability row uses an invalid Gold policy")

            support = int(row["n_support"])
            correct = int(row["n_correct"])
            incorrect = int(row["n_incorrect"])

            if support != EXPECTED_SUPPORT[capability]:
                raise ValueError(
                    f"Support mismatch for {capability}: "
                    f"{support} != {EXPECTED_SUPPORT[capability]}"
                )

            if correct < 0 or incorrect < 0:
                raise ValueError(f"Negative metric count: {capability}")

            if correct + incorrect != support:
                raise ValueError(f"Count mismatch: {capability}")

            mastery = correct / support
            deficit = 1.0 - mastery

            if not math.isfinite(float(row["mastery"])):
                raise ValueError(f"Non-finite mastery: {capability}")

            if not math.isfinite(float(row["deficit"])):
                raise ValueError(f"Non-finite deficit: {capability}")

            if abs(float(row["mastery"]) - mastery) > 1e-12:
                raise ValueError(f"Mastery mismatch: {capability}")

            if abs(float(row["deficit"]) - deficit) > 1e-12:
                raise ValueError(f"Deficit mismatch: {capability}")

            by_capability[capability] = {
                "n_support": support,
                "n_correct": correct,
                "n_incorrect": incorrect,
                "mastery": mastery,
                "deficit": deficit,
                "ci95_low": row.get("ci95_low"),
                "ci95_high": row.get("ci95_high"),
            }

    if set(by_capability) != set(CAPABILITIES):
        raise ValueError(
            "Capability coverage is incomplete: "
            f"{sorted(by_capability)}"
        )

    return by_capability


def publish_exclusive(output: Path, content: bytes) -> None:
    """Publish through a same-directory hard link; never replace a target."""
    output.parent.mkdir(parents=True, exist_ok=True)

    fd, temp_name = tempfile.mkstemp(
        prefix=f".{output.name}.staging.",
        suffix=".tmp",
        dir=str(output.parent),
    )
    temp_path = Path(temp_name)

    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(content)
            stream.flush()
            os.fsync(stream.fileno())

        try:
            os.link(temp_path, output)
        except FileExistsError:
            raise
        except OSError as exc:
            raise RuntimeError(
                "ATOMIC_NO_OVERWRITE_UNAVAILABLE: hard-link publication "
                f"failed; temporary file preserved at {temp_path}: {exc}"
            ) from exc

        temp_path.unlink(missing_ok=True)

    except Exception:
        if temp_path.exists():
            print(f"RECOVERY_TEMP_PRESERVED={temp_path}")
        raise

def sha256_file(path: Path) -> str:
    """Return the SHA-256 digest of a regular file."""
    source = Path(path).resolve()
    if not source.is_file():
        raise FileNotFoundError(f"Input file not found: {source}")
    digest = hashlib.sha256()
    with source.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def canonical_json_bytes(value: Any) -> bytes:
    """Serialize JSON deterministically for stable fingerprints."""
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n").encode("utf-8")


def sha256_json(value: Any) -> str:
    return hashlib.sha256(canonical_json_bytes(value)).hexdigest()


def fingerprint_files(paths: Mapping[str, Path]) -> dict[str, Any]:
    """Fingerprint named input files; names are stable logical identifiers."""
    if not paths:
        raise ValueError("At least one input file is required for a fingerprint")
    entries: dict[str, dict[str, Any]] = {}
    resolved_seen: dict[str, str] = {}
    for name in sorted(paths):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Fingerprint input names must be non-empty strings")
        path = Path(paths[name]).resolve()
        if not path.is_file():
            raise FileNotFoundError(f"Fingerprint input '{name}' not found: {path}")
        normalized = str(path).casefold()
        if normalized in resolved_seen:
            raise ValueError(f"Duplicate fingerprint path for '{name}' and '{resolved_seen[normalized]}'")
        resolved_seen[normalized] = name
        entries[name] = {
            "path": str(path),
            "size_bytes": path.stat().st_size,
            "sha256": sha256_file(path),
        }
    payload = {"schema_version": "p1p2.input-fingerprint.v1", "files": entries}
    payload["fingerprint_sha256"] = sha256_json(payload)
    return payload


def verify_fingerprint(fingerprint: Mapping[str, Any]) -> bool:
    """Verify a fingerprint object's internal canonical digest."""
    if not isinstance(fingerprint, Mapping):
        raise TypeError("Fingerprint must be a mapping")
    if fingerprint.get("schema_version") != "p1p2.input-fingerprint.v1":
        raise ValueError("Unsupported fingerprint schema")
    files = fingerprint.get("files")
    digest = fingerprint.get("fingerprint_sha256")
    if not isinstance(files, Mapping) or not isinstance(digest, str):
        raise ValueError("Fingerprint is missing files or fingerprint_sha256")
    unsigned = {key: value for key, value in fingerprint.items() if key != "fingerprint_sha256"}
    actual = sha256_json(unsigned)
    if actual != digest:
        raise ValueError("Fingerprint digest mismatch")
    return True


def verify_fingerprint_files(fingerprint: Mapping[str, Any]) -> bool:
    """Verify that every recorded file still has its recorded size and hash."""
    verify_fingerprint(fingerprint)
    files = fingerprint["files"]
    for name, entry in files.items():
        if not isinstance(entry, Mapping):
            raise ValueError(f"Invalid fingerprint entry: {name}")
        path = Path(entry.get("path", ""))
        if not path.is_file():
            raise FileNotFoundError(f"Fingerprint source missing: {name}: {path}")
        if path.stat().st_size != entry.get("size_bytes"):
            raise ValueError(f"Fingerprint size mismatch: {name}")
        if sha256_file(path) != entry.get("sha256"):
            raise ValueError(f"Fingerprint SHA-256 mismatch: {name}")
    return True


def ensure_no_overwrite(paths: Iterable[Path]) -> list[Path]:
    """Resolve and preflight output paths without creating or changing them."""
    resolved: list[Path] = []
    seen: set[str] = set()
    for raw_path in paths:
        path = Path(raw_path).resolve()
        key = str(path).casefold()
        if key in seen:
            raise ValueError(f"Duplicate output path: {path}")
        seen.add(key)
        if path.exists():
            raise FileExistsError(f"Refusing to overwrite existing artifact: {path}")
        resolved.append(path)
    if not resolved:
        raise ValueError("At least one output path is required")
    return resolved


def _write_temp_for_target(target: Path, content: bytes) -> Path:
    target.parent.mkdir(parents=True, exist_ok=True)
    fd, raw_temp = tempfile.mkstemp(prefix=f".{target.name}.", suffix=".tmp", dir=str(target.parent))
    temp_path = Path(raw_temp)
    try:
        with os.fdopen(fd, "wb") as handle:
            handle.write(content)
            handle.flush()
            os.fsync(handle.fileno())
    except Exception:
        temp_path.unlink(missing_ok=True)
        raise
    return temp_path


def publish_bytes_no_overwrite(target: Path, content: bytes) -> Path:
    """Atomically publish one file without replacing an existing path."""
    if not isinstance(content, bytes):
        raise TypeError("content must be bytes")
    destination = Path(target).resolve()
    ensure_no_overwrite([destination])
    temp_path = _write_temp_for_target(destination, content)
    try:
        # A same-filesystem hard link is atomic and fails if destination exists.
        os.link(temp_path, destination)
    finally:
        temp_path.unlink(missing_ok=True)
    return destination


def publish_json_no_overwrite(target: Path, value: Any) -> Path:

def publish_pair_no_overwrite(first_path: Path, first_content: bytes,
                              second_path: Path, second_content: bytes) -> tuple[Path, Path]:
    """Publish two files; roll back the first if publishing the second fails."""
    first = Path(first_path).resolve()
    second = Path(second_path).resolve()
    ensure_no_overwrite([first, second])
    first_temp = _write_temp_for_target(first, first_content)
    second_temp: Path | None = None
    first_published = False
    try:
        second_temp = _write_temp_for_target(second, second_content)
        os.link(first_temp, first)
        first_published = True
        os.link(second_temp, second)
    except Exception:
        if first_published and first.exists():
            expected = hashlib.sha256(first_content).hexdigest()
            if sha256_file(first) == expected:
                first.unlink()
        raise
    finally:
        first_temp.unlink(missing_ok=True)
        if second_temp is not None:
            second_temp.unlink(missing_ok=True)
    return first, second

        raise ValueError("manifest must not supply input_fingerprint directly")
    payload["input_fingerprint"] = fingerprint_files(input_paths)
    return publish_json_no_overwrite(target, payload)

def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish a completed staging directory without overwriting a target.

    the staging directory is intentionally preserved for recovery.

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def hash_path(path: Path) -> str:
    path = path.resolve()

    if path.is_file():
        return sha256_file(path)

    if not path.is_dir():
        raise FileNotFoundError(f"Input does not exist: {path}")

    files = sorted(p for p in path.rglob("*") if p.is_file())
    if not files:
        raise ValueError(f"Checkpoint directory is empty: {path}")

    digest = hashlib.sha256()
    for file_path in files:
        relative = file_path.relative_to(path).as_posix()
        digest.update(relative.encode("utf-8"))
        digest.update(b"\0")
        digest.update(sha256_file(file_path).encode("ascii"))
        digest.update(b"\n")

    return digest.hexdigest()

    rows: list[dict[str, Any]] = []

def wilson_interval(
    correct: int,
    support: int,
    z: float = 1.959963984540054,
) -> tuple[float | None, float | None]:
    if support <= 0:
        return None, None

    p = correct / support
    z2 = z * z
    denominator = 1.0 + z2 / support
    center = (p + z2 / (2.0 * support)) / denominator
    margin = (
        z
        * (
            p * (1.0 - p) / support
            + z2 / (4.0 * support * support)
        ) ** 0.5
        / denominator
    )

    return max(0.0, center - margin), min(1.0, center + margin)

def verify_existing_sidecar(
    out_dir: Path,
    input_hashes: dict[str, str],
) -> None:
    manifest_path = (
        out_dir / "student_capability_evaluation_manifest.json"
    )

    if not manifest_path.is_file():
        raise RuntimeError(
            "EXISTING_PARTIAL_SIDECAR: manifest missing; "
            "do not overwrite or auto-delete this directory"
        )

    manifest = read_json(manifest_path)

    if manifest.get("input_hashes") != input_hashes:
        raise RuntimeError(
            "INPUT_FINGERPRINT_MISMATCH: sidecar is immutable"
        )

    outputs = manifest.get("outputs", {})
    expected_names = set(REQUIRED_OUTPUTS) - {
        "student_capability_evaluation_manifest.json"
    }

    if set(outputs) != expected_names:
        raise RuntimeError("Existing sidecar output inventory mismatch")

    for name, metadata in outputs.items():
        path = out_dir / name

        if not path.is_file():
            raise RuntimeError(f"Existing output missing: {path}")

        if sha256_file(path) != metadata.get("sha256"):
            raise RuntimeError(f"Existing output hash mismatch: {path}")

    print(f"REUSE: PASS {out_dir}")

def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish without overwrite; retry transient Windows permission failures."""
    retry_delays = (0.25, 0.5, 1.0, 2.0, 4.0)

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)
            return
        except PermissionError as exc:
            if out_dir.exists():
                raise FileExistsError(
                    "OUTPUT_COLLISION: destination appeared during "
                    "publication; refusing to overwrite: " + str(out_dir)
                ) from exc

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)


def main() -> None:
    args = parse_args()

    input_hashes = build_input_hashes(args)
    out_dir = Path(args.out_root) / args.run_id

    if out_dir.exists():
        verify_existing_sidecar(out_dir, input_hashes)
        return

    out_dir.parent.mkdir(parents=True, exist_ok=True)

    stage = Path(tempfile.mkdtemp(
        prefix=f".{args.run_id}.staging.",
        dir=out_dir.parent,
    ))

    try:
        build_sidecar(args, stage, input_hashes)
        publish_staged_directory(stage, out_dir)
    except Exception as exc:
        if stage.exists():
            print(f"RECOVERY_STAGING_PRESERVED={stage}")
            print(f"RECOVERY_ERROR={type(exc).__name__}: {exc}")
        raise

    print(f"CAPABILITY_EVAL: PASS {out_dir}")


if __name__ == "__main__":
    main()

usage: compute_nll.py [-h] --tag TAG [--adapter ADAPTER] --out OUT
                      [--batch-size BATCH_SIZE] [--limit LIMIT] [--pool POOL]
                      [--model MODEL] [--revision REVISION]
                      [--dtype {auto,bf16,fp32,fp16}]

Score every verified pool trajectory under a Student checkpoint (one forward pass, no backward).

Per trajectory tau (trajectory tokens only, the end-of-turn token excluded):
  nll_sum    = sum_t -log p(tau_t | x, tau_<t)
  ell        = nll_sum / n_tokens            (length-normalized NLL; input of L_proxy, config "learnability")
  brier_sum  = sum_t || pi_t - onehot(tau_t) ||^2
               (Brier residual; stored so the LARK score, arXiv 2605.30651 Eq. 7, can be computed
                later without another pass. Exploratory, not part of the frozen L_proxy.)

L_proxy itself is not computed here: it needs ell over the whole pool (median / IQR rule) and
is derived in the analysis step.

--pool defaults to the official pool v1 (trajectory/pool/pool_v1_verified.jsonl). The Task 5 smoke test
scored the preliminary pool; reproduce it with --pool trajectory/pool/batch_100_teacher_verified.jsonl.
