#!/usr/bin/env python3
"""
Selection rules of B2, B3, B4, B5 and the shuffled-scores ablation (analysis_plan.md section 5,
task7_decisions.md T7-1, T7-2, T7-3). Pure Python: no torch, no files; select_round.py does the I/O.

Budget filling (protocol section 16.1, tokens(B_t) <= beta_round): walk the order and stop at the first
trajectory that would exceed the budget. Later, shorter trajectories are not used to top up, because that
would favour short trajectories over the method's own order. B3 and B4 keep the stopping trajectory for
the next round.

Tokens are response tokens under the Student tokenizer without the end-of-turn token, the unit of L_bar
and therefore of beta_round.

Seeds fixed by the config: B2 random.Random(training_seed * 100 + round); B3 random.Random(training_seed).
The ablation's seed is not in the config; ABLATION_SEED_OFFSET keeps it apart from B2.
"""
import random
from collections import Counter

from learnability import score_checkpoint

METHODS = ("B2", "B3", "B4", "B5", "ablation_shuffled_scores")
NEEDS_NLL = ("B4", "B5", "ablation_shuffled_scores")
ABLATION_SEED_OFFSET = 50
SELECTED_CRITERION = {"B2": "random", "B3": "shortest_correct", "B4": "top_score", "B5": "top_score",
                      "ablation_shuffled_scores": "shuffled_score"}


def fill_budget(order, problems, tokens, budget, n_cap, unavailable=()):
    """Returns (selected ids in order, {id: reason}, tokens used, whether the budget stopped the walk)."""
    unavailable = set(unavailable)
    selected, reasons, per_problem = [], {}, Counter()
    used, stopped = 0, False
    for tid in order:
        if tid in unavailable:
            reasons[tid] = "already_trained"
        elif stopped:
            reasons[tid] = "budget_exhausted"
        elif per_problem[problems[tid]] >= n_cap:
            reasons[tid] = "n_cap_reached"
        elif used + tokens[tid] > budget:
            stopped = True
            reasons[tid] = "budget_exhausted"
        else:
            selected.append(tid)
            reasons[tid] = "selected"
            per_problem[problems[tid]] += 1
            used += tokens[tid]
    return selected, reasons, used, stopped


def rank_by_score(scores):
    """Descending score, ties by trajectory_id ascending (decision #23)."""
    return sorted(scores, key=lambda tid: (-scores[tid], tid))


def random_order(ids, seed):
    order = sorted(ids)
    random.Random(seed).shuffle(order)
    return order


def shuffled_scores(scores, seed):
    """The same score values assigned to trajectories at random."""
    ids = sorted(scores)
    values = [scores[tid] for tid in ids]
    random.Random(seed).shuffle(values)
    return dict(zip(ids, values))


def shortest_per_problem(problems, tokens):
    """Decision #24: per problem the shortest trajectory, ties by smallest trajectory_id."""
    best = {}
    for tid in sorted(problems):
        pid = problems[tid]
        if pid not in best or tokens[tid] < tokens[best[pid]]:
            best[pid] = tid
    return best


def b3_order(problems, tokens, training_seed):
    """B3 order: the shortest trajectory of each problem, problems shuffled once [T7-2]."""
    best = shortest_per_problem(problems, tokens)
    return [best[pid] for pid in random_order(best, training_seed)]


def static_round(order, problems, tokens, budget, n_cap, round_index):
    """
    Round round_index of a once-computed order (B3, B4): walk it without repeating a trajectory; a
    trajectory skipped by n_cap or by the budget stays available for later rounds [T7-1, T7-2].
    """
    used_before = set()
    for r in range(round_index + 1):
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap, used_before)
        if r < round_index:
            used_before |= set(selected)
    return selected, reasons, used, stopped


def select(method, round_index, training_seed, problems, tokens, budget, n_cap, ell_by_id=None, trained_ids=()):
    """
    One round's selection over the eligible pool. Returns the selected ids, tokens used, band parameters
    and one log entry per eligible trajectory (fields of selection_log_schema.json, null where unused).
    ell_by_id: ell at S_t for B5 and the ablation, at S_0 for B4 in every round.
    trained_ids: trained in this run before this round (reference set of B5 and the ablation [T7-3]).
    """
    if method not in METHODS:
        raise ValueError(f"unknown method {method}")
    scoring, scores = None, None
    if method in NEEDS_NLL:
        if ell_by_id is None or set(ell_by_id) != set(problems):
            raise ValueError(f"{method} needs ell for exactly the eligible pool")
        # B4 ranks at S_0, where nothing has been trained: the reference set is the whole eligible pool.
        scoring = score_checkpoint(ell_by_id, () if method == "B4" else trained_ids)
        scores = scoring["l_proxy"]

    if method == "B2":
        order = random_order(problems, training_seed * 100 + round_index)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    elif method == "B3":
        order = b3_order(problems, tokens, training_seed)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    elif method == "B4":
        order = rank_by_score(scores)
        selected, reasons, used, stopped = static_round(order, problems, tokens, budget, n_cap, round_index)
    else:
        if method == "ablation_shuffled_scores":
            scores = shuffled_scores(scores, training_seed * 100 + round_index + ABLATION_SEED_OFFSET)
        order = rank_by_score(scores)
        selected, reasons, used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    if not stopped:
        raise ValueError(f"{method} round {round_index}: order exhausted before the budget ({used}/{budget} tokens)")

    ranks = {tid: i for i, tid in enumerate(order, start=1)}
    has_rank = method != "B2"
    rows = []
    for tid in sorted(problems):
        reason = reasons.get(tid, "not_shortest")
        rows.append({
            "problem_id": problems[tid],
            "trajectory_id": tid,
            "tokens": tokens[tid],
            "nll": ell_by_id[tid] if scoring else None,
            "ell_star": scoring["ell_star"] if scoring else None,
            "sigma": scoring["sigma"] if scoring else None,
            "l_proxy": scoring["l_proxy"][tid] if scoring else None,
            "score": scores[tid] if scores else None,
            "rank": ranks.get(tid) if has_rank else None,
            "selected": reason == "selected",
            "criterion": SELECTED_CRITERION[method] if reason == "selected" else reason,
            "outcome": None,
        })
    return {
        "selected": selected,
        "tokens_used": used,
        "ell_star": scoring["ell_star"] if scoring else None,
        "sigma": scoring["sigma"] if scoring else None,
        "n_reference": scoring["n_reference"] if scoring else None,
        "rows": rows,
    }


def probe_groups(tokens, group_size, seed, n_bins=5):
    """
    L_emp groups (config budget.l_emp_groups): token-length quintile bins of the eligible pool, group_size
    trajectories sampled per bin. n_cap is not applied: the rule does not ask for it.
    """
    ordered = sorted(tokens, key=lambda tid: (tokens[tid], tid))
    bins = [ordered[i * len(ordered) // n_bins: (i + 1) * len(ordered) // n_bins] for i in range(n_bins)]
    rng = random.Random(seed)
    return [sorted(rng.sample(b, group_size)) for b in bins]

#!/usr/bin/env python3
"""
L_proxy (analysis_plan.md section 4, config "learnability"). Pure Python: no torch, no files.

  ell*_t  = median of ell over the reference set at S_t
  sigma_t = IQR of ell over the reference set at S_t / 1.349
  L_proxy = exp(-(ell - ell*_t)^2 / (2 sigma_t^2)), computed for every eligible trajectory

Reference set [T7-3]: eligible trajectories not yet trained in this run (at S_0: all eligible).

Quartiles use statistics.quantiles(n=4) (method "exclusive"), the same call as the Task 5 smoke analysis
(analyze_smoke_test.py). The choice cannot change the ranking: L_proxy order depends only on |ell - ell*_t|.
"""
import math
import statistics


def band_parameters(ells):
    """(ell_star, sigma) of a list of ell values."""
    if len(ells) < 2:
        raise ValueError(f"reference set too small: {len(ells)}")
    if not all(math.isfinite(x) for x in ells):
        raise ValueError("non-finite ell in the reference set")
    q = statistics.quantiles(ells, n=4)
    sigma = (q[2] - q[0]) / 1.349
    if sigma <= 0:
        raise ValueError("IQR of ell is zero: L_proxy is undefined")
    return statistics.median(ells), sigma


def l_proxy(ell, ell_star, sigma):
    return math.exp(-((ell - ell_star) ** 2) / (2 * sigma ** 2))


def score_checkpoint(ell_by_id, trained_ids=()):
    """
    L_proxy of every trajectory in ell_by_id (the whole eligible pool at one checkpoint).
    trained_ids: trajectories already trained in this run; excluded from the reference set only.
    """
    unknown = set(trained_ids) - set(ell_by_id)
    if unknown:
        raise ValueError(f"trained trajectories missing from the scored pool: {sorted(unknown)[:3]}")
    trained = set(trained_ids)
    reference = [ell for tid, ell in ell_by_id.items() if tid not in trained]
    ell_star, sigma = band_parameters(reference)
    return {
        "ell_star": ell_star,
        "sigma": sigma,
        "n_reference": len(reference),
        "l_proxy": {tid: l_proxy(ell, ell_star, sigma) for tid, ell in ell_by_id.items()},
    }

Writes (atomically):

  <run-dir>/round_NNN/selection_log.jsonl  one record per eligible trajectory (selection_log_schema.json)

--pilot allows a smaller budget and seeds outside the config; the output is marked and is not a result.

Usage (from experiments/week2September/manual/):

def round_dir(run_dir, t):

def rel(path):
    resolved = Path(path).resolve()
    return str(resolved.relative_to(MANUAL_DIR)) if resolved.is_relative_to(MANUAL_DIR) else str(resolved)


def write_atomic(path, text):
    tmp = path.with_name(path.name + ".tmp")
    tmp.write_text(text, encoding="utf-8")
    os.replace(tmp, path)


def load_tokens(tokens_path, pool_path):
    summary = json.loads(Path(tokens_path).with_suffix(".summary.json").read_text(encoding="utf-8"))
    if summary["pool_sha256"] != sha256_file(pool_path):
        raise ValueError(f"{tokens_path} was built from another pool")
    rows = read_jsonl(tokens_path)
    return {r["trajectory_id"]: r["problem_id"] for r in rows}, {r["trajectory_id"]: r["response_tokens"] for r in rows}


def load_ell(nll_path, tokens):
    """ell by trajectory; the NLL file must cover the pool with the token counts of the budget."""
    rows = read_jsonl(nll_path)
    ell = {r["trajectory_id"]: r["ell"] for r in rows}
    if len(ell) != len(rows) or set(ell) != set(tokens):
        raise ValueError(f"{nll_path} does not cover exactly the eligible pool")
    wrong = [r["trajectory_id"] for r in rows if r["n_tokens"] != tokens[r["trajectory_id"]]]
    if wrong:
        raise ValueError(f"{len(wrong)} token counts in {nll_path} differ from the token table: {wrong[:3]}")
    return ell

#!/usr/bin/env python3
"""
Score every verified pool trajectory under a Student checkpoint (one forward pass, no backward).

Per trajectory tau (trajectory tokens only, the end-of-turn token excluded):
  nll_sum    = sum_t -log p(tau_t | x, tau_<t)
  ell        = nll_sum / n_tokens            (length-normalized NLL; input of L_proxy, config "learnability")
  brier_sum  = sum_t || pi_t - onehot(tau_t) ||^2
               (Brier residual; stored so the LARK score, arXiv 2605.30651 Eq. 7, can be computed
                later without another pass. Exploratory, not part of the frozen L_proxy.)

L_proxy itself is not computed here: it needs ell over the whole pool (median / IQR rule) and
is derived in the analysis step.

--pool defaults to the official pool v1 (trajectory/pool/pool_v1_verified.jsonl). The Task 5 smoke test
scored the preliminary pool; reproduce it with --pool trajectory/pool/batch_100_teacher_verified.jsonl.

Usage (from experiments/week2September/manual/):
  python scripts/compute_nll.py --tag S0 --out reports/task5/smoke_test/nll_pool_S0.jsonl \
      --pool trajectory/pool/batch_100_teacher_verified.jsonl
  python scripts/compute_nll.py --tag S1_seed13 --adapter <adapter_dir> \
      --out reports/task5/smoke_test/nll_pool_S1_seed13.jsonl --pool trajectory/pool/batch_100_teacher_verified.jsonl
"""
import argparse
import json
import sys
import time
from pathlib import Path

import torch

sys.path.insert(0, str(Path(__file__).resolve().parent))
from student_common import (  # noqa: E402
    IGNORE_INDEX, POOL_VERIFIED, add_model_args, check_prompt_template, encode_example, environment_info,
    load_config, load_model, load_pool, load_tokenizer, pick_dtype, sha256_file, student_from_args, write_json,
)


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--tag", required=True)
    ap.add_argument("--adapter", default=None)
    ap.add_argument("--out", required=True, type=Path)
    ap.add_argument("--batch-size", type=int, default=4)
    ap.add_argument("--limit", type=int, default=None)
    ap.add_argument("--pool", type=Path, default=POOL_VERIFIED, help="verified pool to score (default: pool v1)")
    add_model_args(ap)
    args = ap.parse_args()

    check_prompt_template()
    cfg = load_config()
    model_name, revision = student_from_args(args, cfg)
    dtype = pick_dtype(torch, args.dtype)
    pool = sorted(load_pool(args.pool), key=lambda r: r["trajectory_id"])
    if args.limit:
        pool = pool[: args.limit]

    tok = load_tokenizer(model_name, revision)
    model = load_model(model_name, revision, dtype, adapter=args.adapter)
    examples = [encode_example(tok, r["question"], r["raw_generation"], add_eos=False) for r in pool]

    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    t0 = time.time()
    args.out.parent.mkdir(parents=True, exist_ok=True)
    with args.out.open("w", encoding="utf-8") as f:
        for i in range(0, len(examples), args.batch_size):
            batch = examples[i: i + args.batch_size]
            width = max(len(e["input_ids"]) for e in batch)
            ids = torch.tensor([e["input_ids"] + [tok.pad_token_id] * (width - len(e["input_ids"])) for e in batch])
            mask = torch.tensor([[1] * len(e["input_ids"]) + [0] * (width - len(e["input_ids"])) for e in batch])
            labels = torch.tensor([e["labels"] + [IGNORE_INDEX] * (width - len(e["labels"])) for e in batch])
            ids, mask, labels = ids.to(model.device), mask.to(model.device), labels.to(model.device)
            with torch.no_grad():
                logits = model(input_ids=ids, attention_mask=mask).logits[:, :-1]
            tgt = labels[:, 1:]
            for b, e in enumerate(batch):
                sel = tgt[b] != IGNORE_INDEX
                lp = torch.log_softmax(logits[b][sel].float(), dim=-1)
                y = tgt[b][sel]
                lp_y = lp.gather(1, y[:, None]).squeeze(1)
                p = lp.exp()
                brier = (p.pow(2).sum(-1) - 2 * lp_y.exp() + 1).sum().item()
                nll_sum = -lp_y.sum().item()
                r = pool[i + b]
                f.write(json.dumps({
                    "trajectory_id": r["trajectory_id"], "problem_id": r["problem_id"], "checkpoint": args.tag,
                    "n_tokens": int(sel.sum()), "nll_sum": nll_sum, "ell": nll_sum / int(sel.sum()),
                    "brier_sum": brier,
                }) + "\n")
            print(f"[{args.tag}] {min(i + args.batch_size, len(examples))}/{len(examples)}  {time.time() - t0:.0f}s",
                  flush=True)
    elapsed = time.time() - t0
    write_json(args.out.with_suffix(".summary.json"), {
        "checkpoint": args.tag, "adapter": args.adapter, "model": model_name, "revision": revision,
        "dtype": str(dtype).replace("torch.", ""), "n_trajectories": len(examples),
        "pool": str(args.pool), "pool_sha256": sha256_file(args.pool),
        "forward_passes": len(examples), "elapsed_seconds": round(elapsed, 1),
        "peak_gpu_memory_gb": round(torch.cuda.max_memory_allocated() / 2**30, 2) if torch.cuda.is_available() else None,
        "environment": environment_info(torch),
    })
    print(f"[{args.tag}] scored {len(examples)} trajectories in {elapsed:.0f}s -> {args.out}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Train one round of the Student with LoRA: S_t + B_t -> S_{t+1}.

Hyperparameters come from config/experiment_config.json "training" (LoRA r/alpha/dropout/targets,
AdamW, constant lr, no warmup, max_grad_norm, effective batch, 1 epoch, max_seq_len,
completion-only loss). Effective batch = --micro-batch x gradient accumulation.

Loss: token-level mean over all supervised (response) tokens of the effective batch;
prompt tokens are masked. Seeds: torch seed (LoRA init, dropout) = data order seed = --seed.

Pool: the trajectories are looked up in the pool named by the train set ("pool", checked against
"pool_sha256"), so the Task 5 smoke-test train sets still resolve to the preliminary pool; train sets
without that field use --pool (default: official pool v1).

Outputs:
  <adapter-dir>/                 LoRA adapter (safetensors) + tokenizer files
  <adapter-dir>/train_log.json   per-step loss, tokens, time; totals; config used

Usage (from experiments/week2September/manual/):
  python scripts/train_round.py --seed 13 \
      --train-set reports/task5/smoke_test/train_set_seed13.json \

def collate(examples, pad_id):
    width = max(len(e["input_ids"]) for e in examples)
    ids, labels, mask = [], [], []
    for e in examples:
        pad = width - len(e["input_ids"])
        ids.append(e["input_ids"] + [pad_id] * pad)          # right padding for training
        labels.append(e["labels"] + [IGNORE_INDEX] * pad)
        mask.append([1] * len(e["input_ids"]) + [0] * pad)
    return torch.tensor(ids), torch.tensor(labels), torch.tensor(mask)


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--seed", type=int, required=True)
    ap.add_argument("--train-set", required=True, type=Path, help="JSON with trajectory_ids (select_smoke_train_set.py)")
    ap.add_argument("--adapter-dir", required=True, type=Path)
    ap.add_argument("--init-adapter", default=None, help="continue from an existing adapter (round t > 0)")
    ap.add_argument("--micro-batch", type=int, default=4)
    ap.add_argument("--epochs", type=int, default=None, help="default: config training.epochs_per_round")
    ap.add_argument("--lr", type=float, default=None, help="default: config training.learning_rate")
    ap.add_argument("--no-gradient-checkpointing", action="store_true")
    ap.add_argument("--pool", type=Path, default=None,
                    help="pool for train sets without a 'pool' field (default: official pool v1)")
    add_model_args(ap)
    args = ap.parse_args()

    check_prompt_template()
    cfg = load_config()
    tc = cfg["training"]
    if tc["loss"].split()[0] != "completion-only" or tc["lr_scheduler"] != "constant" or tc["warmup_steps"] != 0:
        raise ValueError("config training block differs from what this script implements")
    eff_batch = tc["batch_size"]
    if eff_batch % args.micro_batch:
        raise ValueError(f"--micro-batch must divide the effective batch {eff_batch}")
    accum = eff_batch // args.micro_batch
    epochs = args.epochs or tc["epochs_per_round"]
    lr = args.lr or tc["learning_rate"]
    model_name, revision = student_from_args(args, cfg)
    dtype = pick_dtype(torch, args.dtype)

    from peft import LoraConfig, PeftModel, get_peft_model
    from transformers import AutoModelForCausalLM, AutoTokenizer

    train_set = json.loads(args.train_set.read_text(encoding="utf-8"))
    if "pool" in train_set:
        pool_path = MANUAL_DIR / train_set["pool"]
        if args.pool and args.pool.resolve() != pool_path.resolve():
            raise ValueError(f"--pool {args.pool} differs from the train set's pool {train_set['pool']}")
        if "pool_sha256" in train_set and sha256_file(pool_path) != train_set["pool_sha256"]:
            raise ValueError(f"{pool_path} changed since the train set was drawn")
    else:
        pool_path = args.pool or POOL_VERIFIED
    by_id = {r["trajectory_id"]: r for r in load_pool(pool_path)}
    rows = [by_id[t] for t in train_set["trajectory_ids"]]

    tok = AutoTokenizer.from_pretrained(model_name, revision=revision)
    examples = [encode_example(tok, r["question"], r["raw_generation"]) for r in rows]
    too_long = [r["trajectory_id"] for r, e in zip(rows, examples) if len(e["input_ids"]) > tc["max_seq_len"]]
    if too_long:
        raise ValueError(f"{len(too_long)} examples exceed max_seq_len {tc['max_seq_len']}: {too_long[:3]}")

    torch.manual_seed(args.seed)
    random.seed(args.seed)
    model = AutoModelForCausalLM.from_pretrained(model_name, revision=revision, dtype=dtype)
    if not args.no_gradient_checkpointing:
        model.gradient_checkpointing_enable(gradient_checkpointing_kwargs={"use_reentrant": False})
        model.enable_input_require_grads()
    if args.init_adapter:
        model = PeftModel.from_pretrained(model, args.init_adapter, is_trainable=True)
    else:
        model = get_peft_model(model, LoraConfig(
            r=tc["lora_r"], lora_alpha=tc["lora_alpha"], lora_dropout=tc["lora_dropout"],
            target_modules=tc["lora_target_modules"], bias="none", task_type="CAUSAL_LM",
        ))
    device = "cuda" if torch.cuda.is_available() else "cpu"
    model.to(device)
    model.train()
    trainable = [p for p in model.parameters() if p.requires_grad]
    n_trainable = sum(p.numel() for p in trainable)
    opt = torch.optim.AdamW(trainable, lr=lr, betas=(tc["adam_beta1"], tc["adam_beta2"]),
                            weight_decay=tc["weight_decay"])  # fresh optimizer state each round (config)

    order_rng = random.Random(args.seed)  # data_order_seed = training_seed
    log, step, t0 = [], 0, time.time()
    total_sup, total_prompt = 0, 0
    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    for epoch in range(epochs):
        order = list(range(len(examples)))
        order_rng.shuffle(order)
        for start in range(0, len(order), eff_batch):
            eff = [examples[i] for i in order[start: start + eff_batch]]
            n_sup = sum(e["n_supervised"] for e in eff)
            loss_sum = 0.0
            opt.zero_grad(set_to_none=True)
            for m in range(0, len(eff), args.micro_batch):
                ids, labels, mask = (x.to(device) for x in collate(eff[m: m + args.micro_batch], tok.pad_token_id))
                logits = model(input_ids=ids, attention_mask=mask).logits[:, :-1]
                tgt = labels[:, 1:]
                sel = tgt != IGNORE_INDEX  # upcast only supervised positions (saves memory)
                tok_loss = torch.nn.functional.cross_entropy(logits[sel].float(), tgt[sel], reduction="sum")
                (tok_loss / n_sup).backward()
                loss_sum += tok_loss.item()
            grad_norm = torch.nn.utils.clip_grad_norm_(trainable, tc["max_grad_norm"]).item()
            opt.step()
            step += 1
            total_sup += n_sup
            total_prompt += sum(e["n_prompt"] for e in eff)
            log.append({"step": step, "epoch": epoch, "loss": loss_sum / n_sup, "grad_norm": grad_norm,
                        "n_examples": len(eff), "supervised_tokens": n_sup, "lr": lr,
                        "elapsed_s": round(time.time() - t0, 1)})
            print(f"step {step:3d}  loss {loss_sum / n_sup:.4f}  grad_norm {grad_norm:.3f}  "
                  f"tokens {n_sup}  {time.time() - t0:.0f}s", flush=True)
            if not math.isfinite(loss_sum):
                raise RuntimeError("non-finite loss")

    args.adapter_dir.mkdir(parents=True, exist_ok=True)
    model.save_pretrained(args.adapter_dir)
    tok.save_pretrained(args.adapter_dir)
    write_json(args.adapter_dir / "train_log.json", {
        "checkpoint_parent": args.init_adapter or "student_ckpt_000",
        "seed": args.seed, "data_order_seed": args.seed,
        "train_set": str(args.train_set), "train_set_sha256": sha256_file(args.train_set),
        "pool": str(pool_path), "pool_sha256": sha256_file(pool_path),
        "n_examples": len(examples), "epochs": epochs, "steps": step,
        "effective_batch": eff_batch, "micro_batch": args.micro_batch, "grad_accum": accum,
        "learning_rate": lr, "lr_from_config": lr == tc["learning_rate"],
        "epochs_from_config": epochs == tc["epochs_per_round"],
        "model": model_name, "revision": revision, "dtype": str(dtype).replace("torch.", ""),
        "gradient_checkpointing": not args.no_gradient_checkpointing,
        "trainable_parameters": n_trainable,
        "cumulative_training_tokens": total_sup,
        "prompt_tokens_masked": total_prompt,
        "trajectory_tokens": sum(e["n_traj"] for e in examples) * epochs,
        "beta_round_tokens_config": cfg["budget"]["beta_round_tokens"],
        "elapsed_seconds": round(time.time() - t0, 1),
        "peak_gpu_memory_gb": round(torch.cuda.max_memory_allocated() / 2**30, 2) if torch.cuda.is_available() else None,
        "config_sha256": sha256_file(CONFIG_PATH),
        "environment": environment_info(torch),
        "steps_log": log,
    })
    print(f"saved adapter -> {args.adapter_dir}  ({step} steps, {total_sup} supervised tokens)")


if __name__ == "__main__":
    main()

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path) -> dict:
    value = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def read_jsonl(path: Path) -> list[dict]:
    rows = []
    with path.open("r", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, start=1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSONL at {path}:{line_number}: {exc}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected JSON object at {path}:{line_number}"
                )
            rows.append(row)

    return rows


def write_temp_exclusive(path: Path, content: bytes) -> Path:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp_path = path.with_name(
        path.name + f".{os.getpid()}.tmp"
    )

    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
    descriptor = os.open(str(temp_path), flags, 0o600)

    try:
        with os.fdopen(descriptor, "wb") as handle:
            handle.write(content)
            handle.flush()
            os.fsync(handle.fileno())
    except Exception:
        try:
            temp_path.unlink()
        except OSError:
            pass
        raise

    return temp_path


def publish_pair_no_overwrite(
    train_set_path: Path,
    selection_log_path: Path,
    train_set_bytes: bytes,
    selection_log_bytes: bytes,
) -> None:
    if train_set_path.exists():
        raise FileExistsError(
            f"Refusing to overwrite existing file: {train_set_path}"
        )

    if selection_log_path.exists():
        raise FileExistsError(
            f"Refusing to overwrite existing file: {selection_log_path}"
        )

    train_temp = write_temp_exclusive(train_set_path, train_set_bytes)
    log_temp = None

    try:
        log_temp = write_temp_exclusive(
            selection_log_path,
            selection_log_bytes,
        )

        if train_set_path.exists() or selection_log_path.exists():
            raise FileExistsError(
                "An output appeared during publication; refusing to overwrite."
            )

        os.rename(train_temp, train_set_path)
        os.rename(log_temp, selection_log_path)

    except Exception:
        for temp_path in (train_temp, log_temp):
            if temp_path is not None:
                try:
                    temp_path.unlink(missing_ok=True)
                except OSError:
                    pass
        raise


def load_pool(
    pool_path: Path,
    d_pool_path: Path,
) -> tuple[dict[str, dict], dict[str, str]]:
    rows = read_jsonl(pool_path)
    d_pool_ids = {
        line.strip()
        for line in d_pool_path.read_text(encoding="utf-8").splitlines()
        if line.strip()
    }

    pool_by_id: dict[str, dict] = {}
    problems: dict[str, str] = {}

    for index, row in enumerate(rows, start=1):
        tid = row.get("trajectory_id")
        pid = row.get("problem_id")

        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Pool row {index}: invalid trajectory_id")

        if not isinstance(pid, str) or not pid:
            raise ValueError(f"Pool row {index}: invalid problem_id")

        if tid in pool_by_id:
            raise ValueError(f"Duplicate trajectory_id in pool: {tid}")

        metadata = row.get("metadata")
        if not isinstance(metadata, dict):
            raise ValueError(f"{tid}: metadata is missing or invalid")

        if metadata.get("verification_status") != "valid":
            raise ValueError(f"{tid}: verification_status is not valid")

        capabilities = row.get("capabilities")
        if not isinstance(capabilities, list) or not capabilities:
            raise ValueError(f"{tid}: capability list is empty or invalid")

        if len(capabilities) != len(set(capabilities)):
            raise ValueError(f"{tid}: duplicate capability tags")

        unknown = sorted(set(capabilities) - set(TAXONOMY))
        if unknown:
            raise ValueError(f"{tid}: unknown capability tags: {unknown}")

        if pid not in d_pool_ids:
            raise ValueError(f"{tid}: problem_id is outside D_pool: {pid}")

        pool_by_id[tid] = row
        problems[tid] = pid

    if not pool_by_id:
        raise ValueError("Verified pool is empty")

    return pool_by_id, problems


def load_tokens(
    tokens_path: Path,
    pool_path: Path,
    problems: dict[str, str],
) -> dict[str, int]:
    summary_path = tokens_path.with_suffix(".summary.json")

    if not summary_path.is_file():
        raise FileNotFoundError(
            f"Token summary not found: {summary_path}"
        )

    summary = read_json(summary_path)
    actual_pool_hash = sha256_file(pool_path)

    if summary.get("pool_sha256") != actual_pool_hash:
        raise ValueError(
            "Token summary pool_sha256 does not match the current pool."
        )

    rows = read_jsonl(tokens_path)
    tokens: dict[str, int] = {}

    for index, row in enumerate(rows, start=1):
        tid = row.get("trajectory_id")
        pid = row.get("problem_id")
        count = row.get("response_tokens")

        if not isinstance(tid, str) or not tid:
            raise ValueError(f"Token row {index}: invalid trajectory_id")

        if tid in tokens:
            raise ValueError(f"Duplicate token row: {tid}")

        if tid not in problems:
            raise ValueError(f"Token row not present in verified pool: {tid}")

        if problems[tid] != pid:
            raise ValueError(f"Token problem_id mismatch: {tid}")

        if not isinstance(count, int) or isinstance(count, bool) or count <= 0:
            raise ValueError(f"Invalid response_tokens for {tid}: {count}")

        tokens[tid] = count

    if set(tokens) != set(problems):
        missing = sorted(set(problems) - set(tokens))
        extra = sorted(set(tokens) - set(problems))
        raise ValueError(
            f"Pool/token IDs differ. Missing={missing[:5]}, extra={extra[:5]}"
        )

    token_sum = sum(tokens.values())

    if len(rows) != summary.get("n_trajectories"):
        raise ValueError("Token summary n_trajectories mismatch")

    if token_sum != summary.get("response_tokens_sum"):
        raise ValueError("Token summary response_tokens_sum mismatch")

    return tokens


def load_capability_deficits(state_path: Path, method: str, seed: int, round_number: int) -> dict[str, float]:
    state = read_json(state_path)

    if state.get("schema_version") != "p1p2.capability-state.v1":
        raise ValueError("Unsupported capability state schema")

    target = state.get("target", {})

    if target.get("method") != method:
        raise ValueError(
            f"State target method mismatch: expected {method}, got {target.get('method')}"
        )

    if target.get("seed") != seed:
        raise ValueError(
            f"State target seed mismatch: expected {seed}, got {target.get('seed')}"
        )

    if target.get("selection_round") != round_number:
        raise ValueError(
            "State selection_round mismatch: "
            f"expected {round_number}, got {target.get('selection_round')}"
        )

    capabilities = state.get("capabilities")
    if not isinstance(capabilities, dict):
        raise ValueError("Capability state has no capabilities object")

    deficits = {}

    for capability in TAXONOMY:
        item = capabilities.get(capability)
        if not isinstance(item, dict):
            raise ValueError(f"Missing capability state: {capability}")

        value = item.get("deficit")
        if (
            not isinstance(value, (int, float))
            or isinstance(value, bool)
            or not math.isfinite(value)
            or not 0.0 <= value <= 1.0
        ):
            raise ValueError(f"Invalid deficit for {capability}: {value}")

        deficits[capability] = float(value)

    return deficits


def load_previous_train_ids(paths: list[Path], problems: dict[str, str]) -> list[str]:
    trained = set()

    for path in paths:
        data = read_json(path)
        ids = data.get("trajectory_ids")

        if not isinstance(ids, list):
            raise ValueError(f"Invalid trajectory_ids in {path}")

        for tid in ids:
            if not isinstance(tid, str) or tid not in problems:
                raise ValueError(
                    f"Previous train set contains unknown trajectory: {tid}"
                )
            trained.add(tid)

    return sorted(trained)


def load_nll(
    nll_path: Path,
    tokens: dict[str, int],
    source_checkpoint: str,
) -> tuple[dict[str, float], str]:
    rows = read_jsonl(nll_path)
    ell_by_id = {}

    for index, row in enumerate(rows, start=1):
        tid = row.get("trajectory_id")

        if not isinstance(tid, str) or not tid:
            raise ValueError(f"NLL row {index}: invalid trajectory_id")

        if tid in ell_by_id:
            raise ValueError(f"Duplicate NLL trajectory_id: {tid}")

        if tid not in tokens:
            raise ValueError(f"NLL trajectory not present in token table: {tid}")

        if row.get("checkpoint") != source_checkpoint:
            raise ValueError(
                f"NLL checkpoint mismatch for {tid}: "
                f"expected {source_checkpoint}, got {row.get('checkpoint')}"
            )

        if row.get("n_tokens") != tokens[tid]:
            raise ValueError(f"NLL token count mismatch for {tid}")

        ell = row.get("ell")
        if (
            not isinstance(ell, (int, float))
            or isinstance(ell, bool)
            or not math.isfinite(ell)
        ):
            raise ValueError(f"Invalid ell value for {tid}: {ell}")

        ell_by_id[tid] = float(ell)

    if set(ell_by_id) != set(tokens):
        missing = sorted(set(tokens) - set(ell_by_id))
        extra = sorted(set(ell_by_id) - set(tokens))
        raise ValueError(
            f"NLL does not cover the token table. "
            f"Missing={missing[:5]}, extra={extra[:5]}"
        )

    summary_path = nll_path.with_suffix(".summary.json")
    if not summary_path.is_file():
        raise FileNotFoundError(f"NLL summary not found: {summary_path}")

    summary = read_json(summary_path)

    if summary.get("checkpoint") != source_checkpoint:
        raise ValueError("NLL summary checkpoint mismatch")

    if summary.get("pool_sha256") != sha256_file(POOL_DEFAULT):
        raise ValueError("NLL summary pool_sha256 mismatch")

    if summary.get("n_trajectories") != len(tokens):
        raise ValueError("NLL summary trajectory count mismatch")

    return ell_by_id, sha256_file(nll_path)


def average_percentile_ranks(scores: dict[str, float]) -> dict[str, float]:
    if not scores:
        raise ValueError("Cannot rank an empty score mapping")

    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    n = len(ordered)

    if n == 1:
        return {ordered[0][0]: 0.5}

    result = {}
    index = 0

    while index < n:
        end = index + 1
        value = ordered[index][1]

        while end < n and ordered[end][1] == value:
            end += 1

        average_rank = ((index + 1) + end) / 2.0
        percentile = (average_rank - 1.0) / (n - 1.0)

        for position in range(index, end):
            result[ordered[position][0]] = percentile

        index = end

    return result


def calculate_p1_scores(
    pool_by_id: dict[str, dict],
    deficits: dict[str, float],
) -> dict[str, float]:
    result = {}

    for tid, row in pool_by_id.items():
        tags = row["capabilities"]
        result[tid] = sum(deficits[tag] for tag in tags) / len(tags)

    return result

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    with path.open("r", encoding="utf-8-sig") as stream:
        value = json.load(stream)
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def stable_state(value: dict[str, Any]) -> dict[str, Any]:

def require_equal(label: str, left: Any, right: Any) -> None:

def validate_source_identity(
    manifest: dict[str, Any],

    for field in required_fields:

    for field in ("run_id", "method", "seed", "round"):
        if field not in manifest:
            raise ValueError(f"Missing manifest field: {field}")

        require_equal(field, manifest[field], lineage[field])

    source_seed = int(lineage["seed"])

def validate_metrics(metrics_path: Path) -> dict[str, dict[str, Any]]:
    by_capability: dict[str, dict[str, Any]] = {}

    with metrics_path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue

            row = json.loads(line)
            if not isinstance(row, dict):
                raise ValueError(f"Invalid metric row at line {line_no}")

            capability = row.get("capability")
            if capability not in CAPABILITIES:
                raise ValueError(f"Unknown capability: {capability!r}")

            if capability in by_capability:
                raise ValueError(f"Duplicate capability row: {capability}")

            if row.get("strict_metric") != "status_equals_valid":
                raise ValueError("Capability row does not use strict metric")

            if row.get("gold_policy") != "approved_85_primary":
                raise ValueError("Capability row uses an invalid Gold policy")

            support = int(row["n_support"])
            correct = int(row["n_correct"])
            incorrect = int(row["n_incorrect"])

            if support != EXPECTED_SUPPORT[capability]:
                raise ValueError(
                    f"Support mismatch for {capability}: "
                    f"{support} != {EXPECTED_SUPPORT[capability]}"
                )

            if correct < 0 or incorrect < 0:
                raise ValueError(f"Negative metric count: {capability}")

            if correct + incorrect != support:
                raise ValueError(f"Count mismatch: {capability}")

            mastery = correct / support
            deficit = 1.0 - mastery

            if not math.isfinite(float(row["mastery"])):
                raise ValueError(f"Non-finite mastery: {capability}")

            if not math.isfinite(float(row["deficit"])):
                raise ValueError(f"Non-finite deficit: {capability}")

            if abs(float(row["mastery"]) - mastery) > 1e-12:
                raise ValueError(f"Mastery mismatch: {capability}")

            if abs(float(row["deficit"]) - deficit) > 1e-12:
                raise ValueError(f"Deficit mismatch: {capability}")

            by_capability[capability] = {
                "n_support": support,
                "n_correct": correct,
                "n_incorrect": incorrect,
                "mastery": mastery,
                "deficit": deficit,
                "ci95_low": row.get("ci95_low"),
                "ci95_high": row.get("ci95_high"),
            }

    if set(by_capability) != set(CAPABILITIES):
        raise ValueError(
            "Capability coverage is incomplete: "
            f"{sorted(by_capability)}"
        )

    return by_capability


def publish_exclusive(output: Path, content: bytes) -> None:
    """Publish through a same-directory hard link; never replace a target."""
    output.parent.mkdir(parents=True, exist_ok=True)

    fd, temp_name = tempfile.mkstemp(
        prefix=f".{output.name}.staging.",
        suffix=".tmp",
        dir=str(output.parent),
    )
    temp_path = Path(temp_name)

    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(content)
            stream.flush()
            os.fsync(stream.fileno())

        try:
            os.link(temp_path, output)
        except FileExistsError:
            raise
        except OSError as exc:
            raise RuntimeError(
                "ATOMIC_NO_OVERWRITE_UNAVAILABLE: hard-link publication "
                f"failed; temporary file preserved at {temp_path}: {exc}"
            ) from exc

        temp_path.unlink(missing_ok=True)

    except Exception:
        if temp_path.exists():
            print(f"RECOVERY_TEMP_PRESERVED={temp_path}")
        raise
