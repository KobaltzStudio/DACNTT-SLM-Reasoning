#!/usr/bin/env python3
"""
Train one round of the Student with LoRA: S_t + B_t -> S_{t+1}.

Hyperparameters come from config/experiment_config.json "training" (LoRA r/alpha/dropout/targets,
AdamW, constant lr, no warmup, max_grad_norm, effective batch, 1 epoch, max_seq_len,
completion-only loss). Effective batch = --micro-batch x gradient accumulation.

Loss: token-level mean over all supervised (response) tokens of the effective batch;
prompt tokens are masked. Seeds: torch seed (LoRA init, dropout) = data order seed = --seed.

Pool: the trajectories are looked up in the pool named by the train set ("pool", checked against
"pool_sha256"), so the Task 5 smoke-test train sets still resolve to the preliminary pool; train sets
without that field use --pool (default: official pool v1).

Outputs:
  <adapter-dir>/                 LoRA adapter (safetensors) + tokenizer files
  <adapter-dir>/train_log.json   per-step loss, tokens, time; totals; config used

Usage (from experiments/week2September/manual/):
  python scripts/train_round.py --seed 13 \
      --train-set reports/task5/smoke_test/train_set_seed13.json \
      --adapter-dir models/smoke/student_ckpt_001_smoke_seed13
"""
import argparse
import json
import math
import random
import sys
import time
from pathlib import Path

import torch

sys.path.insert(0, str(Path(__file__).resolve().parent))
from student_common import (  # noqa: E402
    CONFIG_PATH, IGNORE_INDEX, MANUAL_DIR, POOL_VERIFIED, add_model_args, check_prompt_template, encode_example,
    environment_info, load_config, load_pool, pick_dtype, sha256_file, student_from_args, write_json,
)


def collate(examples, pad_id):
    width = max(len(e["input_ids"]) for e in examples)
    ids, labels, mask = [], [], []
    for e in examples:
        pad = width - len(e["input_ids"])
        ids.append(e["input_ids"] + [pad_id] * pad)          # right padding for training
        labels.append(e["labels"] + [IGNORE_INDEX] * pad)
        mask.append([1] * len(e["input_ids"]) + [0] * pad)
    return torch.tensor(ids), torch.tensor(labels), torch.tensor(mask)


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--seed", type=int, required=True)
    ap.add_argument("--train-set", required=True, type=Path, help="JSON with trajectory_ids (select_smoke_train_set.py)")
    ap.add_argument("--adapter-dir", required=True, type=Path)
    ap.add_argument("--init-adapter", default=None, help="continue from an existing adapter (round t > 0)")
    ap.add_argument("--micro-batch", type=int, default=4)
    ap.add_argument("--epochs", type=int, default=None, help="default: config training.epochs_per_round")
    ap.add_argument("--lr", type=float, default=None, help="default: config training.learning_rate")
    ap.add_argument("--no-gradient-checkpointing", action="store_true")
    ap.add_argument("--pool", type=Path, default=None,
                    help="pool for train sets without a 'pool' field (default: official pool v1)")
    add_model_args(ap)
    args = ap.parse_args()

    check_prompt_template()
    cfg = load_config()
    tc = cfg["training"]
    if tc["loss"].split()[0] != "completion-only" or tc["lr_scheduler"] != "constant" or tc["warmup_steps"] != 0:
        raise ValueError("config training block differs from what this script implements")
    eff_batch = tc["batch_size"]
    if eff_batch % args.micro_batch:
        raise ValueError(f"--micro-batch must divide the effective batch {eff_batch}")
    accum = eff_batch // args.micro_batch
    epochs = args.epochs or tc["epochs_per_round"]
    lr = args.lr or tc["learning_rate"]
    model_name, revision = student_from_args(args, cfg)
    dtype = pick_dtype(torch, args.dtype)

    from peft import LoraConfig, PeftModel, get_peft_model
    from transformers import AutoModelForCausalLM, AutoTokenizer

    train_set = json.loads(args.train_set.read_text(encoding="utf-8"))
    if "pool" in train_set:
        pool_path = MANUAL_DIR / train_set["pool"]
        if args.pool and args.pool.resolve() != pool_path.resolve():
            raise ValueError(f"--pool {args.pool} differs from the train set's pool {train_set['pool']}")
        if "pool_sha256" in train_set and sha256_file(pool_path) != train_set["pool_sha256"]:
            raise ValueError(f"{pool_path} changed since the train set was drawn")
    else:
        pool_path = args.pool or POOL_VERIFIED
    by_id = {r["trajectory_id"]: r for r in load_pool(pool_path)}
    rows = [by_id[t] for t in train_set["trajectory_ids"]]

    tok = AutoTokenizer.from_pretrained(model_name, revision=revision)
    examples = [encode_example(tok, r["question"], r["raw_generation"]) for r in rows]
    too_long = [r["trajectory_id"] for r, e in zip(rows, examples) if len(e["input_ids"]) > tc["max_seq_len"]]
    if too_long:
        raise ValueError(f"{len(too_long)} examples exceed max_seq_len {tc['max_seq_len']}: {too_long[:3]}")

    torch.manual_seed(args.seed)
    random.seed(args.seed)
    model = AutoModelForCausalLM.from_pretrained(model_name, revision=revision, dtype=dtype)
    if not args.no_gradient_checkpointing:
        model.gradient_checkpointing_enable(gradient_checkpointing_kwargs={"use_reentrant": False})
        model.enable_input_require_grads()
    if args.init_adapter:
        model = PeftModel.from_pretrained(model, args.init_adapter, is_trainable=True)
    else:
        model = get_peft_model(model, LoraConfig(
            r=tc["lora_r"], lora_alpha=tc["lora_alpha"], lora_dropout=tc["lora_dropout"],
            target_modules=tc["lora_target_modules"], bias="none", task_type="CAUSAL_LM",
        ))
    device = "cuda" if torch.cuda.is_available() else "cpu"
    model.to(device)
    model.train()
    trainable = [p for p in model.parameters() if p.requires_grad]
    n_trainable = sum(p.numel() for p in trainable)
    opt = torch.optim.AdamW(trainable, lr=lr, betas=(tc["adam_beta1"], tc["adam_beta2"]),
                            weight_decay=tc["weight_decay"])  # fresh optimizer state each round (config)

    order_rng = random.Random(args.seed)  # data_order_seed = training_seed
    log, step, t0 = [], 0, time.time()
    total_sup, total_prompt = 0, 0
    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    for epoch in range(epochs):
        order = list(range(len(examples)))
        order_rng.shuffle(order)
        for start in range(0, len(order), eff_batch):
            eff = [examples[i] for i in order[start: start + eff_batch]]
            n_sup = sum(e["n_supervised"] for e in eff)
            loss_sum = 0.0
            opt.zero_grad(set_to_none=True)
            for m in range(0, len(eff), args.micro_batch):
                ids, labels, mask = (x.to(device) for x in collate(eff[m: m + args.micro_batch], tok.pad_token_id))
                logits = model(input_ids=ids, attention_mask=mask).logits[:, :-1]
                tgt = labels[:, 1:]
                sel = tgt != IGNORE_INDEX  # upcast only supervised positions (saves memory)
                tok_loss = torch.nn.functional.cross_entropy(logits[sel].float(), tgt[sel], reduction="sum")
                (tok_loss / n_sup).backward()
                loss_sum += tok_loss.item()
            grad_norm = torch.nn.utils.clip_grad_norm_(trainable, tc["max_grad_norm"]).item()
            opt.step()
            step += 1
            total_sup += n_sup
            total_prompt += sum(e["n_prompt"] for e in eff)
            log.append({"step": step, "epoch": epoch, "loss": loss_sum / n_sup, "grad_norm": grad_norm,
                        "n_examples": len(eff), "supervised_tokens": n_sup, "lr": lr,
                        "elapsed_s": round(time.time() - t0, 1)})
            print(f"step {step:3d}  loss {loss_sum / n_sup:.4f}  grad_norm {grad_norm:.3f}  "
                  f"tokens {n_sup}  {time.time() - t0:.0f}s", flush=True)
            if not math.isfinite(loss_sum):
                raise RuntimeError("non-finite loss")

    args.adapter_dir.mkdir(parents=True, exist_ok=True)
    model.save_pretrained(args.adapter_dir)
    tok.save_pretrained(args.adapter_dir)
    write_json(args.adapter_dir / "train_log.json", {
        "checkpoint_parent": args.init_adapter or "student_ckpt_000",
        "seed": args.seed, "data_order_seed": args.seed,
        "train_set": str(args.train_set), "train_set_sha256": sha256_file(args.train_set),
        "pool": str(pool_path), "pool_sha256": sha256_file(pool_path),
        "n_examples": len(examples), "epochs": epochs, "steps": step,
        "effective_batch": eff_batch, "micro_batch": args.micro_batch, "grad_accum": accum,
        "learning_rate": lr, "lr_from_config": lr == tc["learning_rate"],
        "epochs_from_config": epochs == tc["epochs_per_round"],
        "model": model_name, "revision": revision, "dtype": str(dtype).replace("torch.", ""),
        "gradient_checkpointing": not args.no_gradient_checkpointing,
        "trainable_parameters": n_trainable,
        "cumulative_training_tokens": total_sup,
        "prompt_tokens_masked": total_prompt,
        "trajectory_tokens": sum(e["n_traj"] for e in examples) * epochs,
        "beta_round_tokens_config": cfg["budget"]["beta_round_tokens"],
        "elapsed_seconds": round(time.time() - t0, 1),
        "peak_gpu_memory_gb": round(torch.cuda.max_memory_allocated() / 2**30, 2) if torch.cuda.is_available() else None,
        "config_sha256": sha256_file(CONFIG_PATH),
        "environment": environment_info(torch),
        "steps_log": log,
    })
    print(f"saved adapter -> {args.adapter_dir}  ({step} steps, {total_sup} supervised tokens)")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Evaluate a Student checkpoint on D_dev or D_test with the frozen evaluation protocol.

Protocol (config/experiment_config.json "evaluation"): prompt v1, greedy decoding,
max_new_tokens 512, answers scored by answer_utils.evaluate_output
(invalid_format counts as incorrect; format compliance reported separately).

Greedy is enforced explicitly: the Qwen generation_config ships sampling defaults and
repetition_penalty 1.05, which would otherwise still apply under do_sample=False.

Also records lenient_correct (last number anywhere in the output, format ignored): the registered
secondary metric of T7-5, always reported with strict accuracy and format compliance. Accuracy per
capability is added on D_dev, over the problems with manual labels.

Backends:
  hf    transformers generate (default; used by the Task 5 smoke test)
  vllm  vLLM with the same greedy settings; needs a bf16 GPU (L4/A100). A LoRA adapter is merged into a
        temporary model directory first, exactly as the hf backend merges it before generating.

Output (resumable: problem_ids already in --out are skipped):
  --out           one JSON line per problem
  --summary       accuracy, format compliance, output length, timing, memory

Usage (from experiments/week2September/manual/):
  python scripts/eval_student.py --tag S0 --out reports/task5/smoke_test/eval_S0_dev.jsonl
  python scripts/eval_student.py --tag S1_seed13 --adapter <adapter_dir> \
      --out reports/task5/smoke_test/eval_S1_dev_seed13.jsonl
  python scripts/eval_student.py --split test --backend vllm --tag S1 --adapter <adapter_dir> --out <out.jsonl>
"""
import argparse
import gc
import os
import sys
import tempfile
import time
from pathlib import Path

# Deterministic scheduling in offline mode (vLLM docs, Reproducibility); must be set before importing vllm.
os.environ.setdefault("VLLM_ENABLE_V1_MULTIPROCESSING", "0")

import torch  # noqa: E402

sys.path.insert(0, str(Path(__file__).resolve().parent))
from eval_metrics import score_output, summarize  # noqa: E402
from student_common import (  # noqa: E402
    add_model_args, append_jsonl, build_prompt, check_prompt_template, end_of_turn_id,
    environment_info, load_capability_labels, load_config, load_dev_rows, load_model, load_test_rows,
    load_tokenizer, pick_dtype, read_jsonl_resumable, student_from_args, write_json,
)


def generate_hf(todo, tok, model_name, revision, dtype, adapter, max_new_tokens, batch_size):
    """Yields lists of (row, text, n_output_tokens), one list per batch."""
    from transformers import GenerationConfig
    model = load_model(model_name, revision, dtype, adapter=adapter)
    gen_cfg = GenerationConfig(
        do_sample=False,
        max_new_tokens=max_new_tokens,
        repetition_penalty=1.0,
        eos_token_id=[end_of_turn_id(tok), tok.eos_token_id],
        pad_token_id=tok.pad_token_id,
    )
    for i in range(0, len(todo), batch_size):
        batch = todo[i: i + batch_size]
        prompts = [build_prompt(tok, r["question"]) for r in batch]
        enc = tok(prompts, return_tensors="pt", padding=True, add_special_tokens=False).to(model.device)
        with torch.no_grad():
            out = model.generate(**enc, generation_config=gen_cfg)
        results = []
        for r, ids in zip(batch, out[:, enc["input_ids"].shape[1]:]):
            ids = ids.tolist()
            n_out = len(ids)
            while n_out and ids[n_out - 1] == tok.pad_token_id:
                n_out -= 1
            results.append((r, tok.decode(ids[:n_out], skip_special_tokens=True), n_out))
        yield results


def generate_vllm(todo, tok, model_name, revision, adapter, max_new_tokens, batch_size, gpu_memory_utilization):
    """Same contract as generate_hf. batch_size here is the number of problems per generate() call."""
    from vllm import LLM, SamplingParams
    if not torch.cuda.is_available() or not torch.cuda.is_bf16_supported():
        raise SystemExit("the vllm backend needs a GPU with bf16 (T6-5)")
    stop_ids = sorted({end_of_turn_id(tok), tok.eos_token_id})
    params = SamplingParams(n=1, temperature=0.0, top_p=1.0, repetition_penalty=1.0,
                            max_tokens=max_new_tokens, stop_token_ids=stop_ids)
    with tempfile.TemporaryDirectory(prefix="merged_student_") as merged:
        if adapter:
            model = load_model(model_name, revision, torch.bfloat16, adapter=adapter)
            model.save_pretrained(merged)
            tok.save_pretrained(merged)
            del model
            gc.collect()
            torch.cuda.empty_cache()
            llm = LLM(model=merged, dtype="bfloat16", generation_config="vllm",
                      gpu_memory_utilization=gpu_memory_utilization)
        else:
            llm = LLM(model=model_name, revision=revision, tokenizer_revision=revision, dtype="bfloat16",
                      generation_config="vllm", gpu_memory_utilization=gpu_memory_utilization)
        for i in range(0, len(todo), batch_size):
            batch = todo[i: i + batch_size]
            outputs = llm.generate([build_prompt(tok, r["question"]) for r in batch], params, use_tqdm=False)
            yield [(r, o.outputs[0].text, len(o.outputs[0].token_ids)) for r, o in zip(batch, outputs)]


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--tag", required=True, help="checkpoint label, e.g. S0 or S1_seed13")
    ap.add_argument("--adapter", default=None, help="LoRA adapter directory (omit for S_0)")
    ap.add_argument("--out", required=True, type=Path)
    ap.add_argument("--summary", type=Path, default=None, help="default: <out>.summary.json")
    ap.add_argument("--split", default="dev", choices=["dev", "test"])
    ap.add_argument("--backend", default="hf", choices=["hf", "vllm"])
    ap.add_argument("--batch-size", type=int, default=None, help="default: 16 for hf, 200 for vllm")
    ap.add_argument("--gpu-memory-utilization", type=float, default=0.85, help="vllm backend only")
    ap.add_argument("--limit", type=int, default=None, help="first N problems only (pipeline tests)")
    add_model_args(ap)
    args = ap.parse_args()

    check_prompt_template()
    cfg = load_config()
    ev = cfg["evaluation"]
    if ev["decoding"] != "greedy":
        raise ValueError("config evaluation.decoding is not greedy")
    model_name, revision = student_from_args(args, cfg)
    dtype = pick_dtype(torch, args.dtype)
    batch_size = args.batch_size or (200 if args.backend == "vllm" else 16)

    rows = load_dev_rows() if args.split == "dev" else load_test_rows()
    if args.limit:
        rows = rows[: args.limit]
    args.out.parent.mkdir(parents=True, exist_ok=True)
    done = {r["problem_id"] for r in read_jsonl_resumable(args.out)}
    todo = [r for r in rows if r["problem_id"] not in done]
    print(f"[{args.tag}] {args.split}: {len(rows)} problems, {len(done)} already done, {len(todo)} to run, "
          f"backend={args.backend}, dtype={dtype}")

    tok = load_tokenizer(model_name, revision)
    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    t0 = time.time()
    n_done = 0
    if todo:
        if args.backend == "vllm":
            batches = generate_vllm(todo, tok, model_name, revision, args.adapter, ev["max_new_tokens"], batch_size,
                                    args.gpu_memory_utilization)
        else:
            batches = generate_hf(todo, tok, model_name, revision, dtype, args.adapter, ev["max_new_tokens"], batch_size)
        for batch in batches:
            for r, text, n_out in batch:
                append_jsonl(args.out, score_output(r, args.tag, text, n_out, ev["max_new_tokens"]))
            n_done += len(batch)
            print(f"[{args.tag}] {n_done}/{len(todo)}  {time.time() - t0:.0f}s", flush=True)
    elapsed = time.time() - t0

    wanted = {x["problem_id"] for x in rows}
    results = [r for r in read_jsonl_resumable(args.out) if r["problem_id"] in wanted]
    if len(results) != len(rows):
        raise RuntimeError(f"{args.out} has {len(results)} of {len(rows)} problems")
    labels = load_capability_labels() if args.split == "dev" else None
    summary = {
        "checkpoint": args.tag,
        "split": args.split,
        "backend": args.backend,
        "model": model_name,
        "revision": revision,
        "adapter": args.adapter,
        "dtype": "bfloat16" if args.backend == "vllm" else str(dtype).replace("torch.", ""),
        **summarize(results, labels),
        "decoding": {"do_sample": False, "max_new_tokens": ev["max_new_tokens"], "repetition_penalty": 1.0,
                     "prompt_version": ev["prompt_version"]},
        "batch_size": batch_size,
        "limit": args.limit,
        "elapsed_seconds_this_run": round(elapsed, 1),
        "problems_this_run": len(todo),
        "peak_gpu_memory_gb": round(torch.cuda.max_memory_allocated() / 2**30, 2) if torch.cuda.is_available() else None,
        "environment": environment_info(torch),
    }
    summary_path = args.summary or args.out.with_suffix(".summary.json")
    write_json(summary_path, summary)
    print(f"[{args.tag}] accuracy {summary['accuracy']:.4f}  lenient {summary['lenient_accuracy']:.4f}  "
          f"format {summary['format_compliance']:.4f}  mean_len {summary['mean_output_tokens']:.1f}  -> {summary_path}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Score every verified pool trajectory under a Student checkpoint (one forward pass, no backward).

Per trajectory tau (trajectory tokens only, the end-of-turn token excluded):
  nll_sum    = sum_t -log p(tau_t | x, tau_<t)
  ell        = nll_sum / n_tokens            (length-normalized NLL; input of L_proxy, config "learnability")
  brier_sum  = sum_t || pi_t - onehot(tau_t) ||^2
               (Brier residual; stored so the LARK score, arXiv 2605.30651 Eq. 7, can be computed
                later without another pass. Exploratory, not part of the frozen L_proxy.)

L_proxy itself is not computed here: it needs ell over the whole pool (median / IQR rule) and
is derived in the analysis step.

--pool defaults to the official pool v1 (trajectory/pool/pool_v1_verified.jsonl). The Task 5 smoke test
scored the preliminary pool; reproduce it with --pool trajectory/pool/batch_100_teacher_verified.jsonl.

Usage (from experiments/week2September/manual/):
  python scripts/compute_nll.py --tag S0 --out reports/task5/smoke_test/nll_pool_S0.jsonl \
      --pool trajectory/pool/batch_100_teacher_verified.jsonl
  python scripts/compute_nll.py --tag S1_seed13 --adapter <adapter_dir> \
      --out reports/task5/smoke_test/nll_pool_S1_seed13.jsonl --pool trajectory/pool/batch_100_teacher_verified.jsonl
"""
import argparse
import json
import sys
import time
from pathlib import Path

import torch

sys.path.insert(0, str(Path(__file__).resolve().parent))
from student_common import (  # noqa: E402
    IGNORE_INDEX, POOL_VERIFIED, add_model_args, check_prompt_template, encode_example, environment_info,
    load_config, load_model, load_pool, load_tokenizer, pick_dtype, sha256_file, student_from_args, write_json,
)


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--tag", required=True)
    ap.add_argument("--adapter", default=None)
    ap.add_argument("--out", required=True, type=Path)
    ap.add_argument("--batch-size", type=int, default=4)
    ap.add_argument("--limit", type=int, default=None)
    ap.add_argument("--pool", type=Path, default=POOL_VERIFIED, help="verified pool to score (default: pool v1)")
    add_model_args(ap)
    args = ap.parse_args()

    check_prompt_template()
    cfg = load_config()
    model_name, revision = student_from_args(args, cfg)
    dtype = pick_dtype(torch, args.dtype)
    pool = sorted(load_pool(args.pool), key=lambda r: r["trajectory_id"])
    if args.limit:
        pool = pool[: args.limit]

    tok = load_tokenizer(model_name, revision)
    model = load_model(model_name, revision, dtype, adapter=args.adapter)
    examples = [encode_example(tok, r["question"], r["raw_generation"], add_eos=False) for r in pool]

    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    t0 = time.time()
    args.out.parent.mkdir(parents=True, exist_ok=True)
    with args.out.open("w", encoding="utf-8") as f:
        for i in range(0, len(examples), args.batch_size):
            batch = examples[i: i + args.batch_size]
            width = max(len(e["input_ids"]) for e in batch)
            ids = torch.tensor([e["input_ids"] + [tok.pad_token_id] * (width - len(e["input_ids"])) for e in batch])
            mask = torch.tensor([[1] * len(e["input_ids"]) + [0] * (width - len(e["input_ids"])) for e in batch])
            labels = torch.tensor([e["labels"] + [IGNORE_INDEX] * (width - len(e["labels"])) for e in batch])
            ids, mask, labels = ids.to(model.device), mask.to(model.device), labels.to(model.device)
            with torch.no_grad():
                logits = model(input_ids=ids, attention_mask=mask).logits[:, :-1]
            tgt = labels[:, 1:]
            for b, e in enumerate(batch):
                sel = tgt[b] != IGNORE_INDEX
                lp = torch.log_softmax(logits[b][sel].float(), dim=-1)
                y = tgt[b][sel]
                lp_y = lp.gather(1, y[:, None]).squeeze(1)
                p = lp.exp()
                brier = (p.pow(2).sum(-1) - 2 * lp_y.exp() + 1).sum().item()
                nll_sum = -lp_y.sum().item()
                r = pool[i + b]
                f.write(json.dumps({
                    "trajectory_id": r["trajectory_id"], "problem_id": r["problem_id"], "checkpoint": args.tag,
                    "n_tokens": int(sel.sum()), "nll_sum": nll_sum, "ell": nll_sum / int(sel.sum()),
                    "brier_sum": brier,
                }) + "\n")
            print(f"[{args.tag}] {min(i + args.batch_size, len(examples))}/{len(examples)}  {time.time() - t0:.0f}s",
                  flush=True)
    elapsed = time.time() - t0
    write_json(args.out.with_suffix(".summary.json"), {
        "checkpoint": args.tag, "adapter": args.adapter, "model": model_name, "revision": revision,
        "dtype": str(dtype).replace("torch.", ""), "n_trajectories": len(examples),
        "pool": str(args.pool), "pool_sha256": sha256_file(args.pool),
        "forward_passes": len(examples), "elapsed_seconds": round(elapsed, 1),
        "peak_gpu_memory_gb": round(torch.cuda.max_memory_allocated() / 2**30, 2) if torch.cuda.is_available() else None,
        "environment": environment_info(torch),
    })
    print(f"[{args.tag}] scored {len(examples)} trajectories in {elapsed:.0f}s -> {args.out}")


if __name__ == "__main__":
    main()

def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish a completed staging directory without overwriting a target.

    Retries transient Windows PermissionError failures. On terminal failure,
    the staging directory is intentionally preserved for recovery.
    """
    retry_delays = (0.25, 0.5, 1.0, 2.0, 4.0)

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)
            return
        except PermissionError as exc:
            if out_dir.exists():
                raise FileExistsError(
                    "OUTPUT_COLLISION: destination appeared during "
                    "publication; refusing to overwrite: " + str(out_dir)
                ) from exc

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)

#!/usr/bin/env python3
"""Build an immutable P1 capability state from a validated sidecar."""
from __future__ import annotations

import argparse
import hashlib
import json
import math
import os
import tempfile
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


CAPABILITIES = (
    "addition",
    "subtraction",
    "multiplication",
    "division",
    "percentage",
    "fraction",
    "ratio",
)

EXPECTED_SUPPORT = {
    "addition": 30,
    "subtraction": 51,
    "multiplication": 73,
    "division": 47,
    "percentage": 29,
    "fraction": 31,
    "ratio": 30,
}

SOURCE_METHODS = {
    "B2", "B3", "B4", "B5", "P1-DEFICIT", "P1P2"
}

EXPECTED_OUTPUT_NAMES = {
    "capability_metrics.csv",
    "capability_metrics.jsonl",
    "prediction_coverage_report.json",
    "checkpoint_lineage.json",
    "input_fingerprint.json",
}


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    with path.open("r", encoding="utf-8-sig") as stream:
        value = json.load(stream)
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def stable_state(value: dict[str, Any]) -> dict[str, Any]:
    return {
        key: item
        for key, item in value.items()
        if key != "created_at_utc"
    }


def require_equal(label: str, left: Any, right: Any) -> None:
    if left != right:
        raise ValueError(
            f"PROVENANCE_MISMATCH {label}: "
            f"manifest={left!r}, lineage={right!r}"
        )


def validate_source_identity(
    manifest: dict[str, Any],
    lineage: dict[str, Any],
    fingerprint: dict[str, Any],
) -> tuple[int, str, int, str]:
    required_fields = (
        "run_id", "method", "seed", "round",
        "checkpoint_id", "checkpoint_sha256",
    )

    for field in required_fields:
        if field not in lineage:
            raise ValueError(f"Missing lineage field: {field}")

    for field in ("run_id", "method", "seed", "round"):
        if field not in manifest:
            raise ValueError(f"Missing manifest field: {field}")

    for field in ("run_id", "method", "seed", "round"):
        require_equal(field, manifest[field], lineage[field])

    source_round = int(lineage["round"])
    source_method = str(lineage["method"])
    source_seed = int(lineage["seed"])
    run_id = str(lineage["run_id"])

    if source_round not in (0, 1, 2, 3):
        raise ValueError(f"Invalid source round: {source_round}")

    if source_round == 0:
        if source_method != "SHARED" or source_seed != 0:
            raise ValueError("S0 must originate from SHARED seed 0")
        expected_run_id = "SHARED_seed0_S0"
    else:
        if source_method not in SOURCE_METHODS:
            raise ValueError(f"Unsupported source method: {source_method}")
        expected_run_id = (
            f"{source_method}_seed{source_seed}_S{source_round}"
        )

    if run_id != expected_run_id:
        raise ValueError(
            f"RUN_ID_MISMATCH: {run_id!r} != {expected_run_id!r}"
        )

    expected_checkpoint = f"S{source_round}"
    if manifest.get("checkpoint") != expected_checkpoint:
        raise ValueError(
            "Manifest checkpoint does not match source round: "
            f"{manifest.get('checkpoint')!r} != {expected_checkpoint!r}"
        )

    checkpoint_id = str(lineage["checkpoint_id"]).strip()
    checkpoint_sha256 = str(lineage["checkpoint_sha256"]).lower()

    if not checkpoint_id:
        raise ValueError("Empty checkpoint_id")

    if len(checkpoint_sha256) != 64:
        raise ValueError("Invalid checkpoint_sha256 length")

    if any(ch not in "0123456789abcdef" for ch in checkpoint_sha256):
        raise ValueError("Invalid checkpoint_sha256 characters")

    manifest_inputs = manifest.get("input_hashes")
    fingerprint_inputs = fingerprint.get("inputs")

    if not isinstance(manifest_inputs, dict):
        raise ValueError("Manifest input_hashes must be an object")

    if not isinstance(fingerprint_inputs, dict):
        raise ValueError("Fingerprint inputs must be an object")

    if manifest_inputs != fingerprint_inputs:
        raise ValueError(
            "Manifest input_hashes do not match fingerprint inputs"
        )

    if fingerprint_inputs.get("checkpoint") != checkpoint_sha256:
        raise ValueError(
            "Checkpoint hash in lineage does not match input fingerprint"
        )

    expected_fingerprint = hashlib.sha256(
        json.dumps(
            fingerprint_inputs,
            sort_keys=True,
            separators=(",", ":"),
        ).encode("utf-8")
    ).hexdigest()

    if fingerprint.get("fingerprint") != expected_fingerprint:
        raise ValueError("Input fingerprint digest is invalid")

    return source_round, source_method, source_seed, run_id


def validate_metrics(metrics_path: Path) -> dict[str, dict[str, Any]]:
    by_capability: dict[str, dict[str, Any]] = {}

    with metrics_path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue

            row = json.loads(line)
            if not isinstance(row, dict):
                raise ValueError(f"Invalid metric row at line {line_no}")

            capability = row.get("capability")
            if capability not in CAPABILITIES:
                raise ValueError(f"Unknown capability: {capability!r}")

            if capability in by_capability:
                raise ValueError(f"Duplicate capability row: {capability}")

            if row.get("strict_metric") != "status_equals_valid":
                raise ValueError("Capability row does not use strict metric")

            if row.get("gold_policy") != "approved_85_primary":
                raise ValueError("Capability row uses an invalid Gold policy")

            support = int(row["n_support"])
            correct = int(row["n_correct"])
            incorrect = int(row["n_incorrect"])

            if support != EXPECTED_SUPPORT[capability]:
                raise ValueError(
                    f"Support mismatch for {capability}: "
                    f"{support} != {EXPECTED_SUPPORT[capability]}"
                )

            if correct < 0 or incorrect < 0:
                raise ValueError(f"Negative metric count: {capability}")

            if correct + incorrect != support:
                raise ValueError(f"Count mismatch: {capability}")

            mastery = correct / support
            deficit = 1.0 - mastery

            if not math.isfinite(float(row["mastery"])):
                raise ValueError(f"Non-finite mastery: {capability}")

            if not math.isfinite(float(row["deficit"])):
                raise ValueError(f"Non-finite deficit: {capability}")

            if abs(float(row["mastery"]) - mastery) > 1e-12:
                raise ValueError(f"Mastery mismatch: {capability}")

            if abs(float(row["deficit"]) - deficit) > 1e-12:
                raise ValueError(f"Deficit mismatch: {capability}")

            by_capability[capability] = {
                "n_support": support,
                "n_correct": correct,
                "n_incorrect": incorrect,
                "mastery": mastery,
                "deficit": deficit,
                "ci95_low": row.get("ci95_low"),
                "ci95_high": row.get("ci95_high"),
            }

    if set(by_capability) != set(CAPABILITIES):
        raise ValueError(
            "Capability coverage is incomplete: "
            f"{sorted(by_capability)}"
        )

    return by_capability


def publish_exclusive(output: Path, content: bytes) -> None:
    """Publish through a same-directory hard link; never replace a target."""
    output.parent.mkdir(parents=True, exist_ok=True)

    fd, temp_name = tempfile.mkstemp(
        prefix=f".{output.name}.staging.",
        suffix=".tmp",
        dir=str(output.parent),
    )
    temp_path = Path(temp_name)

    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(content)
            stream.flush()
            os.fsync(stream.fileno())

        try:
            os.link(temp_path, output)
        except FileExistsError:
            raise
        except OSError as exc:
            raise RuntimeError(
                "ATOMIC_NO_OVERWRITE_UNAVAILABLE: hard-link publication "
                f"failed; temporary file preserved at {temp_path}: {exc}"
            ) from exc

        temp_path.unlink(missing_ok=True)

    except Exception:
        if temp_path.exists():
            print(f"RECOVERY_TEMP_PRESERVED={temp_path}")
        raise

from __future__ import annotations

import hashlib
import json
import os
import tempfile
from pathlib import Path
from typing import Any, Iterable, Mapping


def sha256_file(path: Path) -> str:
    """Return the SHA-256 digest of a regular file."""
    source = Path(path).resolve()
    if not source.is_file():
        raise FileNotFoundError(f"Input file not found: {source}")
    digest = hashlib.sha256()
    with source.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def canonical_json_bytes(value: Any) -> bytes:
    """Serialize JSON deterministically for stable fingerprints."""
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n").encode("utf-8")


def sha256_json(value: Any) -> str:
    return hashlib.sha256(canonical_json_bytes(value)).hexdigest()


def fingerprint_files(paths: Mapping[str, Path]) -> dict[str, Any]:
    """Fingerprint named input files; names are stable logical identifiers."""
    if not paths:
        raise ValueError("At least one input file is required for a fingerprint")
    entries: dict[str, dict[str, Any]] = {}
    resolved_seen: dict[str, str] = {}
    for name in sorted(paths):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Fingerprint input names must be non-empty strings")
        path = Path(paths[name]).resolve()
        if not path.is_file():
            raise FileNotFoundError(f"Fingerprint input '{name}' not found: {path}")
        normalized = str(path).casefold()
        if normalized in resolved_seen:
            raise ValueError(f"Duplicate fingerprint path for '{name}' and '{resolved_seen[normalized]}'")
        resolved_seen[normalized] = name
        entries[name] = {
            "path": str(path),
            "size_bytes": path.stat().st_size,
            "sha256": sha256_file(path),
        }
    payload = {"schema_version": "p1p2.input-fingerprint.v1", "files": entries}
    payload["fingerprint_sha256"] = sha256_json(payload)
    return payload


def verify_fingerprint(fingerprint: Mapping[str, Any]) -> bool:
    """Verify a fingerprint object's internal canonical digest."""
    if not isinstance(fingerprint, Mapping):
        raise TypeError("Fingerprint must be a mapping")
    if fingerprint.get("schema_version") != "p1p2.input-fingerprint.v1":
        raise ValueError("Unsupported fingerprint schema")
    files = fingerprint.get("files")
    digest = fingerprint.get("fingerprint_sha256")
    if not isinstance(files, Mapping) or not isinstance(digest, str):
        raise ValueError("Fingerprint is missing files or fingerprint_sha256")
    unsigned = {key: value for key, value in fingerprint.items() if key != "fingerprint_sha256"}
    actual = sha256_json(unsigned)
    if actual != digest:
        raise ValueError("Fingerprint digest mismatch")
    return True


def verify_fingerprint_files(fingerprint: Mapping[str, Any]) -> bool:
    """Verify that every recorded file still has its recorded size and hash."""
    verify_fingerprint(fingerprint)
    files = fingerprint["files"]
    for name, entry in files.items():
        if not isinstance(entry, Mapping):
            raise ValueError(f"Invalid fingerprint entry: {name}")
        path = Path(entry.get("path", ""))
        if not path.is_file():
            raise FileNotFoundError(f"Fingerprint source missing: {name}: {path}")
        if path.stat().st_size != entry.get("size_bytes"):
            raise ValueError(f"Fingerprint size mismatch: {name}")
        if sha256_file(path) != entry.get("sha256"):
            raise ValueError(f"Fingerprint SHA-256 mismatch: {name}")
    return True


def ensure_no_overwrite(paths: Iterable[Path]) -> list[Path]:
    """Resolve and preflight output paths without creating or changing them."""
    resolved: list[Path] = []
    seen: set[str] = set()
    for raw_path in paths:
        path = Path(raw_path).resolve()
        key = str(path).casefold()
        if key in seen:
            raise ValueError(f"Duplicate output path: {path}")
        seen.add(key)
        if path.exists():
            raise FileExistsError(f"Refusing to overwrite existing artifact: {path}")
        resolved.append(path)
    if not resolved:
        raise ValueError("At least one output path is required")
    return resolved


def _write_temp_for_target(target: Path, content: bytes) -> Path:
    target.parent.mkdir(parents=True, exist_ok=True)
    fd, raw_temp = tempfile.mkstemp(prefix=f".{target.name}.", suffix=".tmp", dir=str(target.parent))
    temp_path = Path(raw_temp)
    try:
        with os.fdopen(fd, "wb") as handle:
            handle.write(content)
            handle.flush()
            os.fsync(handle.fileno())
    except Exception:
        temp_path.unlink(missing_ok=True)
        raise
    return temp_path


def publish_bytes_no_overwrite(target: Path, content: bytes) -> Path:
    """Atomically publish one file without replacing an existing path."""
    if not isinstance(content, bytes):
        raise TypeError("content must be bytes")
    destination = Path(target).resolve()
    ensure_no_overwrite([destination])
    temp_path = _write_temp_for_target(destination, content)
    try:
        # A same-filesystem hard link is atomic and fails if destination exists.
        os.link(temp_path, destination)
    finally:
        temp_path.unlink(missing_ok=True)
    return destination


def publish_json_no_overwrite(target: Path, value: Any) -> Path:
    return publish_bytes_no_overwrite(target, canonical_json_bytes(value))


def publish_pair_no_overwrite(first_path: Path, first_content: bytes,
                              second_path: Path, second_content: bytes) -> tuple[Path, Path]:
    """Publish two files; roll back the first if publishing the second fails."""
    first = Path(first_path).resolve()
    second = Path(second_path).resolve()
    ensure_no_overwrite([first, second])
    first_temp = _write_temp_for_target(first, first_content)
    second_temp: Path | None = None
    first_published = False
    try:
        second_temp = _write_temp_for_target(second, second_content)
        os.link(first_temp, first)
        first_published = True
        os.link(second_temp, second)
    except Exception:
        if first_published and first.exists():
            expected = hashlib.sha256(first_content).hexdigest()
            if sha256_file(first) == expected:
                first.unlink()
        raise
    finally:
        first_temp.unlink(missing_ok=True)
        if second_temp is not None:
            second_temp.unlink(missing_ok=True)
    return first, second


def publish_manifest_no_overwrite(target: Path, manifest: Mapping[str, Any],
                                   input_paths: Mapping[str, Path]) -> Path:
    """Attach a reproducible input fingerprint and publish a run manifest."""
    if not isinstance(manifest, Mapping):
        raise TypeError("manifest must be a mapping")
    payload = dict(manifest)
    if "input_fingerprint" in payload:
        raise ValueError("manifest must not supply input_fingerprint directly")
    payload["input_fingerprint"] = fingerprint_files(input_paths)
    return publish_json_no_overwrite(target, payload)

def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    with Path(path).open("r", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(f"Invalid JSONL at {path}:{line_number}: {exc}") from exc
            if not isinstance(row, dict):
                raise ValueError(f"Expected JSON object at {path}:{line_number}")
            rows.append(row)
    return rows

def average_percentile_ranks(scores: dict[str, float]) -> dict[str, float]:
    if not scores:
        raise ValueError("Cannot rank an empty score mapping")
    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    n = len(ordered)
    if n == 1:
        return {ordered[0][0]: 0.5}
    result: dict[str, float] = {}
    index = 0
    while index < n:
        end = index + 1
        value = ordered[index][1]
        while end < n and ordered[end][1] == value:
            end += 1
        average_rank = ((index + 1) + end) / 2.0
        percentile = (average_rank - 1.0) / (n - 1.0)
        for position in range(index, end):
            result[ordered[position][0]] = percentile
        index = end
    return result


def calculate_p1_scores(pool_by_id: dict[str, dict], deficits: dict[str, float]) -> dict[str, float]:
    result: dict[str, float] = {}
    for tid, row in pool_by_id.items():
        tags = row["capabilities"]
        result[tid] = sum(deficits[tag] for tag in tags) / len(tags)
    return result


def publish_pair_no_overwrite(train_set_path: Path, selection_log_path: Path,
                              train_set_bytes: bytes, selection_log_bytes: bytes) -> None:
    train_set_path, selection_log_path = Path(train_set_path), Path(selection_log_path)
    if train_set_path.resolve() == selection_log_path.resolve():
        raise ValueError("Train-set and selection-log paths must be different")
    train_set_path.parent.mkdir(parents=True, exist_ok=True)
    selection_log_path.parent.mkdir(parents=True, exist_ok=True)
    if train_set_path.exists() or selection_log_path.exists():
        raise FileExistsError("Refusing to overwrite an existing selector output")
    nonce = f"{os.getpid()}"
    train_tmp = train_set_path.with_name(train_set_path.name + f".{nonce}.tmp")
    log_tmp = selection_log_path.with_name(selection_log_path.name + f".{nonce}.tmp")
    created_train = False
    created_temps: list[Path] = []
    try:
        for temp_path, content in ((train_tmp, train_set_bytes), (log_tmp, selection_log_bytes)):
            flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
            fd = os.open(str(temp_path), flags, 0o600)
            created_temps.append(temp_path)
            with os.fdopen(fd, "wb") as handle:
                handle.write(content)
                handle.flush()
                os.fsync(handle.fileno())
        os.link(train_tmp, train_set_path)
        created_train = True
        os.link(log_tmp, selection_log_path)
    except Exception:
        if created_train and train_set_path.exists() and sha256_file(train_set_path) == hashlib.sha256(train_set_bytes).hexdigest():
            train_set_path.unlink()
        raise
    finally:
        for temp_path in created_temps:
            temp_path.unlink(missing_ok=True)

def run_selection(args: argparse.Namespace) -> dict[str, Any]:
    if args.seed < 0:
        raise ValueError("seed must be non-negative")
    if args.method == "P1-DEFICIT" and args.nll is not None:
        raise ValueError("P1-DEFICIT does not accept an NLL input")
    if args.method == "P1P2" and args.nll is None:
        raise ValueError("P1P2 requires --nll for the current source checkpoint")
    source_checkpoint = f"S{args.round - 1}"
    config_path = _require_file(args.config, "Frozen experiment config")
    config = read_json(config_path)
    selection_cfg = config.get("selection")
    budget_cfg = config.get("budget")
    if not isinstance(selection_cfg, dict) or not isinstance(budget_cfg, dict):
        raise ValueError("Frozen config is missing selection or budget section")
    alpha = selection_cfg.get("alpha")
    alpha_tuned = selection_cfg.get("alpha_tuned")
    if alpha != EXPECTED_ALPHA or alpha_tuned is not False:
        raise ValueError(f"Frozen alpha policy mismatch: expected alpha={EXPECTED_ALPHA}, alpha_tuned=false; got {alpha}, {alpha_tuned}")
    budget = budget_cfg.get("beta_round_tokens")
    n_cap = selection_cfg.get("n_cap")
    if not isinstance(budget, int) or isinstance(budget, bool) or budget <= 0:
        raise ValueError("Invalid beta_round_tokens in frozen config")
    if n_cap != 1:
        raise ValueError(f"Frozen protocol requires n_cap=1, found {n_cap}")
    if selection_cfg.get("allow_reselect_across_rounds") is not True:
        raise ValueError("Frozen protocol requires allow_reselect_across_rounds=true")

    tokens_path = _require_file(args.tokens, "Token table")
    pool_hash = sha256_file(pool_path)
    pool_by_id, problems = load_pool(pool_path, args.d_pool)
    tokens = load_tokens(tokens_path, pool_path, problems)
    deficits = load_capability_deficits(args.state, args.method, args.seed, args.round)
    trained_ids = load_previous_train_ids(args.previous_train_set, problems, args.method, args.seed, args.round, pool_hash)
    p1_scores = calculate_p1_scores(pool_by_id, deficits)
    p2_scores = None
    p2_percentiles = None
    ell_by_id = None
    ell_star = sigma = n_reference = nll_sha256 = None
    if args.method == "P1P2":
        ell_by_id, nll_sha256 = load_nll(args.nll, tokens, source_checkpoint, pool_path)
        scoring = score_checkpoint(ell_by_id, trained_ids)
        p2_scores = scoring["l_proxy"]
        ell_star, sigma, n_reference = scoring["ell_star"], scoring["sigma"], scoring["n_reference"]
        p2_percentiles = average_percentile_ranks(p2_scores)
        combined_scores = {tid: EXPECTED_ALPHA * p1_scores[tid] + (1.0 - EXPECTED_ALPHA) * p2_percentiles[tid] for tid in problems}
    else:
        combined_scores = dict(p1_scores)
    order = rank_by_score(combined_scores)
    selected, reasons, tokens_used, stopped = fill_budget(order, problems, tokens, budget, n_cap)
    if not stopped:
        raise ValueError(f"Candidate order exhausted before budget stop: {tokens_used}/{budget}")
    if not selected:
        raise ValueError("Selector returned an empty train set")
    if tokens_used > budget:
        raise ValueError(f"Budget violation: {tokens_used} > {budget}")
    ranks = {tid: rank for rank, tid in enumerate(order, 1)}

    for tid in sorted(problems):
        reason = reasons.get(tid, "not_selected")
        row = pool_by_id[tid]
        log_rows.append({
            "record_type": "selection", "policy_version": POLICY_VERSION, "method": args.method,

            "p2_l_proxy": p2_scores[tid] if p2_scores is not None else None,
            "p2_percentile_rank": p2_percentiles[tid] if p2_percentiles is not None else None,
            "combined_score": combined_scores[tid], "rank": ranks[tid],
            "selected": reason == "selected", "reason": reason,
            "nll_ell": ell_by_id[tid] if ell_by_id is not None else None,
        })
    train_set = {
        "schema_version": "p1p2.train-set.v1", "method": args.method, "training_seed": args.seed,
        "round": args.round, "source_checkpoint": source_checkpoint,
        "priority_policy": {"version": POLICY_VERSION, "alpha": EXPECTED_ALPHA if args.method == "P1P2" else 1.0,
                             "p1_component": "mean_capability_deficit",
                             "p2_component": "average_percentile_rank_of_l_proxy" if args.method == "P1P2" else None,
                             "percentile_tie_policy": "average_rank"},
        "pool": str(pool_path), "pool_sha256": pool_hash,
        "tokens_file": str(tokens_path), "tokens_sha256": sha256_file(tokens_path),
        "tokens_summary_sha256": sha256_file(tokens_path.with_suffix(".summary.json")),
        "config": str(config_path), "config_sha256": sha256_file(config_path),
        "state_file": str(Path(args.state).resolve()), "state_sha256": sha256_file(args.state),
        "nll_file": str(Path(args.nll).resolve()) if args.nll else None, "nll_sha256": nll_sha256,
        "ell_star": ell_star, "sigma": sigma, "n_reference": n_reference,
        "n_eligible": len(problems), "n_trained_before": len(trained_ids),
        "trained_reference_exclusions": trained_ids, "beta_round_tokens": budget, "n_cap": n_cap,
        "allow_reselect_across_rounds": True, "tokens_used": tokens_used, "n_selected": len(selected),
        "trajectory_ids": selected,
    }
    out_dir = Path(args.out_dir).resolve()
    train_path, log_path = out_dir / "train_set.json", out_dir / "selection_log.jsonl"
    train_bytes = (json.dumps(train_set, ensure_ascii=False, indent=2) + "\n").encode("utf-8")
    log_bytes = "".join(json.dumps(row, ensure_ascii=False) + "\n" for row in log_rows).encode("utf-8")
    publish_pair_no_overwrite(train_path, log_path, train_bytes, log_bytes)
    return {"method": args.method, "seed": args.seed, "round": args.round, "source_checkpoint": source_checkpoint,
            "n_eligible": len(problems), "n_trained_before": len(trained_ids), "n_selected": len(selected),
            "tokens_used": tokens_used, "budget": budget, "train_set": train_path, "selection_log": log_path}

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def hash_path(path: Path) -> str:
    path = path.resolve()

    if path.is_file():
        return sha256_file(path)

    if not path.is_dir():
        raise FileNotFoundError(f"Input does not exist: {path}")

    files = sorted(p for p in path.rglob("*") if p.is_file())
    if not files:
        raise ValueError(f"Checkpoint directory is empty: {path}")

    digest = hashlib.sha256()
    for file_path in files:
        relative = file_path.relative_to(path).as_posix()
        digest.update(relative.encode("utf-8"))
        digest.update(b"\0")
        digest.update(sha256_file(file_path).encode("ascii"))
        digest.update(b"\n")

    return digest.hexdigest()


def read_json(path: Path) -> dict[str, Any]:
    with path.open("r", encoding="utf-8-sig") as stream:
        value = json.load(stream)
    if not isinstance(value, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return value


def read_jsonl(path: Path) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []

    with path.open("r", encoding="utf-8-sig") as stream:
        for line_no, line in enumerate(stream, 1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSONL at {path}:{line_no}: {exc}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected JSON object at {path}:{line_no}"
                )
            rows.append(row)

    return rows


def read_id_file(path: Path) -> set[str]:
    values = [
        line.strip()
        for line in path.read_text(encoding="utf-8-sig").splitlines()
        if line.strip()
    ]

    if len(values) != len(set(values)):
        raise ValueError(f"Duplicate IDs in {path}")

    return set(values)


def wilson_interval(
    correct: int,
    support: int,
    z: float = 1.959963984540054,
) -> tuple[float | None, float | None]:
    if support <= 0:
        return None, None

    p = correct / support
    z2 = z * z
    denominator = 1.0 + z2 / support
    center = (p + z2 / (2.0 * support)) / denominator
    margin = (
        z
        * (
            p * (1.0 - p) / support
            + z2 / (4.0 * support * support)
        ) ** 0.5
        / denominator
    )

    return max(0.0, center - margin), min(1.0, center + margin)


def strict_correct(row: dict[str, Any]) -> bool:
    """P2 primary strict metric: only status == 'valid' is correct."""
    status = row.get("status")

    if status not in {"valid", "invalid_answer", "invalid_format"}:
        raise ValueError(
            "Prediction must contain a recognized P2 status; "
            f"got status={status!r}, problem_id={row.get('problem_id')!r}"
        )

    return status == "valid"


def load_approved_gold(
    gold_path: Path,
    approval_path: Path,
    dev_ids: set[str],
) -> dict[str, dict[str, Any]]:
    gold_rows = read_jsonl(gold_path)

    if len(gold_rows) != 85:
        raise ValueError(
            f"Expected exactly 85 Gold rows; found {len(gold_rows)}"
        )

    gold_by_id: dict[str, dict[str, Any]] = {}

    for row in gold_rows:
        problem_id = str(row.get("problem_id", "")).strip()
        if not problem_id:
            raise ValueError("Gold row missing problem_id")
        if problem_id in gold_by_id:
            raise ValueError(f"Duplicate Gold problem_id: {problem_id}")

        labels = row.get("capabilities_manual")
        if not isinstance(labels, list) or not labels:
            raise ValueError(
                f"Gold capability labels missing/invalid: {problem_id}"
            )

        unknown = set(labels) - set(CAPABILITIES)
        if unknown:
            raise ValueError(
                f"Unknown capabilities for {problem_id}: {sorted(unknown)}"
            )

        gold_by_id[problem_id] = row

    gold_ids = set(gold_by_id)

    if len(gold_ids) != 85:
        raise ValueError("Gold IDs are not 85 unique IDs")

    if not gold_ids.issubset(dev_ids):
        outside = sorted(gold_ids - dev_ids)
        raise ValueError(
            f"Gold IDs outside D_dev: {outside[:10]}"
        )

    approval_by_id: dict[str, dict[str, str]] = {}

    with approval_path.open(
        "r", encoding="utf-8-sig", newline=""
    ) as stream:
        reader = csv.DictReader(stream)

        required_columns = {
            "problem_id",
            "review_status",
            "human_approval",
        }
        actual_columns = set(reader.fieldnames or [])
        missing_columns = required_columns - actual_columns

        if missing_columns:
            raise ValueError(
                f"Approval CSV missing columns: {sorted(missing_columns)}"
            )

        for row in reader:
            problem_id = str(row.get("problem_id", "")).strip()
            if not problem_id:
                raise ValueError("Approval CSV contains empty problem_id")
            if problem_id in approval_by_id:
                raise ValueError(
                    f"Duplicate approval problem_id: {problem_id}"
                )
            approval_by_id[problem_id] = row

    approval_ids = set(approval_by_id)

    if approval_ids != gold_ids:
        missing = sorted(gold_ids - approval_ids)
        extra = sorted(approval_ids - gold_ids)
        raise ValueError(
            "Approval/Gold ID set mismatch: "
            f"missing_approval={missing[:10]}, "
            f"extra_approval={extra[:10]}"
        )

    for problem_id, row in approval_by_id.items():
        review_status = str(row.get("review_status", "")).strip()
        human_approval = str(row.get("human_approval", "")).strip()

        approved_review_statuses = {
            "APPROVED_NO_LABEL_CHANGE",
            "APPROVED_WITH_SOURCE_CLARIFICATION",
        }

        if review_status not in approved_review_statuses:
            raise ValueError(
                f"Gold review status is not approved: "
                f"{problem_id}={review_status!r}"
            )

        if not human_approval.startswith("APPROVED_BY_USER_"):
            raise ValueError(
                f"Gold lacks explicit user approval: "
                f"{problem_id}={human_approval!r}"
            )

    return gold_by_id


def build_input_hashes(args: argparse.Namespace) -> dict[str, str]:
    input_paths = {
        "checkpoint": Path(args.checkpoint),
        "dev_predictions": Path(args.dev_predictions),
        "gold_manifest": Path(args.gold_manifest),
        "gold_labels": Path(args.gold_labels),
        "approval_csv": Path(args.approval_csv),
        "dev_ids": Path(args.dev_ids),
        "config": Path(args.config),
    }

    optional_paths = {
        "train_set": args.train_set,
        "selection_log": args.selection_log,
        "train_log": args.train_log,
        "checkpoint_log": args.checkpoint_log,
    }

    for name, value in optional_paths.items():
        if value:
            input_paths[name] = Path(value)

    hashes: dict[str, str] = {}
    for name, path in input_paths.items():
        if not path.exists():
            raise FileNotFoundError(f"Missing input {name}: {path}")
        hashes[name] = hash_path(path)

    return hashes

def build_sidecar(
    args: argparse.Namespace,
    stage: Path,
    input_hashes: dict[str, str],
) -> None:
    dev_ids = read_id_file(Path(args.dev_ids))

    if len(dev_ids) != 500:
        raise ValueError(
            f"Expected 500 unique D_dev IDs; found {len(dev_ids)}"
        )

    predictions = read_jsonl(Path(args.dev_predictions))
    prediction_ids = [
        str(row.get("problem_id", "")).strip()
        for row in predictions
    ]

    if any(not value for value in prediction_ids):
        raise ValueError("Prediction contains empty problem_id")

    if len(prediction_ids) != len(set(prediction_ids)):
        raise ValueError("Duplicate problem_id in D_dev predictions")

    if set(prediction_ids) != dev_ids or len(predictions) != 500:
        missing = sorted(dev_ids - set(prediction_ids))
        extra = sorted(set(prediction_ids) - dev_ids)
        raise ValueError(
            "Prediction coverage mismatch: "
            f"rows={len(predictions)}, "
            f"missing={missing[:10]}, extra={extra[:10]}"
        )

    prediction_by_id = {
        str(row["problem_id"]): row for row in predictions
    }

    # Validate status for every prediction, not only the 85 Gold rows.
    for row in predictions:
        strict_correct(row)

    validate_manifest_count(Path(args.gold_manifest))

    gold_by_id = load_approved_gold(
        Path(args.gold_labels),
        Path(args.approval_csv),
        dev_ids,
    )

    metrics: list[dict[str, Any]] = []

    for capability in CAPABILITIES:
        capability_ids = sorted(
            problem_id
            for problem_id, gold in gold_by_id.items()
            if capability in gold["capabilities_manual"]
        )

        support = len(capability_ids)
        correct = sum(
            strict_correct(prediction_by_id[problem_id])
            for problem_id in capability_ids
        )

        ci_low, ci_high = wilson_interval(correct, support)

        metrics.append({
            "run_id": args.run_id,
            "method": args.method,
            "seed": args.seed,
            "checkpoint": f"S{args.round}",
            "checkpoint_id": Path(args.checkpoint).name,
            "capability": capability,
            "n_support": support,
            "n_correct": correct,
            "n_incorrect": support - correct,
            "mastery": correct / support if support else None,
            "deficit": 1.0 - correct / support if support else None,
            "ci95_low": ci_low,
            "ci95_high": ci_high,
            "ci_method": "Wilson_95_percent",
            "strict_metric": "status_equals_valid",
            "gold_policy": "approved_85_primary",
        })

    observed_support = {
        row["capability"]: row["n_support"] for row in metrics
    }

    if observed_support != EXPECTED_SUPPORT:
        raise ValueError(
            "Gold capability support differs from the frozen expected "
            f"vector. observed={observed_support}; "
            f"expected={EXPECTED_SUPPORT}"
        )

    with (stage / "capability_metrics.csv").open(
        "w", encoding="utf-8-sig", newline=""
    ) as stream:
        writer = csv.DictWriter(
            stream,
            fieldnames=list(metrics[0].keys()),
        )
        writer.writeheader()
        writer.writerows(metrics)

    with (stage / "capability_metrics.jsonl").open(
        "w", encoding="utf-8"
    ) as stream:
        for row in metrics:
            stream.write(
                json.dumps(row, ensure_ascii=False) + "\n"
            )

    coverage = {
        "status": "PASS",
        "split": "D_dev",
        "expected_dev_ids": 500,
        "prediction_records": len(predictions),
        "prediction_unique_ids": len(set(prediction_ids)),
        "gold_records": len(gold_by_id),
        "gold_unique_ids": len(gold_by_id),
        "gold_matched_prediction_ids": len(
            set(gold_by_id) & set(prediction_ids)
        ),
        "missing_prediction_ids": [],
        "gold_policy": "approved_85_primary",
        "strict_correctness_policy": "status == valid",
        "lenient_correct_excluded": True,
        "capability_support": observed_support,
    }

    (stage / "prediction_coverage_report.json").write_text(
        json.dumps(coverage, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    lineage = {
        "run_id": args.run_id,
        "method": args.method,
        "seed": args.seed,
        "round": args.round,
        "checkpoint_id": Path(args.checkpoint).name,
        "checkpoint_path": str(Path(args.checkpoint)),
        "checkpoint_sha256": input_hashes["checkpoint"],
        "shared_checkpoint_group": args.shared_checkpoint_group,
        "train_set": args.train_set,
        "selection_log": args.selection_log,
        "train_log": args.train_log,
        "checkpoint_log": args.checkpoint_log,
        "config_sha256": input_hashes["config"],
        "prediction_sha256": input_hashes["dev_predictions"],
        "gold_manifest_sha256": input_hashes["gold_manifest"],
        "gold_labels_sha256": input_hashes["gold_labels"],
        "approval_csv_sha256": input_hashes["approval_csv"],
        "dev_ids_sha256": input_hashes["dev_ids"],
    }

    (stage / "checkpoint_lineage.json").write_text(
        json.dumps(lineage, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    fingerprint = hashlib.sha256(
        json.dumps(
            input_hashes,
            sort_keys=True,
            separators=(",", ":"),
        ).encode("utf-8")
    ).hexdigest()

    (stage / "input_fingerprint.json").write_text(
        json.dumps({
            "fingerprint": fingerprint,
            "inputs": input_hashes,
        }, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    manifest = {
        "schema_version": "p1p2.capability-manifest.v1",
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "run_id": args.run_id,
        "method": args.method,
        "seed": args.seed,
        "round": args.round,
        "checkpoint": f"S{args.round}",
        "gold_policy": "approved_85_primary",
        "strict_metric": "status == valid",
        "capabilities": list(CAPABILITIES),
        "n_capability_rows": len(metrics),
        "input_hashes": input_hashes,
        "outputs": {},
        "overwrite_policy": "immutable_exact_fingerprint_reuse_only",
    }

    for name in REQUIRED_OUTPUTS:
        if name == "student_capability_evaluation_manifest.json":
            continue

        output_path = stage / name
        manifest["outputs"][name] = {
            "sha256": sha256_file(output_path),
            "size_bytes": output_path.stat().st_size,
        }

    (stage / "student_capability_evaluation_manifest.json").write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )


def verify_existing_sidecar(
    out_dir: Path,
    input_hashes: dict[str, str],
) -> None:
    manifest_path = (
        out_dir / "student_capability_evaluation_manifest.json"
    )

    if not manifest_path.is_file():
        raise RuntimeError(
            "EXISTING_PARTIAL_SIDECAR: manifest missing; "
            "do not overwrite or auto-delete this directory"
        )

    manifest = read_json(manifest_path)

    if manifest.get("input_hashes") != input_hashes:
        raise RuntimeError(
            "INPUT_FINGERPRINT_MISMATCH: sidecar is immutable"
        )

    outputs = manifest.get("outputs", {})
    expected_names = set(REQUIRED_OUTPUTS) - {
        "student_capability_evaluation_manifest.json"
    }

    if set(outputs) != expected_names:
        raise RuntimeError("Existing sidecar output inventory mismatch")

    for name, metadata in outputs.items():
        path = out_dir / name

        if not path.is_file():
            raise RuntimeError(f"Existing output missing: {path}")

        if sha256_file(path) != metadata.get("sha256"):
            raise RuntimeError(f"Existing output hash mismatch: {path}")

    print(f"REUSE: PASS {out_dir}")

    )

    if args.run_id != expected_run_id:
        parser.error(
            f"--run-id must be {expected_run_id!r} "
            f"for the supplied method/seed/round"
        )

    if args.method == "SHARED" and (
        args.seed != 0 or args.round != 0
    ):
        parser.error("SHARED baseline must use seed=0 and round=0")

    if args.method != "SHARED" and args.round == 0:
        parser.error("S0 is a shared baseline, not a method-specific run")

    return args



def publish_staged_directory(stage: Path, out_dir: Path) -> None:
    """Publish without overwrite; retry transient Windows permission failures."""
    retry_delays = (0.25, 0.5, 1.0, 2.0, 4.0)

    for attempt in range(len(retry_delays) + 1):
        if out_dir.exists():
            raise FileExistsError(
                "OUTPUT_COLLISION: destination already exists; "
                "refusing to overwrite: " + str(out_dir)
            )

        try:
            os.rename(stage, out_dir)
            return
        except PermissionError as exc:
            if out_dir.exists():
                raise FileExistsError(
                    "OUTPUT_COLLISION: destination appeared during "
                    "publication; refusing to overwrite: " + str(out_dir)
                ) from exc

            if attempt >= len(retry_delays):
                raise RuntimeError(
                    "PUBLICATION_FAILED_AFTER_RETRIES: "
                    f"{type(exc).__name__}: {exc}. "
                    f"Recovery staging directory preserved at: {stage}"
                ) from exc

            delay = retry_delays[attempt]
            print(
                "PUBLICATION_RETRY="
                f"{attempt + 1}/{len(retry_delays)} "
                f"DELAY_SECONDS={delay}"
            )
            time.sleep(delay)


def main() -> None:
    args = parse_args()

    input_hashes = build_input_hashes(args)
    out_dir = Path(args.out_root) / args.run_id

    if out_dir.exists():
        verify_existing_sidecar(out_dir, input_hashes)
        return

    out_dir.parent.mkdir(parents=True, exist_ok=True)

    stage = Path(tempfile.mkdtemp(
        prefix=f".{args.run_id}.staging.",
        dir=out_dir.parent,
    ))

    try:
        build_sidecar(args, stage, input_hashes)
        publish_staged_directory(stage, out_dir)
    except Exception as exc:
        if stage.exists():
            print(f"RECOVERY_STAGING_PRESERVED={stage}")
            print(f"RECOVERY_ERROR={type(exc).__name__}: {exc}")
        raise

    print(f"CAPABILITY_EVAL: PASS {out_dir}")


if __name__ == "__main__":
    main()

            stage.mkdir()
            target.mkdir()
            marker = stage / "marker"

        with tempfile.TemporaryDirectory() as temp:
            root = Path(temp)
            stage, target = root / ".stage", root / "final"
            stage.mkdir()
            real_rename = evaluator.os.rename
            calls = {"n": 0}

            def flaky(src, dst):
                calls["n"] += 1
                if calls["n"] == 1:
                    raise PermissionError(5, "transient")
                return real_rename(src, dst)

            with patch.object(evaluator.os, "rename", side_effect=flaky):
                with patch.object(evaluator.time, "sleep", return_value=None):
                    evaluator.publish_staged_directory(stage, target)
            self.assertEqual(calls["n"], 2)
            self.assertTrue(target.is_dir())

    def test_persistent_permission_error_keeps_stage(self):

            root = Path(temp)
            stage, target = root / ".stage", root / "final"
            stage.mkdir()

            with patch.object(
                evaluator.os, "rename",
                side_effect=PermissionError(5, "persistent"),
            ):
                with patch.object(evaluator.time, "sleep", return_value=None):
                    with self.assertRaisesRegex(
                        RuntimeError, "PUBLICATION_FAILED_AFTER_RETRIES"
