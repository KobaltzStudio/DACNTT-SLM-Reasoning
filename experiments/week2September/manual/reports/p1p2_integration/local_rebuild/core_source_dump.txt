#!/usr/bin/env python3
"""
Run one method and one training seed through the Task 7 schedule (analysis_plan.md section 3).

  S_0 (shared by all runs): eval D_dev, eval D_test, NLL of the pool
  t = 0..M-1: select B_t -> train S_t + B_t -> S_{t+1} -> eval D_dev, eval D_test -> NLL when needed

NLL per method: B4, B5 and the ablation at S_1 and S_2 (selection and/or T7-6, T7-7); B2 too when
--b2-nll is given (sensitivity analysis of T7-7); B0 and B3 never beyond S_0. S_3 is never scored.

Each step is one subprocess, so GPU memory is released between steps. A finished step leaves a marker in
<run-dir>/.done/ with its command and duration; re-running the same command skips finished steps and redoes
the interrupted one (eval_student.py also resumes inside a step). Durations are appended to
<runs-dir>/timing.jsonl, which is what the cost estimate of the main run comes from.

B4 and B5 select the same B_0 at S_0, so with the same seed S_1 is shared (task7_decisions.md): when the
other method's round 0 is finished and its trajectory list is identical, its adapter, evaluations and NLL
are copied, and the marker records the source.

Layout:
  <runs-dir>/S0/{eval_dev,eval_test,nll}.jsonl
  <runs-dir>/<method>_seed<seed>/round_NNN/{train_set.json,selection_log.jsonl}
  <runs-dir>/<method>_seed<seed>/S<t>/{adapter/,eval_dev.jsonl,eval_test.jsonl,nll.jsonl}
  <runs-dir>/<method>_seed<seed>/checkpoint_log.jsonl

--pilot is for testing the pipeline only: --pilot-problems builds a small pool from the first problems of the
official pool, with a smaller budget and evaluation limit. Keep --runs-dir apart from the main runs.

Usage (from experiments/week2September/manual/):
  python scripts/run_experiment.py --method B5 --seed 13 --runs-dir runs/main
  python scripts/run_experiment.py --method B5 --seed 13 --runs-dir runs/main --dry-run
"""
import argparse
import json
import os
import shlex
import shutil
import subprocess
import sys
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
from selection import METHODS as SELECTION_METHODS  # noqa: E402
from select_round import DEFAULT_TOKENS, round_dir  # noqa: E402
from student_common import CONFIG_PATH, POOL_VERIFIED, load_config, read_jsonl, sha256_file  # noqa: E402

SCRIPTS = Path(__file__).resolve().parent
METHODS = ("B0",) + SELECTION_METHODS
SHARED_ROUND0 = {"B4": "B5", "B5": "B4"}


class Runner:
    """Runs steps once, keeps markers and timing."""

    def __init__(self, runs_dir, dry_run=False):
        self.runs_dir = Path(runs_dir)
        self.dry_run = dry_run

    def marker(self, scope_dir, step):
        return Path(scope_dir) / ".done" / f"{step}.json"

    def is_done(self, scope_dir, step):
        return self.marker(scope_dir, step).exists()

    def mark(self, scope_dir, step, info):
        path = self.marker(scope_dir, step)
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(info, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")

    def run(self, scope_dir, step, cmd):
        printable = " ".join(shlex.quote(str(c)) for c in cmd)
        if self.is_done(scope_dir, step):
            print(f"[skip] {scope_dir}/{step}")
            return
        print(f"[run] {scope_dir}/{step}\n      {printable}", flush=True)
        if self.dry_run:
            return
        t0 = time.time()
        subprocess.run([str(c) for c in cmd], check=True)
        elapsed = round(time.time() - t0, 1)
        self.mark(scope_dir, step, {"command": printable, "elapsed_seconds": elapsed})
        self.runs_dir.mkdir(parents=True, exist_ok=True)
        with (self.runs_dir / "timing.jsonl").open("a", encoding="utf-8") as f:
            f.write(json.dumps({"scope": str(scope_dir), "step": step, "elapsed_seconds": elapsed}) + "\n")


def needs_nll_at(method, t, b2_nll):
    """NLL at S_{t}, t >= 1 (S_0 is always scored in the shared directory)."""
    if t not in (1, 2):
        return False
    return method in ("B4", "B5", "ablation_shuffled_scores") or (method == "B2" and b2_nll)


def build_pilot_pool(args, runs_dir, model_args):
    """Pool of the first problems of the official pool, and its token table."""
    pool_path = runs_dir / "pilot_pool.jsonl"
    tokens_path = runs_dir / "pilot_tokens.jsonl"
    if args.dry_run or tokens_path.with_suffix(".summary.json").exists():
        return pool_path, tokens_path
    runs_dir.mkdir(parents=True, exist_ok=True)
    keep, lines = set(), []
    with POOL_VERIFIED.open(encoding="utf-8") as f:
        for line in f:
            pid = json.loads(line)["problem_id"]
            if pid not in keep and len(keep) == args.pilot_problems:
                break
            keep.add(pid)
            lines.append(line)
    pool_path.write_text("".join(lines), encoding="utf-8")
    subprocess.run([sys.executable, SCRIPTS / "build_pool_tokens.py", "--pool", pool_path, "--out", tokens_path,
                    *model_args], check=True)
    return pool_path, tokens_path


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--method", required=True, choices=METHODS)
    ap.add_argument("--seed", required=True, type=int)
    ap.add_argument("--runs-dir", required=True, type=Path)
    ap.add_argument("--backend", default="vllm", choices=["hf", "vllm"], help="evaluation backend")
    ap.add_argument("--eval-batch", type=int, default=None)
    ap.add_argument("--nll-batch", type=int, default=16)
    ap.add_argument("--micro-batch", type=int, default=8)
    ap.add_argument("--b2-nll", action="store_true", help="score S_1, S_2 for B2 too (T7-7 sensitivity)")
    ap.add_argument("--dry-run", action="store_true", help="print the steps without running them")
    ap.add_argument("--pilot", action="store_true")
    ap.add_argument("--pilot-problems", type=int, default=60)
    ap.add_argument("--pilot-budget", type=int, default=2000)
    ap.add_argument("--pilot-eval-limit", type=int, default=20)
    ap.add_argument("--model", default=None, help="pilot only")
    ap.add_argument("--revision", default=None, help="pilot only")
    ap.add_argument("--dtype", default="auto", choices=["auto", "bf16", "fp32", "fp16"])
    args = ap.parse_args()

    cfg = load_config()
    if not args.pilot:
        if args.model:
            raise SystemExit("--model needs --pilot")
        if args.seed not in cfg["seeds"]["training_seeds"] + cfg["seeds"]["optional_extra_seeds"]:
            raise SystemExit(f"seed {args.seed} is not in the config")

    runs_dir = args.runs_dir.resolve()
    model_args = (["--model", args.model] + (["--revision", args.revision] if args.revision else [])) if args.model else []
    gpu_args = model_args + ["--dtype", args.dtype]
    eval_args = ["--backend", args.backend] + gpu_args
    if args.eval_batch:
        eval_args += ["--batch-size", args.eval_batch]
    select_args = []
    pool, tokens = POOL_VERIFIED, DEFAULT_TOKENS
    if args.pilot:
        pool, tokens = build_pilot_pool(args, runs_dir, model_args)
        eval_args += ["--limit", args.pilot_eval_limit]
        select_args = ["--pilot", "--budget", args.pilot_budget]
    elif not args.dry_run and not tokens.exists():
        raise SystemExit(f"{tokens} is missing: run scripts/build_pool_tokens.py first")

    runner = Runner(runs_dir, args.dry_run)
    py = sys.executable
    tag_run = f"{args.method}_seed{args.seed}"
    run_dir = runs_dir / tag_run
    s0 = runs_dir / "S0"
    n_pool = 0 if args.dry_run else len(read_jsonl(tokens))
    M = cfg["schedule"]["rounds_M"]
    ckpt = cfg["schedule"]["checkpoint_name"]
    config_hash = sha256_file(CONFIG_PATH)

    def evaluate(scope, ckpt_dir, tag, adapter):
        for split in ("dev", "test"):
            cmd = [py, SCRIPTS / "eval_student.py", "--tag", tag, "--split", split,
                   "--out", ckpt_dir / f"eval_{split}.jsonl", *eval_args]
            runner.run(scope, f"eval_{split}_{tag}", cmd + (["--adapter", adapter] if adapter else []))

    def score(scope, ckpt_dir, tag, adapter):
        cmd = [py, SCRIPTS / "compute_nll.py", "--tag", tag, "--out", ckpt_dir / "nll.jsonl", "--pool", pool,
               "--batch-size", args.nll_batch, *gpu_args]
        runner.run(scope, f"nll_{tag}", cmd + (["--adapter", adapter] if adapter else []))

    evaluate(s0, s0, "S0", None)
    score(s0, s0, "S0", None)

    if args.method == "B0":
        print("B0 is S_0 itself: nothing to train.")
        return

    def write_checkpoint_record(t, selected, parent, cumulative_tokens, cumulative_steps):
        row = {"record_type": "checkpoint", "checkpoint": ckpt.format(t=t), "parent_checkpoint": parent,
               "round": t, "method": args.method, "training_seed": args.seed, "data_order_seed": args.seed,
               "cumulative_training_tokens": cumulative_tokens, "cumulative_optimization_steps": cumulative_steps,
               "selected_trajectory_ids": selected,
               "beta_est_forward_passes": n_pool if t == 0 or needs_nll_at(args.method, t, args.b2_nll) else 0,
               "config_hash": config_hash}
        rows = [r for r in read_jsonl(log_path) if r["checkpoint"] != row["checkpoint"]] if log_path.exists() else []
        log_path.write_text("".join(json.dumps(r, ensure_ascii=False) + "\n" for r in rows + [row]), encoding="utf-8")

    log_path = run_dir / "checkpoint_log.jsonl"
    if not args.dry_run:
        run_dir.mkdir(parents=True, exist_ok=True)
        write_checkpoint_record(0, [], None, 0, 0)

    def try_reuse_round0():
        """Copy S_1 from the other of B4/B5 when its B_0 is identical. True when copied."""
        other = SHARED_ROUND0.get(args.method)
        if other is None or runner.is_done(run_dir, "train_0"):
            return False
        other_dir = runs_dir / f"{other}_seed{args.seed}"
        other_tag = f"{other}_seed{args.seed}_S1"
        steps = ["train_0", f"eval_dev_{other_tag}", f"eval_test_{other_tag}", f"nll_{other_tag}"]
        if not all(runner.is_done(other_dir, s) for s in steps):
            return False
        own = json.loads((round_dir(run_dir, 0) / "train_set.json").read_text(encoding="utf-8"))
        theirs = json.loads((round_dir(other_dir, 0) / "train_set.json").read_text(encoding="utf-8"))
        if own["trajectory_ids"] != theirs["trajectory_ids"]:
            raise RuntimeError(f"{args.method} and {other} differ at round 0: the shared S_1 assumption fails")
        shutil.copytree(other_dir / "S1", run_dir / "S1", dirs_exist_ok=True)
        tag = f"{tag_run}_S1"
        info = {"reused_from": str(other_dir / "S1"), "reason": "identical B_0 and seed (task7_decisions.md)"}
        for step in ("train_0", f"eval_dev_{tag}", f"eval_test_{tag}", f"nll_{tag}"):
            runner.mark(run_dir, step, info)
        print(f"[reuse] S_1 of {tag_run} copied from {other_dir.name}")
        return True

    cumulative_tokens, cumulative_steps = 0, 0
    for t in range(M):
        nll = s0 / "nll.jsonl" if t == 0 or args.method == "B4" else run_dir / f"S{t}" / "nll.jsonl"
        cmd = [py, SCRIPTS / "select_round.py", "--method", args.method, "--seed", args.seed, "--round", t,
               "--run-dir", run_dir, "--pool", pool, "--tokens", tokens, *select_args]
        if args.method in ("B4", "B5", "ablation_shuffled_scores"):
            cmd += ["--nll", nll]
        runner.run(run_dir, f"select_{t}", cmd)
        runner.run(run_dir, f"validate_log_{t}",
                   [sys.executable, "-c",
                    "import subprocess,sys;r=subprocess.run([sys.executable,sys.argv[1],sys.argv[2]],"
                    "capture_output=True,text=True);print(r.stdout.splitlines()[-1]);sys.exit(r.returncode)",
                    SCRIPTS / "validate_selection_log.py", round_dir(run_dir, t) / "selection_log.jsonl"])

        ckpt_dir = run_dir / f"S{t + 1}"
        adapter = ckpt_dir / "adapter"
        tag = f"{tag_run}_S{t + 1}"
        if not args.dry_run and t == 0:
            try_reuse_round0()
        train_cmd = [py, SCRIPTS / "train_round.py", "--seed", args.seed,
                     "--train-set", round_dir(run_dir, t) / "train_set.json", "--adapter-dir", adapter,
                     "--micro-batch", args.micro_batch, *gpu_args]
        if t > 0:
            train_cmd += ["--init-adapter", run_dir / f"S{t}" / "adapter"]
        runner.run(run_dir, f"train_{t}", train_cmd)
        evaluate(run_dir, ckpt_dir, tag, adapter)
        if needs_nll_at(args.method, t + 1, args.b2_nll):
            score(run_dir, ckpt_dir, tag, adapter)

        if not args.dry_run:
            train_log = json.loads((adapter / "train_log.json").read_text(encoding="utf-8"))
            cumulative_tokens += train_log["cumulative_training_tokens"]
            cumulative_steps += train_log["steps"]
            selected = json.loads((round_dir(run_dir, t) / "train_set.json").read_text(encoding="utf-8"))["trajectory_ids"]
            write_checkpoint_record(t + 1, selected, ckpt.format(t=t), cumulative_tokens, cumulative_steps)

    print(f"{tag_run} finished -> {run_dir}")


if __name__ == "__main__":
    os.chdir(SCRIPTS.parent)
    main()

#!/usr/bin/env python3
"""
Select B_t for one method, seed and round; writes the train set and the selection log (selection.py has the rules).

Reads:
  --tokens          token table of the pool (build_pool_tokens.py)
  --nll             compute_nll.py output: at S_t for B5 and the ablation, at S_0 for B4 in every round
  <run-dir>/round_NNN/train_set.json of earlier rounds (trajectories already trained in this run, T7-3)

Writes (atomically):
  <run-dir>/round_NNN/train_set.json       input of train_round.py
  <run-dir>/round_NNN/selection_log.jsonl  one record per eligible trajectory (selection_log_schema.json)

--pilot allows a smaller budget and seeds outside the config; the output is marked and is not a result.

Usage (from experiments/week2September/manual/):
  python scripts/select_round.py --method B5 --seed 13 --round 1 --run-dir runs/main/B5_seed13 \
      --nll runs/main/B5_seed13/S1/nll.jsonl
"""
import argparse
import json
import os
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
from selection import METHODS, NEEDS_NLL, select  # noqa: E402
from student_common import CONFIG_PATH, MANUAL_DIR, POOL_VERIFIED, load_config, read_jsonl, sha256_file  # noqa: E402

DEFAULT_TOKENS = POOL_VERIFIED.with_name("pool_v1_tokens.jsonl")


def round_dir(run_dir, t):
    return Path(run_dir) / f"round_{t:03d}"


def rel(path):
    resolved = Path(path).resolve()
    return str(resolved.relative_to(MANUAL_DIR)) if resolved.is_relative_to(MANUAL_DIR) else str(resolved)


def write_atomic(path, text):
    tmp = path.with_name(path.name + ".tmp")
    tmp.write_text(text, encoding="utf-8")
    os.replace(tmp, path)


def load_tokens(tokens_path, pool_path):
    summary = json.loads(Path(tokens_path).with_suffix(".summary.json").read_text(encoding="utf-8"))
    if summary["pool_sha256"] != sha256_file(pool_path):
        raise ValueError(f"{tokens_path} was built from another pool")
    rows = read_jsonl(tokens_path)
    return {r["trajectory_id"]: r["problem_id"] for r in rows}, {r["trajectory_id"]: r["response_tokens"] for r in rows}


def load_ell(nll_path, tokens):
    """ell by trajectory; the NLL file must cover the pool with the token counts of the budget."""
    rows = read_jsonl(nll_path)
    ell = {r["trajectory_id"]: r["ell"] for r in rows}
    if len(ell) != len(rows) or set(ell) != set(tokens):
        raise ValueError(f"{nll_path} does not cover exactly the eligible pool")
    wrong = [r["trajectory_id"] for r in rows if r["n_tokens"] != tokens[r["trajectory_id"]]]
    if wrong:
        raise ValueError(f"{len(wrong)} token counts in {nll_path} differ from the token table: {wrong[:3]}")
    return ell


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--method", required=True, choices=METHODS)
    ap.add_argument("--seed", required=True, type=int)
    ap.add_argument("--round", required=True, type=int)
    ap.add_argument("--run-dir", required=True, type=Path)
    ap.add_argument("--pool", type=Path, default=POOL_VERIFIED)
    ap.add_argument("--tokens", type=Path, default=DEFAULT_TOKENS)
    ap.add_argument("--nll", type=Path, default=None)
    ap.add_argument("--budget", type=int, default=None, help="pilot only (default: config beta_round_tokens)")
    ap.add_argument("--pilot", action="store_true")
    args = ap.parse_args()

    cfg = load_config()
    if not args.pilot:
        if args.seed not in cfg["seeds"]["training_seeds"] + cfg["seeds"]["optional_extra_seeds"]:
            raise SystemExit(f"seed {args.seed} is not in the config")
        if args.budget is not None:
            raise SystemExit("--budget needs --pilot")
    if not 0 <= args.round < cfg["schedule"]["rounds_M"]:
        raise SystemExit(f"round must be in 0..{cfg['schedule']['rounds_M'] - 1}")
    if args.method in NEEDS_NLL and args.nll is None:
        raise SystemExit(f"{args.method} needs --nll")
    budget = args.budget or cfg["budget"]["beta_round_tokens"]
    n_cap = cfg["selection"]["n_cap"]

    problems, tokens = load_tokens(args.tokens, args.pool)
    trained = []
    for t in range(args.round):
        trained += json.loads((round_dir(args.run_dir, t) / "train_set.json").read_text(encoding="utf-8"))["trajectory_ids"]
    ell = load_ell(args.nll, tokens) if args.method in NEEDS_NLL else None

    res = select(args.method, args.round, args.seed, problems, tokens, budget, n_cap, ell, sorted(set(trained)))

    config_hash = sha256_file(CONFIG_PATH)
    source_t = 0 if args.method == "B4" else args.round  # B4 is ranked at S_0 in every round
    source = cfg["schedule"]["checkpoint_name"].format(t=source_t)
    out_dir = round_dir(args.run_dir, args.round)
    out_dir.mkdir(parents=True, exist_ok=True)
    write_atomic(out_dir / "selection_log.jsonl", "".join(
        json.dumps({"record_type": "selection", "method": args.method, "round": args.round,
                    "source_checkpoint": source, "training_seed": args.seed, **row, "config_hash": config_hash},
                   ensure_ascii=False) + "\n"
        for row in res["rows"]))
    write_atomic(out_dir / "train_set.json", json.dumps({
        "pilot": args.pilot,
        "method": args.method, "round": args.round, "training_seed": args.seed, "source_checkpoint": source,
        "pool": rel(args.pool), "pool_sha256": sha256_file(args.pool),
        "config_sha256": config_hash,
        "nll_file": rel(args.nll) if ell else None,
        "nll_sha256": sha256_file(args.nll) if ell else None,
        "ell_star": res["ell_star"], "sigma": res["sigma"], "n_reference": res["n_reference"],
        "n_eligible": len(problems), "n_trained_before": len(set(trained)),
        "beta_round_tokens": budget, "n_cap": n_cap,
        "tokens_used": res["tokens_used"], "n_selected": len(res["selected"]),
        "trajectory_ids": res["selected"],
    }, ensure_ascii=False, indent=2) + "\n")
    print(f"[{args.method} seed {args.seed} round {args.round}] {len(res['selected'])} trajectories, "
          f"{res['tokens_used']}/{budget} tokens -> {out_dir}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Train one round of the Student with LoRA: S_t + B_t -> S_{t+1}.

Hyperparameters come from config/experiment_config.json "training" (LoRA r/alpha/dropout/targets,
AdamW, constant lr, no warmup, max_grad_norm, effective batch, 1 epoch, max_seq_len,
completion-only loss). Effective batch = --micro-batch x gradient accumulation.

Loss: token-level mean over all supervised (response) tokens of the effective batch;
prompt tokens are masked. Seeds: torch seed (LoRA init, dropout) = data order seed = --seed.

Pool: the trajectories are looked up in the pool named by the train set ("pool", checked against
"pool_sha256"), so the Task 5 smoke-test train sets still resolve to the preliminary pool; train sets
without that field use --pool (default: official pool v1).

Outputs:
  <adapter-dir>/                 LoRA adapter (safetensors) + tokenizer files
  <adapter-dir>/train_log.json   per-step loss, tokens, time; totals; config used

Usage (from experiments/week2September/manual/):
  python scripts/train_round.py --seed 13 \
      --train-set reports/task5/smoke_test/train_set_seed13.json \
      --adapter-dir models/smoke/student_ckpt_001_smoke_seed13
"""

def collate(examples, pad_id):
    width = max(len(e["input_ids"]) for e in examples)
    ids, labels, mask = [], [], []
    for e in examples:
        pad = width - len(e["input_ids"])
        ids.append(e["input_ids"] + [pad_id] * pad)          # right padding for training
        labels.append(e["labels"] + [IGNORE_INDEX] * pad)
        mask.append([1] * len(e["input_ids"]) + [0] * pad)
    return torch.tensor(ids), torch.tensor(labels), torch.tensor(mask)


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--seed", type=int, required=True)
    ap.add_argument("--train-set", required=True, type=Path, help="JSON with trajectory_ids (select_smoke_train_set.py)")
    ap.add_argument("--adapter-dir", required=True, type=Path)
    ap.add_argument("--init-adapter", default=None, help="continue from an existing adapter (round t > 0)")
    ap.add_argument("--micro-batch", type=int, default=4)
    ap.add_argument("--epochs", type=int, default=None, help="default: config training.epochs_per_round")
    ap.add_argument("--lr", type=float, default=None, help="default: config training.learning_rate")
    ap.add_argument("--no-gradient-checkpointing", action="store_true")
    ap.add_argument("--pool", type=Path, default=None,
                    help="pool for train sets without a 'pool' field (default: official pool v1)")
    add_model_args(ap)
    args = ap.parse_args()

    check_prompt_template()
    cfg = load_config()
    tc = cfg["training"]
    if tc["loss"].split()[0] != "completion-only" or tc["lr_scheduler"] != "constant" or tc["warmup_steps"] != 0:
        raise ValueError("config training block differs from what this script implements")
    eff_batch = tc["batch_size"]
    if eff_batch % args.micro_batch:
        raise ValueError(f"--micro-batch must divide the effective batch {eff_batch}")
    accum = eff_batch // args.micro_batch
    epochs = args.epochs or tc["epochs_per_round"]
    lr = args.lr or tc["learning_rate"]

def build_prompt(tokenizer, question):
    """Chat-formatted prompt v1, ending with the assistant generation header."""
    messages = [{"role": "user", "content": PROMPT_TEMPLATE.format(question=question)}]
    return tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)


def response_text(raw_generation):
    """Training/scoring target: the Teacher's raw output (REASONING: ... ANSWER: ...)."""
    return raw_generation.strip()


def encode_example(tokenizer, question, raw_generation, add_eos=True):
    """
    Token ids for prompt + response, labels masked (-100) on the prompt.

    Prompt and response are tokenized separately, which is exactly what the model sees at
    generation time. The end-of-turn token is appended to the supervised part when add_eos,
    so the Student learns to stop. n_traj counts only the trajectory tokens (no end-of-turn),
    matching L_bar in the config.
    """
    prompt_ids = tokenizer(build_prompt(tokenizer, question), add_special_tokens=False)["input_ids"]
    traj_ids = tokenizer(response_text(raw_generation), add_special_tokens=False)["input_ids"]
    resp_ids = traj_ids + ([end_of_turn_id(tokenizer)] if add_eos else [])
    input_ids = prompt_ids + resp_ids
    labels = [IGNORE_INDEX] * len(prompt_ids) + resp_ids
    return {
        "input_ids": input_ids,
        "labels": labels,
        "n_prompt": len(prompt_ids),
        "n_traj": len(traj_ids),
        "n_supervised": len(resp_ids),
    }


def end_of_turn_id(tokenizer):
    tid = tokenizer.convert_tokens_to_ids("<|im_end|>")
    if tid is None or tid == tokenizer.unk_token_id:
        raise ValueError("tokenizer has no <|im_end|> token")
    return tid


def pick_dtype(torch, requested="auto"):
    """bf16 when the GPU supports it (config), otherwise fp32 (fp16 is avoided: Qwen2.5 can overflow)."""
    if requested != "auto":
        return {"bf16": torch.bfloat16, "fp32": torch.float32, "fp16": torch.float16}[requested]
    if torch.cuda.is_available() and torch.cuda.is_bf16_supported():
        return torch.bfloat16
    return torch.float32

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def hash_path(path: Path) -> str:
    path = path.resolve()

    if path.is_file():
        return sha256_file(path)

    if not path.is_dir():
        raise FileNotFoundError(f"Input does not exist: {path}")

    files = sorted(p for p in path.rglob("*") if p.is_file())
    if not files:
        raise ValueError(f"Checkpoint directory is empty: {path}")

    digest = hashlib.sha256()
    for file_path in files:
        relative = file_path.relative_to(path).as_posix()
        digest.update(relative.encode("utf-8"))
        digest.update(b"\0")
        digest.update(sha256_file(file_path).encode("ascii"))
        digest.update(b"\n")

    return digest.hexdigest()

    rows: list[dict[str, Any]] = []

            try:
                row = json.loads(line)

                ) from exc

            if not isinstance(row, dict):

            rows.append(row)

    return rows

def wilson_interval(
    correct: int,
    support: int,
    z: float = 1.959963984540054,
) -> tuple[float | None, float | None]:
    if support <= 0:
        return None, None

    p = correct / support
    z2 = z * z
    denominator = 1.0 + z2 / support
    center = (p + z2 / (2.0 * support)) / denominator
    margin = (
        z
        * (
            p * (1.0 - p) / support
            + z2 / (4.0 * support * support)
        ) ** 0.5
        / denominator
    )

    return max(0.0, center - margin), min(1.0, center + margin)

def load_approved_gold(
    gold_path: Path,
    approval_path: Path,
    dev_ids: set[str],
) -> dict[str, dict[str, Any]]:
    gold_rows = read_jsonl(gold_path)

    if len(gold_rows) != 85:
        raise ValueError(
            f"Expected exactly 85 Gold rows; found {len(gold_rows)}"
        )

    gold_by_id: dict[str, dict[str, Any]] = {}

    for row in gold_rows:
        problem_id = str(row.get("problem_id", "")).strip()
        if not problem_id:
            raise ValueError("Gold row missing problem_id")
        if problem_id in gold_by_id:
            raise ValueError(f"Duplicate Gold problem_id: {problem_id}")

        labels = row.get("capabilities_manual")
        if not isinstance(labels, list) or not labels:
            raise ValueError(
                f"Gold capability labels missing/invalid: {problem_id}"
            )

        unknown = set(labels) - set(CAPABILITIES)
        if unknown:
            raise ValueError(
                f"Unknown capabilities for {problem_id}: {sorted(unknown)}"
            )

        gold_by_id[problem_id] = row

    gold_ids = set(gold_by_id)

    if len(gold_ids) != 85:
        raise ValueError("Gold IDs are not 85 unique IDs")

    if not gold_ids.issubset(dev_ids):
        outside = sorted(gold_ids - dev_ids)
        raise ValueError(
            f"Gold IDs outside D_dev: {outside[:10]}"
        )

    approval_by_id: dict[str, dict[str, str]] = {}

    with approval_path.open(
        "r", encoding="utf-8-sig", newline=""
    ) as stream:
        reader = csv.DictReader(stream)

        required_columns = {
            "problem_id",
            "review_status",
            "human_approval",
        }
        actual_columns = set(reader.fieldnames or [])
        missing_columns = required_columns - actual_columns

        if missing_columns:
            raise ValueError(
                f"Approval CSV missing columns: {sorted(missing_columns)}"
            )

        for row in reader:
            problem_id = str(row.get("problem_id", "")).strip()
            if not problem_id:
                raise ValueError("Approval CSV contains empty problem_id")
            if problem_id in approval_by_id:
                raise ValueError(
                    f"Duplicate approval problem_id: {problem_id}"
                )
            approval_by_id[problem_id] = row

    approval_ids = set(approval_by_id)

    if approval_ids != gold_ids:
        missing = sorted(gold_ids - approval_ids)
        extra = sorted(approval_ids - gold_ids)
        raise ValueError(
            "Approval/Gold ID set mismatch: "
            f"missing_approval={missing[:10]}, "
            f"extra_approval={extra[:10]}"
        )

    for problem_id, row in approval_by_id.items():
        review_status = str(row.get("review_status", "")).strip()
        human_approval = str(row.get("human_approval", "")).strip()

        approved_review_statuses = {
            "APPROVED_NO_LABEL_CHANGE",
            "APPROVED_WITH_SOURCE_CLARIFICATION",
        }

        if review_status not in approved_review_statuses:
            raise ValueError(
                f"Gold review status is not approved: "
                f"{problem_id}={review_status!r}"
            )

        if not human_approval.startswith("APPROVED_BY_USER_"):
            raise ValueError(
                f"Gold lacks explicit user approval: "
                f"{problem_id}={human_approval!r}"
            )

    return gold_by_id

        "train_set": args.train_set,
        "selection_log": args.selection_log,
        "train_log": args.train_log,
        "checkpoint_log": args.checkpoint_log,
    }

    for name, value in optional_paths.items():
        if value:
            input_paths[name] = Path(value)

    hashes: dict[str, str] = {}
    for name, path in input_paths.items():
        if not path.exists():
            raise FileNotFoundError(f"Missing input {name}: {path}")
        hashes[name] = hash_path(path)

    return hashes


def validate_manifest_count(path: Path) -> None:
    manifest = read_json(path)

    if manifest.get("gold_version") != "vFinal":
        raise ValueError(
            "Expected Gold manifest gold_version='vFinal'"
        )

    if manifest.get("supplement_count") != 85:
        raise ValueError(
            "Gold manifest supplement_count must equal 85"
        )


def build_sidecar(
    args: argparse.Namespace,
    stage: Path,
    input_hashes: dict[str, str],
) -> None:
    dev_ids = read_id_file(Path(args.dev_ids))

    if len(dev_ids) != 500:
        raise ValueError(
            f"Expected 500 unique D_dev IDs; found {len(dev_ids)}"
        )

    predictions = read_jsonl(Path(args.dev_predictions))
    prediction_ids = [
        str(row.get("problem_id", "")).strip()
        for row in predictions
    ]

    if any(not value for value in prediction_ids):
        raise ValueError("Prediction contains empty problem_id")

    if len(prediction_ids) != len(set(prediction_ids)):
        raise ValueError("Duplicate problem_id in D_dev predictions")

    if set(prediction_ids) != dev_ids or len(predictions) != 500:
        missing = sorted(dev_ids - set(prediction_ids))
        extra = sorted(set(prediction_ids) - dev_ids)
        raise ValueError(
            "Prediction coverage mismatch: "
            f"rows={len(predictions)}, "
            f"missing={missing[:10]}, extra={extra[:10]}"
        )

    prediction_by_id = {
        str(row["problem_id"]): row for row in predictions
    }

    # Validate status for every prediction, not only the 85 Gold rows.
    for row in predictions:
        strict_correct(row)

    validate_manifest_count(Path(args.gold_manifest))

    gold_by_id = load_approved_gold(
        Path(args.gold_labels),
        Path(args.approval_csv),
        dev_ids,
    )

    metrics: list[dict[str, Any]] = []

    for capability in CAPABILITIES:
        capability_ids = sorted(
            problem_id
            for problem_id, gold in gold_by_id.items()
            if capability in gold["capabilities_manual"]
        )

        support = len(capability_ids)
        correct = sum(
            strict_correct(prediction_by_id[problem_id])
            for problem_id in capability_ids
        )

        ci_low, ci_high = wilson_interval(correct, support)

        metrics.append({
            "run_id": args.run_id,
            "method": args.method,
            "seed": args.seed,
            "checkpoint": f"S{args.round}",
            "checkpoint_id": Path(args.checkpoint).name,
            "capability": capability,
            "n_support": support,
            "n_correct": correct,
            "n_incorrect": support - correct,
            "mastery": correct / support if support else None,
            "deficit": 1.0 - correct / support if support else None,
            "ci95_low": ci_low,
            "ci95_high": ci_high,
            "ci_method": "Wilson_95_percent",
            "strict_metric": "status_equals_valid",
            "gold_policy": "approved_85_primary",
        })

    observed_support = {
        row["capability"]: row["n_support"] for row in metrics
    }

    if observed_support != EXPECTED_SUPPORT:
        raise ValueError(
            "Gold capability support differs from the frozen expected "
            f"vector. observed={observed_support}; "
            f"expected={EXPECTED_SUPPORT}"
        )

    with (stage / "capability_metrics.csv").open(
        "w", encoding="utf-8-sig", newline=""
    ) as stream:
        writer = csv.DictWriter(
            stream,
            fieldnames=list(metrics[0].keys()),
        )
        writer.writeheader()
        writer.writerows(metrics)

    with (stage / "capability_metrics.jsonl").open(
        "w", encoding="utf-8"
    ) as stream:
        for row in metrics:
            stream.write(
                json.dumps(row, ensure_ascii=False) + "\n"
            )

    coverage = {
        "status": "PASS",
        "split": "D_dev",
        "expected_dev_ids": 500,
        "prediction_records": len(predictions),
        "prediction_unique_ids": len(set(prediction_ids)),
        "gold_records": len(gold_by_id),
        "gold_unique_ids": len(gold_by_id),
        "gold_matched_prediction_ids": len(
            set(gold_by_id) & set(prediction_ids)
        ),
        "missing_prediction_ids": [],
        "gold_policy": "approved_85_primary",
        "strict_correctness_policy": "status == valid",
        "lenient_correct_excluded": True,
        "capability_support": observed_support,
    }

    (stage / "prediction_coverage_report.json").write_text(
        json.dumps(coverage, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    lineage = {
        "run_id": args.run_id,
        "method": args.method,
        "seed": args.seed,
        "round": args.round,
        "checkpoint_id": Path(args.checkpoint).name,
        "checkpoint_path": str(Path(args.checkpoint)),
        "checkpoint_sha256": input_hashes["checkpoint"],
        "shared_checkpoint_group": args.shared_checkpoint_group,
        "train_set": args.train_set,
        "selection_log": args.selection_log,
        "train_log": args.train_log,
        "checkpoint_log": args.checkpoint_log,
        "config_sha256": input_hashes["config"],
        "prediction_sha256": input_hashes["dev_predictions"],
        "gold_manifest_sha256": input_hashes["gold_manifest"],
        "gold_labels_sha256": input_hashes["gold_labels"],
        "approval_csv_sha256": input_hashes["approval_csv"],
        "dev_ids_sha256": input_hashes["dev_ids"],
    }

    (stage / "checkpoint_lineage.json").write_text(
        json.dumps(lineage, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    fingerprint = hashlib.sha256(
        json.dumps(
            input_hashes,
            sort_keys=True,
            separators=(",", ":"),
        ).encode("utf-8")
    ).hexdigest()

    (stage / "input_fingerprint.json").write_text(
        json.dumps({
            "fingerprint": fingerprint,
            "inputs": input_hashes,
        }, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    manifest = {
        "schema_version": "p1p2.capability-manifest.v1",
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "run_id": args.run_id,
        "method": args.method,
        "seed": args.seed,
        "round": args.round,
        "checkpoint": f"S{args.round}",
        "gold_policy": "approved_85_primary",
        "strict_metric": "status == valid",
        "capabilities": list(CAPABILITIES),
        "n_capability_rows": len(metrics),
        "input_hashes": input_hashes,
        "outputs": {},
        "overwrite_policy": "immutable_exact_fingerprint_reuse_only",
    }

    for name in REQUIRED_OUTPUTS:
        if name == "student_capability_evaluation_manifest.json":
            continue

        output_path = stage / name
        manifest["outputs"][name] = {
            "sha256": sha256_file(output_path),
            "size_bytes": output_path.stat().st_size,
        }

    (stage / "student_capability_evaluation_manifest.json").write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )


def verify_existing_sidecar(
    out_dir: Path,
    input_hashes: dict[str, str],
) -> None:
    manifest_path = (
        out_dir / "student_capability_evaluation_manifest.json"
    )

    if not manifest_path.is_file():
        raise RuntimeError(
            "EXISTING_PARTIAL_SIDECAR: manifest missing; "
            "do not overwrite or auto-delete this directory"
        )

    manifest = read_json(manifest_path)

    if manifest.get("input_hashes") != input_hashes:
        raise RuntimeError(
            "INPUT_FINGERPRINT_MISMATCH: sidecar is immutable"
        )

    outputs = manifest.get("outputs", {})
    expected_names = set(REQUIRED_OUTPUTS) - {
        "student_capability_evaluation_manifest.json"
    }

    if set(outputs) != expected_names:
        raise RuntimeError("Existing sidecar output inventory mismatch")

    for name, metadata in outputs.items():
        path = out_dir / name

        if not path.is_file():
            raise RuntimeError(f"Existing output missing: {path}")

        if sha256_file(path) != metadata.get("sha256"):
            raise RuntimeError(f"Existing output hash mismatch: {path}")

    print(f"REUSE: PASS {out_dir}")


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="CPU-only P1/P2 capability sidecar evaluator"
    )

    parser.add_argument("--run-id", required=True)
    parser.add_argument(
        "--method",
        required=True,
        choices=["SHARED", "B2", "B3", "B4", "B5"],
    )
    parser.add_argument("--seed", required=True, type=int)
    parser.add_argument(
        "--round",
        required=True,
        type=int,
        choices=[0, 1, 2, 3],
    )

    parser.add_argument("--checkpoint", required=True)
    parser.add_argument("--dev-predictions", required=True)
    parser.add_argument("--gold-manifest", required=True)
    parser.add_argument("--gold-labels", required=True)
    parser.add_argument("--approval-csv", required=True)
    parser.add_argument("--dev-ids", required=True)
    parser.add_argument("--config", required=True)

    parser.add_argument("--train-set")
    parser.add_argument("--selection-log")
    parser.add_argument("--train-log")
    parser.add_argument("--checkpoint-log")
    parser.add_argument("--shared-checkpoint-group")
    parser.add_argument(
        "--out-root",
        default="reports/capability",
    )

    args = parser.parse_args()

    expected_run_id = (
        "SHARED_seed0_S0"
        if args.method == "SHARED"
        else f"{args.method}_seed{args.seed}_S{args.round}"
    )

    if args.run_id != expected_run_id:
        parser.error(
            f"--run-id must be {expected_run_id!r} "
            f"for the supplied method/seed/round"
        )

    if args.method == "SHARED" and (
        args.seed != 0 or args.round != 0
    ):
        parser.error("SHARED baseline must use seed=0 and round=0")

    if args.method != "SHARED" and args.round == 0:
        parser.error("S0 is a shared baseline, not a method-specific run")

    return args


def main() -> None:
    args = parse_args()

    input_hashes = build_input_hashes(args)
    out_dir = Path(args.out_root) / args.run_id

    if out_dir.exists():
        verify_existing_sidecar(out_dir, input_hashes)
        return

    out_dir.parent.mkdir(parents=True, exist_ok=True)

    stage = Path(tempfile.mkdtemp(
        prefix=f".{args.run_id}.staging.",
        dir=out_dir.parent,
    ))

    try:
        build_sidecar(args, stage, input_hashes)

        # Atomic publish. If another process created the target first,
        # do not replace it.
        if out_dir.exists():
            raise FileExistsError(
                f"Output appeared during evaluation: {out_dir}"
            )

        os.rename(stage, out_dir)

    except Exception:
        shutil.rmtree(stage, ignore_errors=True)
        raise

    print(f"CAPABILITY_EVAL: PASS {out_dir}")


if __name__ == "__main__":
    main()
