SUPPLEMENT_INPUT = SOURCE_DIR / "supplement_annotations_vnext.jsonl"

SUPPLEMENT_OUTPUT = OUTPUT_DIR / "supplement_annotations_vfinal.jsonl"

VALID_CAPABILITIES = {

    "division",

VALID_DIFFICULTIES = {"easy", "medium", "hard"}

EXPECTED_BATCH = 100

EXPECTED_TOTAL = 185


def read_jsonl(path):
    if not path.exists():
        raise FileNotFoundError(f"Missing input file: {path}")

    records = []
    with path.open("r", encoding="utf-8-sig") as f:
        for line_number, line in enumerate(f, start=1):
            if not line.strip():
                continue
            try:
                record = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSON at {path}:{line_number}"
                ) from exc

            if not isinstance(record, dict):
                raise ValueError(
                    f"Expected JSON object at {path}:{line_number}"
                )

            records.append(record)

    return records


def write_jsonl(path, records):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="\n") as f:
        for record in records:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")


def parse_capabilities(value, field_name, problem_id):
    if isinstance(value, list):
        labels = value
    else:
        raw = str(value or "").strip()

        if not raw:
            raise ValueError(
                f"Empty {field_name} for {problem_id}"
            )

        # Accept JSON array or semicolon-separated values.
        if raw.startswith("["):
            try:
                labels = json.loads(raw)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSON in {field_name} for {problem_id}"
                ) from exc
        else:
            labels = [x.strip() for x in raw.split(";") if x.strip()]

    if not isinstance(labels, list):
        raise ValueError(
            f"{field_name} must be a list for {problem_id}"
        )

    labels = [str(x).strip() for x in labels]

    if len(labels) != len(set(labels)):
        raise ValueError(
            f"Duplicate capabilities for {problem_id}: {labels}"
        )

    invalid = sorted(set(labels) - VALID_CAPABILITIES)
    if invalid:
        raise ValueError(
            f"Invalid capabilities for {problem_id}: {invalid}"
        )

    return sorted(labels)


def load_human_confirmations():
    if not HUMAN_INPUT.exists():
        raise FileNotFoundError(HUMAN_INPUT)

    with HUMAN_INPUT.open(
        "r", encoding="utf-8-sig", newline=""
    ) as f:
        rows = list(csv.DictReader(f))

    if len(rows) != 22:
        raise ValueError(
            f"Expected 22 adjudication rows, found {len(rows)}"
        )

    confirmations = {}
    confirmed_rows = []

    for row in rows:
        problem_id = row.get("problem_id", "").strip()

        if not problem_id:
            raise ValueError("An adjudication row has an empty problem_id")

        if problem_id in confirmations:
            raise ValueError(f"Duplicate adjudication ID: {problem_id}")

        if str(row.get("question_matches_raw", "")).lower() != "true":
            raise ValueError(
                f"Question/raw mismatch for {problem_id}"
            )

        # The project team explicitly confirmed the current AI decisions
        # in the conversation on 2026-11-06, without requesting revisions.
        capabilities = parse_capabilities(
            row.get("final_capabilities", ""),
            "final_capabilities",
            problem_id,
        )

        difficulty = str(
            row.get("final_difficulty", "")
        ).strip().lower()

        if difficulty not in VALID_DIFFICULTIES:
            raise ValueError(
                f"Invalid final difficulty for {problem_id}: {difficulty}"
            )

        confirmed = dict(row)
        confirmed["human_review_status"] = "CONFIRMED"
        confirmed["human_confirmed_capabilities"] = ";".join(capabilities)
        confirmed["human_confirmed_difficulty"] = difficulty
        confirmed["human_reviewer"] = "Project team (chat confirmation)"
        confirmed["human_review_notes"] = (
            "The project team explicitly confirmed the current AI "
            "adjudication in the conversation on 2026-11-06. "
            "No label or difficulty revision was requested."
        )

        confirmed_rows.append(confirmed)
        confirmations[problem_id] = {
            "capabilities": capabilities,
            "difficulty": difficulty,
        }

    fields = list(confirmed_rows[0].keys())

    HUMAN_CONFIRMED.parent.mkdir(parents=True, exist_ok=True)
    with HUMAN_CONFIRMED.open(
        "w", encoding="utf-8-sig", newline=""
    ) as f:
        writer = csv.DictWriter(
            f,
            fieldnames=fields,
            extrasaction="ignore",
        )
        writer.writeheader()
        writer.writerows(confirmed_rows)

    return confirmations


def index_records(records, label):
    result = {}

    for record in records:
        problem_id = record.get("problem_id")

        if not problem_id:
            raise ValueError(f"{label} contains a record without problem_id")

        if problem_id in result:
            raise ValueError(
                f"Duplicate problem_id in {label}: {problem_id}"
            )

        result[problem_id] = record

    return result


def apply_confirmations(records, confirmations, source_name):
    output = []

    for original in records:
        record = dict(original)
        problem_id = record["problem_id"]

        if problem_id in confirmations:
            if "capabilities_manual" not in record:
                raise ValueError(
                    f"{source_name}: missing capabilities_manual "
                    f"for adjudicated ID {problem_id}"
                )

            if "difficulty_manual" not in record:
                raise ValueError(
                    f"{source_name}: missing difficulty_manual "
                    f"for adjudicated ID {problem_id}"
                )

            decision = confirmations[problem_id]
            record["capabilities_manual"] = decision["capabilities"]
            record["difficulty_manual"] = decision["difficulty"]

        output.append(record)

    return output


def compare_records(source_records, final_records, confirmations):
    source_by_id = index_records(source_records, "Gold vNext")
    final_by_id = index_records(final_records, "Gold vFinal")

    if set(source_by_id) != set(final_by_id):
        raise ValueError("Gold vNext and Gold vFinal ID inventories differ")

    changed_capabilities = 0
    changed_difficulties = 0
    unadjudicated_changes = []
    immutable_field_changes = []

    for problem_id, source in source_by_id.items():
        final = final_by_id[problem_id]

        if source.get("capabilities_manual") != final.get("capabilities_manual"):
            changed_capabilities += 1

        if source.get("difficulty_manual") != final.get("difficulty_manual"):
            changed_difficulties += 1

        if problem_id not in confirmations and source != final:
            unadjudicated_changes.append(problem_id)

        # All fields other than the two adjudicated label fields must remain
        # unchanged for every record.
        source_other = {
            k: v for k, v in source.items()
            if k not in {"capabilities_manual", "difficulty_manual"}
        }
        final_other = {
            k: v for k, v in final.items()
            if k not in {"capabilities_manual", "difficulty_manual"}
        }

        if source_other != final_other:
            immutable_field_changes.append(problem_id)

    if unadjudicated_changes:
        raise ValueError(
            f"Unadjudicated records changed: {unadjudicated_changes[:10]}"
        )

    if immutable_field_changes:
        raise ValueError(
            "Non-label fields changed: "
            f"{immutable_field_changes[:10]}"
        )

    return {
        "changed_capability_records": changed_capabilities,
        "changed_difficulty_records": changed_difficulties,
        "unadjudicated_changes": len(unadjudicated_changes),
        "non_label_field_changes": len(immutable_field_changes),
    }

from pathlib import Path
from collections import Counter

def read_jsonl(path):
    if not path.exists():
        raise FileNotFoundError(path)

    rows = []
    seen = set()

    with path.open("r", encoding="utf-8-sig") as f:
        for line_no, line in enumerate(f, start=1):
            if not line.strip():
                continue
            row = json.loads(line)
            pid = row.get("problem_id")

            if not pid:
                raise ValueError(f"Missing problem_id: {path}:{line_no}")
            if pid in seen:
                raise ValueError(f"Duplicate ID in {path}: {pid}")

            seen.add(pid)
            rows.append(row)

    return rows


def write_jsonl(path, rows):
    with path.open("w", encoding="utf-8") as f:
        for row in rows:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")


def main():
    batch = read_jsonl(BATCH_INPUT)
    supplement = read_jsonl(SUPPLEMENT_INPUT)

    decisions = json.loads(DECISIONS_INPUT.read_text(encoding="utf-8"))

    if len(batch) != 100:
        raise ValueError(f"Expected 100 batch records, got {len(batch)}")
    if len(supplement) != 85:
        raise ValueError(f"Expected 85 supplement records, got {len(supplement)}")

    batch_ids = {r["problem_id"] for r in batch}
    supplement_ids = {r["problem_id"] for r in supplement}

    if batch_ids & supplement_ids:
        raise ValueError("Duplicate problem IDs across batch and supplement.")

    all_ids = batch_ids | supplement_ids

    missing_decision_ids = set(decisions) - all_ids
    if missing_decision_ids:
        raise ValueError(
            "Adjudication IDs not found in Gold v2: "
            + ", ".join(sorted(missing_decision_ids))
        )

    if len(decisions) != 22:
        raise ValueError(f"Expected 22 adjudication decisions, got {len(decisions)}")

    original_by_id = {
        r["problem_id"]: r for r in batch + supplement
    }

    changed_ids = set()

    def apply_decisions(rows):
        output = []

        for original in rows:
            row = dict(original)
            pid = row["problem_id"]

            if pid in decisions:
                decision = decisions[pid]
                capabilities = decision["capabilities"]
                difficulty = decision["difficulty"]

                if not capabilities:
                    raise ValueError(f"Empty capabilities for {pid}")
                if len(capabilities) != len(set(capabilities)):
                    raise ValueError(f"Duplicate capabilities for {pid}")
                if not set(capabilities).issubset(VALID_CAPABILITIES):
                    raise ValueError(f"Invalid capabilities for {pid}")
                if difficulty not in VALID_DIFFICULTIES:
                    raise ValueError(f"Invalid difficulty for {pid}")

                row["capabilities_manual"] = capabilities
                row["difficulty_manual"] = difficulty

                old_notes = row.get("notes", "")
                provenance = (
                    "[D3 AI adjudication proposal; pending human confirmation] "
                    + decision["rationale"]
                )
                row["notes"] = (
                    (old_notes.rstrip() + " " + provenance).strip()
                    if old_notes else provenance
                )

                changed_ids.add(pid)

            output.append(row)

        return output

    batch_vnext = apply_decisions(batch)
    supplement_vnext = apply_decisions(supplement)

    if changed_ids != set(decisions):
        missing = set(decisions) - changed_ids
        raise ValueError(
            "Some decisions were not applied: " + ", ".join(sorted(missing))
        )

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    write_jsonl(BATCH_OUTPUT, batch_vnext)
    write_jsonl(SUPPLEMENT_OUTPUT, supplement_vnext)

    final_ids = {
        r["problem_id"] for r in batch_vnext + supplement_vnext
    }

    if len(final_ids) != 185:
        raise ValueError(f"Expected 185 unique IDs, got {len(final_ids)}")

    # Ensure ground truth and all non-adjudicated records remain unchanged.
    for row in batch_vnext + supplement_vnext:
        pid = row["problem_id"]
        original = original_by_id[pid]

        if row.get("ground_truth") != original.get("ground_truth"):
            raise ValueError(f"Ground truth changed for {pid}")

        if pid not in decisions:
            if row.get("capabilities_manual") != original.get("capabilities_manual"):
                raise ValueError(f"Unadjudicated capabilities changed: {pid}")
            if row.get("difficulty_manual") != original.get("difficulty_manual"):
                raise ValueError(f"Unadjudicated difficulty changed: {pid}")

    capability_support = Counter()
    difficulty_support = Counter()

    for row in batch_vnext + supplement_vnext:
        labels = row.get("capabilities_manual", [])
        difficulty = row.get("difficulty_manual")

        if not isinstance(labels, list) or not labels:
            raise ValueError(f"Invalid capabilities_manual: {row['problem_id']}")
        if not set(labels).issubset(VALID_CAPABILITIES):
            raise ValueError(f"Unknown capability: {row['problem_id']}")
        if difficulty not in VALID_DIFFICULTIES:
            raise ValueError(f"Invalid difficulty: {row['problem_id']}")

        capability_support.update(labels)
        difficulty_support.update([difficulty])

    manifest = {
        "dataset_name": "Gold vNext",
        "version": "vNext",
        "records": len(final_ids),
        "batch_100_records": len(batch_vnext),
        "supplement_records": len(supplement_vnext),
        "adjudicated_records": len(changed_ids),
        "adjudication_method": "AI-assisted adjudication; pending human confirmation",
        "source_files": [
            str(BATCH_INPUT),
            str(SUPPLEMENT_INPUT)
        ],
        "adjudication_decisions": str(DECISIONS_INPUT),
        "output_files": [
            str(BATCH_OUTPUT),
            str(SUPPLEMENT_OUTPUT)
        ],
        "ground_truth_modified": False,
        "non_adjudicated_labels_modified": False,
        "capability_support": dict(capability_support),
        "difficulty_support": dict(difficulty_support)
    }

    MANIFEST_OUTPUT.write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2),

    )

    print("GOLD VNEXT BUILD: PASS")

    print("Supplement records:", len(supplement_vnext))
    print("Total unique records:", len(final_ids))
    print("Adjudicated records applied:", len(changed_ids))

def read_jsonl(path):
    rows = []
    with path.open("r", encoding="utf-8-sig") as f:
        for line_no, line in enumerate(f, 1):
            if line.strip():
                try:
                    rows.append(json.loads(line))
                except json.JSONDecodeError as exc:

def index_unique(rows, source_name):
    result = {}
    for row in rows:
        pid = row.get("problem_id")
        if not pid:
            raise ValueError(f"Missing problem_id in {source_name}")
        if pid in result:
            raise ValueError(f"Duplicate problem_id in {source_name}: {pid}")
        result[pid] = row
    return result

        if old is None:
            continue

        if old.get("ground_truth") != new.get("ground_truth"):
            ground_truth_changes.append(pid)

        if old.get("capabilities_manual") != new.get("capabilities_manual"):
            changed_capabilities.append(pid)

        if old.get("difficulty_manual") != new.get("difficulty_manual"):
            changed_difficulties.append(pid)

        if pid not in adjudicated_ids:
            if (
                old.get("capabilities_manual") != new.get("capabilities_manual")
                or old.get("difficulty_manual") != new.get("difficulty_manual")
            ):
                unadjudicated_changes.append(pid)

        labels = new.get("capabilities_manual")
        difficulty = new.get("difficulty_manual")

        if not isinstance(labels, list) or not labels:
            errors.append(f"Invalid/empty capabilities_manual: {pid}")
        elif not set(labels).issubset(VALID_CAPABILITIES):
            errors.append(f"Unknown capability label: {pid}")

        if difficulty not in VALID_DIFFICULTIES:
            errors.append(f"Invalid difficulty: {pid}")

    checks["ground_truth_unchanged"] = len(ground_truth_changes) == 0
    checks["no_unadjudicated_label_changes"] = len(unadjudicated_changes) == 0
    checks["all_labels_valid"] = not any(
        "capabilities_manual" in e or "capability label" in e for e in errors
    )
    checks["all_difficulties_valid"] = not any(
        "difficulty" in e for e in errors
    )
    checks["all_adjudicated_ids_present"] = adjudicated_ids.issubset(output)

    report = {
        "dataset": "Gold vNext",
        "status": "PASS" if all(checks.values()) and not errors else "FAIL",
        "checks": checks,
        "source_record_count": len(source),
        "output_record_count": len(output),
        "adjudication_decision_count": len(adjudicated_ids),
        "changed_capability_ids": sorted(changed_capabilities),
        "changed_difficulty_ids": sorted(changed_difficulties),
        "ground_truth_changed_ids": sorted(ground_truth_changes),
        "unadjudicated_label_changed_ids": sorted(unadjudicated_changes),
        "errors": errors
    }

    REPORT_PATH.parent.mkdir(parents=True, exist_ok=True)
    REPORT_PATH.write_text(
        json.dumps(report, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )

    print("GOLD VNEXT VALIDATION:", report["status"])
    for name, passed in checks.items():
        print(f"{name}: {'PASS' if passed else 'FAIL'}")
    print("Source records:", len(source))
    print("Output records:", len(output))
    print("Changed capability records:", len(changed_capabilities))
    print("Changed difficulty records:", len(changed_difficulties))
    print("Ground truth changes:", len(ground_truth_changes))
    print("Unadjudicated label changes:", len(unadjudicated_changes))
    print("Report:", REPORT_PATH)

    if errors:
        for error in errors:
            print("ERROR:", error)

    if report["status"] != "PASS":
        raise SystemExit(1)


if __name__ == "__main__":
    main()

def read_jsonl(path):
    if not path.is_file():
        raise FileNotFoundError(path)

    rows = []
    with path.open("r", encoding="utf-8-sig") as f:
        for line_number, line in enumerate(f, start=1):
            if not line.strip():
                continue
            try:
                rows.append(json.loads(line))
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSONL at {path}:{line_number}: {exc}"
                )
    return rows


def read_csv(path):
    if not path.is_file():
        raise FileNotFoundError(path)

    with path.open("r", encoding="utf-8-sig", newline="") as f:
        return list(csv.DictReader(f))


def read_json(path):
    if not path.is_file():
        raise FileNotFoundError(path)
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha256(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def unique_ids(rows, name):
    ids = [row["problem_id"] for row in rows]

    if len(ids) != len(set(ids)):
        raise ValueError(f"{name}: duplicate problem_id detected.")

    return set(ids)


def check(condition, name, detail, results):
    results.append({

        "detail": str(detail),

def read_jsonl(path: Path) -> list[dict]:
    if not path.is_file():
        raise FileNotFoundError(path)

    rows = []
    with path.open("r", encoding="utf-8-sig") as f:
        for line_no, line in enumerate(f, start=1):
            if not line.strip():
                continue
            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(f"Invalid JSONL: {path}:{line_no}") from exc
            if not isinstance(row, dict):
                raise ValueError(f"Expected JSON object: {path}:{line_no}")
            rows.append(row)
    return rows


def read_csv(path: Path) -> list[dict]:
    if not path.is_file():
        raise FileNotFoundError(path)
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        return list(csv.DictReader(f))


def read_json(path: Path) -> dict:
    if not path.is_file():
        raise FileNotFoundError(path)
    return json.loads(path.read_text(encoding="utf-8-sig"))


def write_jsonl(path: Path, rows: list[dict]) -> None:
    with path.open("w", encoding="utf-8", newline="\n") as f:
        for row in rows:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def index_by_id(rows: list[dict], label: str) -> dict[str, dict]:
    result = {}
    for row in rows:
        pid = row.get("problem_id")
        if not pid:
            raise ValueError(f"{label}: record without problem_id")
        if pid in result:
            raise ValueError(f"{label}: duplicate problem_id {pid}")
        result[pid] = row
    return result


def parse_labels(value, pid: str) -> list[str]:
    if isinstance(value, list):
        labels = value
    elif isinstance(value, str):
        labels = json.loads(value)
    else:
        raise ValueError(f"Invalid capabilities in worklist for {pid}")

    if not isinstance(labels, list) or not labels:
        raise ValueError(f"Empty/invalid capabilities for {pid}")

    labels = [str(x).strip() for x in labels]
    if len(labels) != len(set(labels)):
        raise ValueError(f"Duplicate capabilities for {pid}")

    invalid = set(labels) - VALID_CAPABILITIES
    if invalid:
        raise ValueError(f"Invalid capabilities for {pid}: {sorted(invalid)}")

    return labels


def check(condition: bool, name: str, detail, checks: list) -> None:
    checks.append({
        "name": name,
        "passed": bool(condition),
        "detail": detail,
    })


def append_note(old: str, new: str, mode: str) -> str:
    if mode == "replace":
        return new
    old = str(old or "").strip()
    return f"{old}\n[{new}]" if old else new


def write_csv(path: Path, rows: list[dict]) -> None:
    if not rows:
        raise ValueError(f"Refusing to write empty CSV: {path}")
    fields = list(rows[0].keys())
    with path.open("w", encoding="utf-8-sig", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)

            "answer_decision": answer_decision,
            "source_difficulty": source_difficulty,
            "final_difficulty": final_difficulty,
            "difficulty_status": difficulty_status,
            "ambiguity_flag": str(ambiguity).lower(),
            "ambiguity_status": ambiguity_status,
            "decision_reason": decision_reason,
            "source_question": source_question,
            "source_annotation_notes": source_notes,
            "final_annotation_notes": final_notes,
            "reviewer": "User-confirmed / assistant-reviewed",
            "review_date": "2026-11-16",
        }
        ledger.append(ledger_row)

        if pid in DECISIONS:
            disagreement.append({
                "problem_id": pid,
                "split": worklist_by_id[pid].get("split", "D_dev"),
                "source_capabilities": json.dumps(
                    source_labels, ensure_ascii=False
                ),
                "final_capabilities": json.dumps(
                    final_labels, ensure_ascii=False
                ),
                "capability_changed": str(
                    source_labels != final_labels
                ).lower(),
                "source_difficulty": source_difficulty,
                "final_difficulty": final_difficulty,
                "ambiguity_flag": str(ambiguity).lower(),
                "answer_status": answer_status,
                "source_ground_truth": source_ground_truth,
                "answer_decision": answer_decision,
                "decision_reason": decision_reason,
            })

        candidate_supplement.append(row)

    ledger_by_id = index_by_id(ledger, "adjudication ledger")
    check(
        len(ledger) == 63 and set(ledger_by_id) == worklist_ids,
        "ledger_covers_all_63",
        {"rows": len(ledger), "ids": len(ledger_by_id)},
        checks,
    )

    changed_label_ids = {
        row["problem_id"]
        for row in ledger
        if row["capability_changed"] == "true"
    }
    ambiguous_ids = {
        row["problem_id"]
        for row in ledger
        if row["ambiguity_flag"] == "true"
    }

    # Candidate supplement must preserve exactly 85 records.
    candidate_supplement_by_id = index_by_id(
        candidate_supplement, "candidate supplement"
    )
    check(
        len(candidate_supplement) == 85,
        "candidate_supplement_count",
        len(candidate_supplement),
        checks,
    )

    # Prepare a staging folder inside the repository. Never use TEMP/AppData.
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    STAGING_DIR.mkdir(parents=False, exist_ok=False)

    try:
        staged_batch = STAGING_DIR / BATCH_NAME
        staged_supplement = STAGING_DIR / SUPPLEMENT_NAME
        staged_manifest = STAGING_DIR / MANIFEST_NAME

        # Batch 100 is copied byte-for-byte, not reserialized.
        shutil.copyfile(PATHS["batch"], staged_batch)
        write_jsonl(staged_supplement, candidate_supplement)

        candidate_batch = read_jsonl(staged_batch)
        candidate_supp = read_jsonl(staged_supplement)
        candidate_batch_by_id = index_by_id(
            candidate_batch, "candidate batch"
        )
        candidate_supp_by_id = index_by_id(
            candidate_supp, "candidate supplement"
        )

        post_checks = []

        check(
            len(candidate_batch) == 100,
            "candidate_batch_count",
            len(candidate_batch),
            post_checks,
        )
        check(
            len(candidate_supp) == 85,
            "candidate_supplement_count",
            len(candidate_supp),
            post_checks,
        )
        check(
            len(set(candidate_batch_by_id) | set(candidate_supp_by_id)) == 185,
            "candidate_total_unique",
            len(set(candidate_batch_by_id) | set(candidate_supp_by_id)),
            post_checks,
        )
        check(
            set(candidate_batch_by_id) == batch_ids,
            "candidate_batch_inventory_unchanged",
            True,
            post_checks,
        )
        check(
            set(candidate_supp_by_id) == supplement_ids,
            "candidate_supplement_inventory_unchanged",
            True,
            post_checks,
        )
        check(
            sha256(staged_batch) == sha256(PATHS["batch"]),
            "batch_byte_identical",
            sha256(staged_batch),
            post_checks,
        )

        # Only the 22 explicitly decided pending IDs may differ.
        unexpected_changes = []
        for pid, old in supplement_by_id.items():
            new = candidate_supp_by_id[pid]

            if pid in DECISIONS:
                allowed_old = {
                    k: v for k, v in old.items()
                    if k not in {"capabilities_manual", "notes"}
                }
                allowed_new = {
                    k: v for k, v in new.items()
                    if k not in {"capabilities_manual", "notes"}
                }
                if allowed_old != allowed_new:
                    unexpected_changes.append(pid)
            elif old != new:
                unexpected_changes.append(pid)

        check(
            not unexpected_changes,
            "only_explicit_decisions_changed",
            unexpected_changes,
            post_checks,
        )

        human_changed = []
        for pid in human_ids:
            if supplement_by_id[pid] != candidate_supp_by_id[pid]:
                human_changed.append(pid)
        check(
            not human_changed,
            "human_confirmed_22_unchanged",
            human_changed,
            post_checks,
        )

        invalid_records = []
        for source_name, records in [
            ("batch", candidate_batch),
            ("supplement", candidate_supp),
        ]:
            for record in records:
                pid = record["problem_id"]
                if set(record.keys()) != EXPECTED_FIELDS:
                    invalid_records.append(f"{source_name}:{pid}:fields")
                    continue
                labels = record["capabilities_manual"]
                if (
                    not isinstance(labels, list)
                    or not labels
                    or not set(labels) <= VALID_CAPABILITIES
                ):
                    invalid_records.append(f"{source_name}:{pid}:capabilities")
                if record["difficulty_manual"] not in VALID_DIFFICULTIES:
                    invalid_records.append(f"{source_name}:{pid}:difficulty")
                if record["schema_version"] != "v1":
                    invalid_records.append(f"{source_name}:{pid}:schema")

        check(
            not invalid_records,
            "candidate_schema_and_labels",
            invalid_records[:20],
            post_checks,
        )

        check(
            len(ledger) == 63,
            "ledger_record_count",
            len(ledger),
            post_checks,
        )
        check(
            len(DECISIONS) == 22 and len(no_change_ids) == 41,
            "review_decision_partition",
            {"user_confirmed": len(DECISIONS), "no_change": len(no_change_ids)},
            post_checks,
        )

        failed_postchecks = [
            item for item in post_checks if not item["passed"]
        ]

        validation = {
            "review_date": "2026-11-16",
            "status": "PASS" if not failed_postchecks else "FAIL",
            "freeze_status": "NOT_FROZEN",
            "candidate_status": (
                "VALIDATED_PENDING_USER_APPROVAL"
                if not failed_postchecks else "VALIDATION_FAILED"
            ),
            "counts": {
                "batch": len(candidate_batch),
                "supplement": len(candidate_supp),
                "total": len(candidate_batch) + len(candidate_supp),
                "reviewed_pending": len(ledger),
                "user_confirmed_decisions": len(DECISIONS),
                "reviewed_no_change": len(no_change_ids),
                "capability_changed_records": len(changed_label_ids),
                "ambiguity_flagged_records": len(ambiguous_ids),
                "human_confirmed_preserved": len(human_ids),
            },
            "changed_capability_ids": sorted(changed_label_ids),
            "ambiguity_ids": sorted(ambiguous_ids),
            "source_hashes_before_build": source_hashes,
            "checks": checks + post_checks,
            "failed_checks": failed_postchecks,
            "gold_vfinal_modified": False,
            "raw_ground_truth_modified": False,
            "human_confirmed_modified": False,
        }

        if failed_postchecks:
            (REPORT_DIR / "gold_review_63_validation_report.json").write_text(
                json.dumps(validation, ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
            raise RuntimeError(
                "Candidate validation failed. Staging folder was retained "
                "for inspection; it was NOT promoted to candidate."
            )

        # Create candidate manifest only after all validations pass.
        manifest = {
            "dataset_name": "Gold vNext candidate",
            "version": "vNext-candidate-2026-11-16",
            "candidate_status": "VALIDATED_PENDING_USER_APPROVAL",
            "freeze_status": "NOT_FROZEN",
            "created_date": "2026-11-16",
            "records": 185,
            "batch_100_records": 100,
            "supplement_records": 85,
            "D_pool_records": 100,
            "D_dev_records": 85,
            "reviewed_pending_records": 63,
            "user_confirmed_decisions": 22,
            "reviewed_no_change_records": 41,
            "capability_changed_records": len(changed_label_ids),
            "ambiguity_flagged_records": len(ambiguous_ids),
            "human_confirmed_records_preserved": 22,
            "source_files": {
                "batch": str(PATHS["batch"].relative_to(BASE)),
                "supplement": str(PATHS["supplement"].relative_to(BASE)),
                "raw_train": str(PATHS["raw_train"].relative_to(BASE)),
                "supplement_raw": str(
                    PATHS["supplement_raw"].relative_to(BASE)
                ),
            },
            "output_files": {
                "batch": f"data/{CANDIDATE_NAME}/{BATCH_NAME}",
                "supplement": f"data/{CANDIDATE_NAME}/{SUPPLEMENT_NAME}",
            },
            "source_sha256": source_hashes,
            "candidate_sha256": {
                "batch": sha256(staged_batch),
                "supplement": sha256(staged_supplement),
            },
            "ground_truth_modified": False,
            "difficulty_modified": False,
            "gold_vfinal_modified": False,
            "human_confirmed_records_modified": False,
            "validation_status": "PASS",
            "approval_required_before_freeze": True,
        }

        staged_manifest.write_text(
            json.dumps(manifest, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

        # Verify original source hashes again before promotion.
        source_hashes_after = {
            name: sha256(path)
            for name, (_, path) in source_hash_keys.items()
        }
        source_unchanged = all(
            source_hashes_after[name] == values["actual_sha256"]
            for name, values in source_hashes.items()
        )
        if not source_unchanged:
            raise RuntimeError(
                "A source file changed during build. Candidate not promoted."
            )

        # Atomic rename within the same repository filesystem.
        os.replace(STAGING_DIR, CANDIDATE_DIR)

        final_batch = CANDIDATE_DIR / BATCH_NAME
        final_supplement = CANDIDATE_DIR / SUPPLEMENT_NAME
        final_manifest = CANDIDATE_DIR / MANIFEST_NAME

        # Re-check final candidate paths after promotion.
        final_checks = [
            {
                "name": "candidate_directory_promoted",
                "passed": CANDIDATE_DIR.is_dir(),
                "detail": str(CANDIDATE_DIR),
            },
            {
                "name": "candidate_files_exist",
                "passed": all(
                    p.is_file()
                    for p in [final_batch, final_supplement, final_manifest]
                ),
                "detail": "batch, supplement, manifest",
            },
            {
                "name": "source_hashes_unchanged_after_build",
                "passed": source_unchanged,
                "detail": source_hashes_after,
            },
        ]

        final_pass = all(item["passed"] for item in final_checks)
        validation["checks"].extend(final_checks)
        validation["status"] = "PASS" if final_pass else "FAIL"
        validation["candidate_status"] = (
            "VALIDATED_PENDING_USER_APPROVAL"
            if final_pass else "POST_PROMOTION_CHECK_FAILED"
        )
        validation["source_hashes_after_build"] = source_hashes_after
        validation["candidate_sha256_final"] = {
            "batch": sha256(final_batch),
            "supplement": sha256(final_supplement),
            "manifest": sha256(final_manifest),
        }

        # Reports.
        REPORT_DIR.mkdir(parents=True, exist_ok=True)

        ledger_path = REPORT_DIR / "user_adjudication_2026-11-16.csv"
        disagreement_path = (
            REPORT_DIR / "gold_review_63_disagreement_report.csv"
        )
        final_report_path = (
            REPORT_DIR / "gold_review_63_final_adjudication.md"
        )
        validation_json_path = (
            REPORT_DIR / "gold_review_63_validation_report.json"
        )
        validation_md_path = (
            REPORT_DIR / "gold_review_63_validation_report.md"
        )

        write_csv(ledger_path, ledger)
        write_csv(disagreement_path, disagreement)

        decision_status_counts = Counter(
            row["review_status"] for row in ledger
        )
        ambiguity_status_counts = Counter(
            row["ambiguity_status"] for row in ledger
        )

        final_lines = [
            "# Gold Review — Final Adjudication 2026-11-16",
            "",
            "## Review scope",
            "",
            f"- Pending worklist records: **{len(worklist)}**",
            f"- User-confirmed decisions: **{len(DECISIONS)}**",
            f"- Reviewed with no annotation change: **{len(no_change_ids)}**",
            f"- Total reviewed: **{len(ledger)}**",
            f"- Capability-changed records: **{len(changed_label_ids)}**",
            f"- Ambiguity-flagged records: **{len(ambiguous_ids)}**",
            "- Difficulty changes: **0**",
            "- Raw ground-truth changes: **0**",
            "- Gold vFinal modifications: **0**",
            "",
            "## Review status",
            "",
        ]

        for key, value in sorted(decision_status_counts.items()):
            final_lines.append(f"- `{key}`: {value}")

        final_lines += ["", "## Ambiguity status", ""]
        for key, value in sorted(ambiguity_status_counts.items()):
            final_lines.append(f"- `{key}`: {value}")

        final_lines += [
            "",
            "## Explicitly retained assumptions and source caveats",
            "",
            "- `gsm8k_train_00239`: 5% overlap is interpreted as "
            "5 percentage points of the total germ population; answer 30%.",
            "- `gsm8k_train_00713`: target answer 15% is retained under the "
            "assumptions that infected mosquito groups are disjoint and "
            "the vaccine reduction applies to both viruses, despite the "
            "malaria-vaccine wording.",
            "- `gsm8k_train_00830`: the current raw dataset record has "
            "ground truth 25 for total meal cost. The separately supplied "
            "variant asks for the lassi price and derives $2.50. The two "
            "targets are not interchangeable.",
            "- `gsm8k_train_01200`: interpretation A is accepted.",
            "- `gsm8k_train_03048`: the three book categories are assumed "
            "mutually exclusive.",
            "",
            "## Candidate",
            "",
            f"- Candidate directory: `{CANDIDATE_DIR}`",
            "- Candidate validation: **PASS**",
            "- Freeze status: **NOT FROZEN**",
            "- User approval required before release/freeze.",
            "",
        ]
        final_report_path.write_text(
            "\n".join(final_lines), encoding="utf-8"
        )

        validation_json_path.write_text(
            json.dumps(validation, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

        validation_lines = [
            "# Gold vNext Candidate Validation — 2026-11-16",
            "",
            f"- Status: **{validation['status']}**",
            f"- Candidate status: **{validation['candidate_status']}**",
            "- Freeze status: **NOT FROZEN**",
            "",
            "## Counts",
            "",
        ]
        for key, value in validation["counts"].items():
            validation_lines.append(f"- {key}: **{value}**")

        validation_lines += ["", "## Checks", ""]
        for item in validation["checks"]:
            status = "PASS" if item["passed"] else "FAIL"
            validation_lines.append(
                f"- **{status}** `{item['name']}`: {item['detail']}"
            )

        validation_lines += [
            "",
            "## Source protection",
            "",
            "- Gold vFinal source files unchanged.",
            "- Raw source files unchanged.",
            "- Human-confirmed records unchanged.",
            "- Batch 100 copied byte-for-byte.",
            "- No ground-truth or difficulty field was modified.",
            "",
        ]
        validation_md_path.write_text(
            "\n".join(validation_lines), encoding="utf-8"
        )

        print()
        print("GOLD VNEXT CANDIDATE BUILD:", validation["status"])
        print("Batch records:", len(candidate_batch))
        print("Supplement records:", len(candidate_supp))
        print("Total records:", len(candidate_batch) + len(candidate_supp))
        print("Pending records reviewed:", len(ledger))
        print("User-confirmed decisions:", len(DECISIONS))
        print("No-change reviewed records:", len(no_change_ids))
        print("Capability-changed records:", len(changed_label_ids))
        print("Ambiguity-flagged records:", len(ambiguous_ids))
        print("Human-confirmed records preserved:", len(human_ids))
        print("Gold vFinal modified: False")
        print("Raw ground truth modified: False")
        print("Freeze status: NOT FROZEN")
        print("Candidate:", CANDIDATE_DIR)
        print("Ledger:", ledger_path)
        print("Disagreement report:", disagreement_path)
        print("Final adjudication report:", final_report_path)
        print("Validation report:", validation_json_path)
        print("Validation Markdown:", validation_md_path)

        if not final_pass:
            raise SystemExit(1)

    except Exception:
        # Keep any failed staging directory inside the repository for audit.
        print("BUILD FAILED. Candidate was not approved or frozen.")
        if STAGING_DIR.exists():
            print("Staging retained for inspection:", STAGING_DIR)
        raise


if __name__ == "__main__":
    main()
